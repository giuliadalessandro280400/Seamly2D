99// @brief VToolSplinePath constructor.
100// @param doc dom document container.
101// @param data container with variables.
102// @param id object id in container.
103// @param typeCreation way we create this tool.
104// @param parent parent object.
105VToolSplinePath::VToolSplinePath(VAbstractPattern *doc, VContainer *data, quint32 id, const Source &typeCreation,
106 const QVector<qreal> &refChords, QGraphicsItem *parent)
107 : VAbstractSpline(doc, data, id, parent)
108 , oldPosition()
109 , splIndex(-1)
110 , m_refChords(refChords)
111{
112 m_sceneType = SceneObject::SplinePath;
113
114 this->setFlag(QGraphicsItem::ItemIsMovable, true);
115 this->setFlag(QGraphicsItem::ItemIsFocusable, true);// For keyboard input focus
116
117 const QSharedPointer<VSplinePath> splPath = data->GeometricObject<VSplinePath>(id);
118 for (qint32 i = 1; i<=splPath->CountSubSpl(); ++i)
119 {
120 const VSpline spl = splPath->GetSpline(i);
121
122 const bool freeAngle1 = qmu::QmuTokenParser::IsSingle(spl.GetStartAngleFormula());
123 const bool freeLength1 = qmu::QmuTokenParser::IsSingle(spl.GetC1LengthFormula());
124
125 auto *controlPoint = new VControlPointSpline(i, SplinePointPosition::FirstPoint,
126 static_cast<QPointF>(spl.GetP2()),
127 freeAngle1, freeLength1, this);
128 connect(controlPoint, &VControlPointSpline::controlPointPositionChanged, this,
129 &VToolSplinePath::controlPointPositionChanged);
130 connect(this, &VToolSplinePath::setEnabledPoint, controlPoint, &VControlPointSpline::setEnabledPoint);
131 connect(controlPoint, &VControlPointSpline::showContextMenu, this, &VToolSplinePath::contextMenuEvent);
132 m_controlPoints.append(controlPoint);
133
134 const bool freeAngle2 = qmu::QmuTokenParser::IsSingle(spl.GetEndAngleFormula());
135 const bool freeLength2 = qmu::QmuTokenParser::IsSingle(spl.GetC2LengthFormula());
136
137 controlPoint = new VControlPointSpline(i, SplinePointPosition::LastPoint, static_cast<QPointF>(spl.GetP3()),
138 freeAngle2, freeLength2, this);
139 connect(controlPoint, &VControlPointSpline::controlPointPositionChanged, this,
140 &VToolSplinePath::controlPointPositionChanged);
141 connect(this, &VToolSplinePath::setEnabledPoint, controlPoint, &VControlPointSpline::setEnabledPoint);
142 connect(controlPoint, &VControlPointSpline::showContextMenu, this, &VToolSplinePath::contextMenuEvent);
143 m_controlPoints.append(controlPoint);
144 }
145
146 VToolSplinePath::refreshCtrlPoints();
147 showHandles(m_piecesMode);
148 ToolCreation(typeCreation);
149}

166// @param dialog dialog.
167// @param scene pointer to scene.
168// @param doc dom document container.
169// @param data container with variables.
170VToolSplinePath* VToolSplinePath::Create(QSharedPointer<DialogTool> dialog, VMainGraphicsScene *scene,
171 VAbstractPattern *doc, VContainer *data)
172{
173 SCASSERT(!dialog.isNull())
174 QSharedPointer<DialogSplinePath> dialogTool = dialog.objectCast<DialogSplinePath>();
175 SCASSERT(!dialogTool.isNull())
176 VSplinePath *path = new VSplinePath(dialogTool->GetPath());
177 for (qint32 i = 0; i < path->CountPoints(); ++i)
178 {
179 doc->IncrementReferens((*path)[i].P().getIdTool());
180 }
181
182 path->setLineColor(dialogTool->getLineColor());
183 path->SetPenStyle(dialogTool->getPenStyle());
184 path->setLineWeight(dialogTool->getLineWeight());
185
186 // Proportional handles: the current chords become the reference, so the curve looks exactly as drawn now.
187 const QVector<qreal> refChords = dialogTool->getProportionalHandles() ? segmentChords(*path) : QVector<qreal>();
188
189 VToolSplinePath* spl = Create(0, path, scene, doc, data, Document::FullParse, Source::FromGui, refChords);
190 if (spl != nullptr)
191 {
192 spl->m_dialog = dialogTool;
193 }
194 return spl;
195}

197// @brief Create help create tool.
198// @param _id tool id, 0 if tool doesn't exist yet.
199// @param path spline path.
200// @param scene pointer to scene.
201// @param doc dom document container.
202// @param data container with variables.
203// @param parse parser file mode.
204// @param typeCreation way we create this tool.
205VToolSplinePath* VToolSplinePath::Create(const quint32 _id, VSplinePath *path, VMainGraphicsScene *scene,
206 VAbstractPattern *doc, VContainer *data, const Document &parse,
207 const Source &typeCreation, const QVector<qreal> &refChords)
208{
209 quint32 id = _id;
210
211 if (typeCreation == Source::FromGui)
212 {
213 id = data->AddGObject(path);
214 data->AddCurveWithSegments(data->GeometricObject<VAbstractCubicBezierPath>(id), id);
215 }
216 else
217 {
218 data->UpdateGObject(id, path);
219 data->AddCurveWithSegments(data->GeometricObject<VAbstractCubicBezierPath>(id), id);
220 if (parse != Document::FullParse)
221 {
222 doc->UpdateToolData(id, data);
223 }
224 }
225
226 if (parse == Document::FullParse)
227 {
228 VDrawTool::AddRecord(id, Tool::SplinePath, doc);
229 VToolSplinePath *spl = new VToolSplinePath(doc, data, id, typeCreation, refChords);
230 scene->addItem(spl);
231 initSplinePathToolConnections(scene, spl);
232 VAbstractPattern::AddTool(id, spl);
233 return spl;
234 }
235 return nullptr;
236}

238//---------------------------------------------------------------------------------------------------------------------
239VToolSplinePath *VToolSplinePath::Create(const quint32 _id, const QVector<quint32> &points, QVector<QString> &a1,
240 QVector<QString> &a2, QVector<QString> &l1, QVector<QString> &l2,
241 const QString &color, const QString &penStyle, const QString &lineWeight,
242 quint32 duplicate, VMainGraphicsScene *scene, VAbstractPattern *doc,
243 VContainer *data, const Document &parse, const Source &typeCreation,
244 const QVector<qreal> &refChords)
245{
246 auto path = new VSplinePath();
247
248 if (duplicate > 0)
249 {
250 path->SetDuplicate(duplicate);
251 }
252
253 // Proportional handles: a handle belongs to the segment it points into. Its length follows the chord of that
254 // segment (factor 1 when the option is off).
255 const bool proportional = refChords.size() == points.size() - 1;
256 auto segmentFactor = [&](int segment)
257 {
258 if (not proportional || segment < 0 || segment >= refChords.size())
259 {
260 return 1.0;
261 }
262 const QPointF a = static_cast<QPointF>(*data->GeometricObject<VPointF>(points.at(segment)));
263 const QPointF b = static_cast<QPointF>(*data->GeometricObject<VPointF>(points.at(segment + 1)));
264 return ProportionalHandles::factor(ProportionalHandles::chord(a, b), refChords.at(segment));
265 };
266
267 for (int i = 0; i < points.size(); ++i)
268 {
269 const qreal calcAngle1 = CheckFormula(_id, a1[i], data);
270 const qreal calcAngle2 = CheckFormula(_id, a2[i], data);
271
272 const qreal calcLength1 = qApp->toPixel(CheckFormula(_id, l1[i], data)) * segmentFactor(i - 1);
273 const qreal calcLength2 = qApp->toPixel(CheckFormula(_id, l2[i], data)) * segmentFactor(i);
274
275 const auto p = *data->GeometricObject<VPointF>(points.at(i));
276
277 path->append(VSplinePoint(p, calcAngle1, a1.at(i), calcAngle2, a2.at(i), calcLength1, l1.at(i), calcLength2,
278 l2.at(i)));
279 }
280
281 path->setLineColor(color);
282 path->SetPenStyle(penStyle);
283 path->setLineWeight(lineWeight);
284
285 return VToolSplinePath::Create(_id, path, scene, doc, data, parse, typeCreation, refChords);
286}

295/**
296 * @brief segmentChords returns the chord (distance between the end points) of every segment, in pattern units.
297 */
298QVector<qreal> VToolSplinePath::segmentChords(const VSplinePath &path)
299{
300 QVector<qreal> chords;
301 for (int i = 0; i + 1 < path.CountPoints(); ++i)
302 {
303 chords.append(ProportionalHandles::chord(static_cast<QPointF>(path.at(i).P()),
304 static_cast<QPointF>(path.at(i + 1).P())));
305 }
306 return chords;
307}

310/**
311 * @brief toStoredLengths converts the actual handle lengths of a path to the values stored for the reference chords.
312 * Only plain numbers are rewritten, formulas are kept.
313 */
314VSplinePath VToolSplinePath::toStoredLengths(const VSplinePath &path, const QVector<qreal> &refChords)
315{
316 VSplinePath result = path;
317 const QVector<qreal> chords = segmentChords(path);
318 if (refChords.size() != chords.size())
319 {
320 return result;
321 }
322
323 for (int i = 0; i < result.CountPoints(); ++i)
324 {
325 VSplinePoint point = result.at(i);
326 if (i > 0 && ProportionalHandles::isNumber(point.Length1Formula()))
327 {
328 point.SetLength1(point.Length1(), ProportionalHandles::toStored(qApp->fromPixel(point.Length1()),
329 chords.at(i - 1), refChords.at(i - 1)));
330 }
331 if (i + 1 < result.CountPoints() && ProportionalHandles::isNumber(point.Length2Formula()))
332 {
333 point.SetLength2(point.Length2(), ProportionalHandles::toStored(qApp->fromPixel(point.Length2()),
334 chords.at(i), refChords.at(i)));
335 }
336 result[i] = point;
337 }
338 return result;
339}

376// @brief updateControlPoints update position points control points in file.
377// @param spl spline what was changed.
378// @param splPath spline path.
379// @param splineIndex index spline in spline path.
380void VToolSplinePath::updateControlPoints(const VSpline &spl, VSplinePath &splPath, const qint32 &splineIndex) const
381{
382 VSplinePoint point = splPath.GetSplinePoint(splineIndex, SplinePointPosition::FirstPoint);
383 point.SetAngle2(spl.GetStartAngle(), spl.GetStartAngleFormula());
384 point.SetLength2(spl.GetC1Length(), spl.GetC1LengthFormula());
385 splPath.UpdatePoint(splineIndex, SplinePointPosition::FirstPoint, point);
386
387 point = splPath.GetSplinePoint(splineIndex, SplinePointPosition::LastPoint);
388 point.SetAngle1(spl.GetEndAngle(), spl.GetEndAngleFormula());
389 point.SetLength1(spl.GetC2Length(), spl.GetC2LengthFormula());
390 splPath.UpdatePoint(splineIndex, SplinePointPosition::LastPoint, point);
391}

522// @brief SaveDialog save options into file after change in dialog.
523void VToolSplinePath::SaveDialog(QDomElement &domElement)
524{
525 SCASSERT(!m_dialog.isNull())
526 QSharedPointer<DialogSplinePath> dialogTool = m_dialog.objectCast<DialogSplinePath>();
527 SCASSERT(!dialogTool.isNull())
528
529 VSplinePath splPath = dialogTool->GetPath();
530
531 // Proportional handles on/off. The lengths in the dialog are the stored (reference) values.
532 const QVector<qreal> currentChords = segmentChords(splPath);
533 if (dialogTool->getProportionalHandles())
534 {
535 if (m_refChords.size() != currentChords.size())
536 {
537 m_refChords = currentChords; // switched on (or points changed): keep the current shape in this size
538 }
539 }
540 else if (not m_refChords.isEmpty())
541 {
542 // Switched off: store the actual lengths of the current size so the shape does not change.
543 if (m_refChords.size() == currentChords.size())
544 {
545 for (int i = 0; i < splPath.CountPoints(); ++i)
546 {
547 VSplinePoint point = splPath.at(i);
548 if (i > 0 && ProportionalHandles::isNumber(point.Length1Formula()))
549 {
550 const qreal f = ProportionalHandles::factor(currentChords.at(i - 1), m_refChords.at(i - 1));
551 point.SetLength1(point.Length1(), QString::number(point.Length1Formula().toDouble() * f));
552 }
553 if (i + 1 < splPath.CountPoints() && ProportionalHandles::isNumber(point.Length2Formula()))
554 {
555 const qreal f = ProportionalHandles::factor(currentChords.at(i), m_refChords.at(i));
556 point.SetLength2(point.Length2(), QString::number(point.Length2Formula().toDouble() * f));
557 }
558 splPath[i] = point;
559 }
560 }
561 m_refChords.clear();
562 }
563
564 for (qint32 i = 1; i <= splPath.CountSubSpl(); ++i)
565 {
566 VSpline spl = splPath.GetSpline(i);
567 qint32 j = i*2;
568
569 m_controlPoints[j-2]->blockSignals(true);
570 m_controlPoints[j-1]->blockSignals(true);
571
572 m_controlPoints[j-2]->setPos(static_cast<QPointF>(spl.GetP2()));
573 m_controlPoints[j-1]->setPos(static_cast<QPointF>(spl.GetP3()));
574
575 m_controlPoints[j-2]->blockSignals(false);
576 m_controlPoints[j-1]->blockSignals(false);
577 }
578
579 doc->SetAttribute(domElement, AttrColor, dialogTool->getLineColor());
580 doc->SetAttribute(domElement, AttrPenStyle, dialogTool->getPenStyle());
581 doc->SetAttribute(domElement, AttrLineWeight, dialogTool->getLineWeight());
582 SetSplinePathAttributes(domElement, splPath);
583}

596// @brief mousePressEvent handle mouse press events.

598void VToolSplinePath::mousePressEvent(QGraphicsSceneMouseEvent *event)
599{
600 // Ignore middle button events used for panning.
601 if (event->button() == Qt::MiddleButton)
602 {
603 return;
604 }
605 if (flags() & QGraphicsItem::ItemIsMovable)
606 {
607 if (event->button() == Qt::LeftButton && event->type() != QEvent::GraphicsSceneMouseDoubleClick)
608 {
609 oldPosition = event->scenePos();
610 const auto splPath = VAbstractTool::data.GeometricObject<VSplinePath>(m_id);
611 splIndex = splPath->Segment(oldPosition);
612 if (IsMovable(splIndex))
613 {
614 SetItemOverrideCursor(this, cursorArrowCloseHand, 1, 1);
615 event->accept();
616 }
617 }
618 }

620}

632 {

642// @brief mouseMoveEvent handle mouse move events.
643// @param event mouse move event.
644void VToolSplinePath::mouseMoveEvent(QGraphicsSceneMouseEvent *event)
645{
646 // Ignore middle button events used for panning.
647 if (event->button() == Qt::MiddleButton)
648 {
649 return;
650 }
651 // Don't need to check if left mouse button was pressed. According to the Qt documentation "If you do receive this
652 // event, you can be certain that this item also received a mouse press event, and that this item is the current
653 // mouse grabber.".
654
655 if (IsMovable(splIndex))
656 {
657 VSplinePath oldSplPath = *VAbstractTool::data.GeometricObject<VSplinePath>(m_id);
658 VSplinePath newSplPath = oldSplPath;
659
660 VSpline spline = newSplPath.GetSpline(splIndex);
661 const qreal t = spline.ParamT(oldPosition);
662
663 if (qFloor(t) == -1)
664 {
665 return;
666 }
667
668 // Magic Bezier Drag Equations follow!
669 // "weight" describes how the influence of the drag should be distributed
670 // among the handles; 0 = front handle only, 1 = back handle only.
671 double weight;
672 if (t <= 1.0 / 6.0)
673 {
674 weight = 0;
675 }
676 else if (t <= 0.5)
677 {
678 weight = (pow((6 * t - 1) / 2.0, 3)) / 2;
679 }
680 else if (t <= 5.0 / 6.0)
681 {
682 weight = (1 - pow((6 * (1-t) - 1) / 2.0, 3)) / 2 + 0.5;
683 }
684 else
685 {
686 weight = 1;
687 }
688
689 const QPointF delta = event->scenePos() - oldPosition;
690 const QPointF offset0 = ((1-weight)/(3*t*(1-t)*(1-t))) * delta;
691 const QPointF offset1 = (weight/(3*t*t*(1-t))) * delta;
692
693 const QPointF p2 = static_cast<QPointF>(spline.GetP2()) + offset0;
694 const QPointF p3 = static_cast<QPointF>(spline.GetP3()) + offset1;
695
696 oldPosition = event->scenePos(); // Now mouse here
697
698 const VSpline spl = VSpline(spline.GetP1(), p2, p3, spline.GetP4());
699
700 updateControlPoints(spl, newSplPath, splIndex);
701
702 MoveSplinePath *moveSplPath = new MoveSplinePath(doc, oldSplPath, newSplPath, m_id);
703 connect(moveSplPath, &VUndoCommand::NeedLiteParsing, doc, &VAbstractPattern::LiteParseTree);
704 qApp->getUndoStack()->push(moveSplPath);
705
706 // Each time we move something we call recalculation scene rect. In some cases this can cause moving
707 // objects positions. And this cause infinite redrawing. That's why we wait the finish of saving the last move.
708 static bool changeFinished = true;
709 if (changeFinished)
710 {
711 changeFinished = false;
712
713 const QList<QGraphicsView *> viewList = scene()->views();
714 if (!viewList.isEmpty())
715 {
716 if (QGraphicsView *view = viewList.at(0))
717 {
718 VMainGraphicsScene *currentScene = qobject_cast<VMainGraphicsScene *>(scene());
719 SCASSERT(currentScene)
720 const QPointF cursorPosition = currentScene->getScenePos();
721 view->ensureVisible(QRectF(cursorPosition.x()-5, cursorPosition.y()-5, 10, 10));
722 }
723 }
724 changeFinished = true;
725 }
726 }
727}

794//---------------------------------------------------------------------------------------------------------------------
795void VToolSplinePath::refreshCtrlPoints()
796{
797 // Very important to disable control points. Without it the pogram can't move the curve.
798 foreach (auto *point, m_controlPoints)
799 {
800 point->setFlag(QGraphicsItem::ItemSendsGeometryChanges, false);
801 }
802
803 const auto splPath = VAbstractTool::data.GeometricObject<VSplinePath>(m_id);
804
805 for (qint32 i = 1; i<=splPath->CountSubSpl(); ++i)
806 {
807 const qint32 j = i*2;
808
809 m_controlPoints[j-2]->blockSignals(true);
810 m_controlPoints[j-1]->blockSignals(true);
811
812 const auto spl = splPath->GetSpline(i);
813
814 {
815 const bool freeAngle1 = qmu::QmuTokenParser::IsSingle(spl.GetStartAngleFormula());
816 const bool freeLength1 = qmu::QmuTokenParser::IsSingle(spl.GetC1LengthFormula());
817
818 const auto splinePoint = spl.GetP1();
819 m_controlPoints[j-2]->refreshCtrlPoint(i, SplinePointPosition::FirstPoint, static_cast<QPointF>(spl.GetP2()),
820 static_cast<QPointF>(splinePoint), freeAngle1, freeLength1);
821 }
822
823 {
824 const bool freeAngle2 = qmu::QmuTokenParser::IsSingle(spl.GetEndAngleFormula());
825 const bool freeLength2 = qmu::QmuTokenParser::IsSingle(spl.GetC2LengthFormula());
826
827 const auto splinePoint = spl.GetP4();
828 m_controlPoints[j-1]->refreshCtrlPoint(i, SplinePointPosition::LastPoint, static_cast<QPointF>(spl.GetP3()),
829 static_cast<QPointF>(splinePoint), freeAngle2, freeLength2);
830 }
831
832 m_controlPoints[j-2]->blockSignals(false);
833 m_controlPoints[j-1]->blockSignals(false);
834 }
835
836 foreach (auto *point, m_controlPoints)
837 {
838 point->setFlag(QGraphicsItem::ItemSendsGeometryChanges, true);
839 }
