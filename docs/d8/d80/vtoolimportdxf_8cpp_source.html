100/**
101 * @brief The Deformation class moves any point as the tied points moved: moving least squares (Schaefer, McPhail,
102 * Warren 2006). One tie: the whole moves. Two: it moves, turns and grows. Three or more: it also changes shape,
103 * smoothly, each point following mostly the ties near it.
104 */
105class Deformation
106{
107public:
108 Deformation(const QVector<QPointF> &from, const QVector<QPointF> &to) : m_from(from), m_to(to) {}
109
110 QPointF map(const QPointF &v) const
111 {
112 const int n = m_from.size();
113 if (n == 0)
114 {
115 return v;
116 }
117 if (n == 1)
118 {
119 return v + (m_to.first() - m_from.first());
120 }
121
122 QVector<qreal> w(n);
123 qreal sum = 0;
124 QPointF pStar, qStar;
125 for (int i = 0; i < n; ++i)
126 {
127 const QPointF d = m_from.at(i) - v;
128 const qreal d2 = d.x() * d.x() + d.y() * d.y();
129 if (d2 < 1e-9)
130 {
131 return m_to.at(i);
132 }
133 w[i] = 1.0 / d2;
134 sum += w.at(i);
135 pStar += m_from.at(i) * w.at(i);
136 qStar += m_to.at(i) * w.at(i);
137 }
138 pStar /= sum;
139 qStar /= sum;
140
141 if (n >= 3)
142 {
143 // affine: M = (sum w p^T p)^-1 (sum w p^T q), row vectors
144 qreal a11 = 0, a12 = 0, a22 = 0, b11 = 0, b12 = 0, b21 = 0, b22 = 0;
145 for (int i = 0; i < n; ++i)
146 {
147 const QPointF p = m_from.at(i) - pStar;
148 const QPointF q = m_to.at(i) - qStar;
149 a11 += w.at(i) * p.x() * p.x();
150 a12 += w.at(i) * p.x() * p.y();
151 a22 += w.at(i) * p.y() * p.y();
152 b11 += w.at(i) * p.x() * q.x();
153 b12 += w.at(i) * p.x() * q.y();
154 b21 += w.at(i) * p.y() * q.x();
155 b22 += w.at(i) * p.y() * q.y();
156 }
157 const qreal det = a11 * a22 - a12 * a12;
158 if (qAbs(det) > 1e-9 * qMax(1.0, a11 * a22))
159 {
160 const qreal i11 = a22 / det, i12 = -a12 / det, i22 = a11 / det;
161 const qreal m11 = i11 * b11 + i12 * b21;
162 const qreal m12 = i11 * b12 + i12 * b22;
163 const qreal m21 = i12 * b11 + i22 * b21;
164 const qreal m22 = i12 * b12 + i22 * b22;
165 const QPointF r = v - pStar;
166 return QPointF(r.x() * m11 + r.y() * m21, r.x() * m12 + r.y() * m22) + qStar;
167 }
168 }
169
170 // similarity (two ties, or ties in a line): a complex factor a = sum w conj(p) q / sum w |p|^2
171 qreal re = 0, im = 0, norm = 0;
172 for (int i = 0; i < n; ++i)
173 {
174 const QPointF p = m_from.at(i) - pStar;
175 const QPointF q = m_to.at(i) - qStar;
176 re += w.at(i) * (p.x() * q.x() + p.y() * q.y());
177 im += w.at(i) * (p.x() * q.y() - p.y() * q.x());
178 norm += w.at(i) * (p.x() * p.x() + p.y() * p.y());
179 }
180 if (norm < 1e-12)
181 {
182 return v + (qStar - pStar);
183 }
184 re /= norm;
185 im /= norm;
186 const QPointF r = v - pStar;
187 return QPointF(r.x() * re - r.y() * im, r.x() * im + r.y() * re) + qStar;
188 }
189
190private:
191 QVector<QPointF> m_from;
192 QVector<QPointF> m_to;
193};

196struct Placement
197{
198 QPointF anchorRef;
199 QPointF dxfAnchor;
200 qreal angle {0};
201 Deformation deformation {{}, {}};
202
203 QPointF map(const QPointF &mm) const
204 {
205 return deformation.map(anchorRef + rotated(mmToPixel(mm) - dxfAnchor, angle));

210Placement makePlacement(const VContainer *data, quint32 anchor, int dxfAnchor, quint32 direction, int dxfDirection,
211 qreal angle, const QVector<DxfShape> &shapes, const QPointF &anchorReference,
212 const QPointF &directionReference, const QVector<DxfLink> &links)
213{
214 Placement placement;
215 QVector<QPointF> from, to;
216
217 const QPointF anchorNow = static_cast<QPointF>(*data->GeometricObject<VPointF>(anchor));
218 placement.anchorRef = unitsToPixel(anchorReference);
219 from.append(placement.anchorRef);
220 to.append(anchorNow);
221
222 if (dxfAnchor >= 0 && dxfAnchor < shapes.size() && not shapes.at(dxfAnchor).points.isEmpty())
223 {
224 placement.dxfAnchor = mmToPixel(shapes.at(dxfAnchor).points.first());
225 }
226
227 qreal turn = 0;
228 if (direction != NULL_ID && dxfDirection >= 0 && dxfDirection < shapes.size()
229 && not shapes.at(dxfDirection).points.isEmpty())
230 {
231 const QPointF directionRef = unitsToPixel(directionReference);
232 const QPointF dxfDirectionPoint = mmToPixel(shapes.at(dxfDirection).points.first());
233 turn = QLineF(placement.anchorRef, directionRef).angle()
234 - QLineF(placement.dxfAnchor, dxfDirectionPoint).angle();
235 from.append(directionRef);
236 to.append(static_cast<QPointF>(*data->GeometricObject<VPointF>(direction)));
237 }
238 placement.angle = turn + angle;
239
240 for (const DxfLink &link : links)
241 {
242 try
243 {
244 to.append(static_cast<QPointF>(*data->GeometricObject<VPointF>(link.point)));
245 from.append(unitsToPixel(link.reference));
246 }
247 catch (const VException &)
248 {
249 // the draft point was removed: the tie is ignored
250 }
251 }
252 placement.deformation = Deformation(from, to);
253 return placement;
254}

257/**
258 * @brief throughPoints a smooth curve through the points (Catmull-Rom), straight if there are only two.
259 */
260VSplinePath *throughPoints(const QVector<QPointF> &points, const QString &name)
261{
262 const int n = points.size();
263 QVector<VSplinePoint> splinePoints;
264 for (int i = 0; i < n; ++i)
265 {
266 const QPointF p = points.at(i);
267 QPointF tangent;
268 if (n == 2)
269 {
270 tangent = points.at(1) - points.at(0);
271 }
272 else if (i == 0)
273 {
274 tangent = points.at(1) - points.at(0);
275 }
276 else if (i == n - 1)
277 {
278 tangent = points.at(n - 1) - points.at(n - 2);
279 }
280 else
281 {
282 tangent = (points.at(i + 1) - points.at(i - 1)) / 2;
283 }
284 const qreal angle2 = QLineF(QPointF(), tangent).angle();
285 const qreal angle1 = std::fmod(angle2 + 180, 360);
286 const qreal before = i > 0 ? QLineF(points.at(i - 1), p).length() / 3 : 0;
287 const qreal after = i < n - 1 ? QLineF(p, points.at(i + 1)).length() / 3 : 0;
288 splinePoints.append(VSplinePoint(VPointF(p), angle1, QString::number(angle1), angle2, QString::number(angle2),
289 before, QString::number(qApp->fromPixel(before)), after,
290 QString::number(qApp->fromPixel(after))));
291 }
292 auto *path = new VSplinePath(splinePoints);
293 path->setName(name);
294 return path;
295}

321VToolImportDxf *VToolImportDxf::Create(const quint32 _id, quint32 anchor, int dxfAnchor, quint32 direction,
322 int dxfDirection, QString &angle, const QVector<DxfShape> &shapes,
323 const QPointF &anchorReference, const QPointF &directionReference,
324 const QVector<DxfLink> &links, const QVector<DestinationItem> &destination,
325 VMainGraphicsScene *scene, VAbstractPattern *doc, VContainer *data,
326 const Document &parse, const Source &typeCreation)
327{
328 const qreal angleValue = CheckFormula(_id, angle, data);
329 const Placement placement = makePlacement(data, anchor, dxfAnchor, direction, dxfDirection, angleValue, shapes,
330 anchorReference, directionReference, links);
331
332 QVector<DestinationItem> dest = destination;
333 quint32 id = _id;
334 const bool fromGui = typeCreation == Source::FromGui;
335 if (fromGui)
336 {
337 dest.clear();
338 id = VContainer::getNextId();
339 }
340
341 for (int i = 0; i < shapes.size(); ++i)
342 {
343 const DxfShape &shape = shapes.at(i);
344 if (shape.point)
345 {
346 const QPointF p = placement.map(shape.points.value(0));
347 VPointF point(p.x(), p.y(), shape.name, qApp->toPixel(0.13), qApp->toPixel(0.26));
348 point.setIdObject(id);
349 if (fromGui)
350 {
351 DestinationItem item;
352 item.mx = point.mx();
353 item.my = point.my();
354 item.showPointName = true;
355 item.id = data->AddGObject(new VPointF(point));
356 dest.append(item);
357 }
358 else if (i < dest.size())
359 {
360 const DestinationItem &item = dest.at(i);
361 point.setMx(item.mx);
362 point.setMy(item.my);
363 point.setShowPointName(item.showPointName);
364 data->UpdateGObject(item.id, new VPointF(point));
365 }
366 continue;
367 }
368
369 QVector<QPointF> mapped;
370 for (const QPointF &mm : shape.points)
371 {
372 mapped.append(placement.map(mm));
373 }
374 VSplinePath *path = throughPoints(mapped, shape.name);
375 path->setIdObject(id);
376 quint32 curveId = NULL_ID;
377 if (fromGui)
378 {
379 DestinationItem item;
380 item.mx = INT_MAX;
381 item.my = INT_MAX;
382 item.id = data->AddGObject(path);
383 curveId = item.id;
384 dest.append(item);
385 }
386 else if (i < dest.size())
387 {
388 curveId = dest.at(i).id;
389 data->UpdateGObject(curveId, path);
390 }
391 else
392 {
393 delete path;
394 }
395 if (curveId != NULL_ID)
396 {
397 data->AddCurveWithSegments(data->GeometricObject<VAbstractCubicBezierPath>(curveId), curveId);
398 }
399 }
400
401 if (not fromGui && parse != Document::FullParse)
402 {
403 doc->UpdateToolData(id, data);
404 }
405
406 if (parse == Document::FullParse)
407 {
408 VDrawTool::AddRecord(id, Tool::ImportDxf, doc);
409 VToolImportDxf *tool = new VToolImportDxf(doc, data, id, anchor, dxfAnchor, direction, dxfDirection, angle,
410 shapes, anchorReference, directionReference, links, dest,
411 typeCreation);
412 scene->addItem(tool);
413 initOperationToolConnections(scene, tool);
414 VAbstractPattern::AddTool(id, tool);
415 doc->IncrementReferens(data->GeometricObject<VPointF>(anchor)->getIdTool());
416 if (direction != NULL_ID)
417 {
418 doc->IncrementReferens(data->GeometricObject<VPointF>(direction)->getIdTool());
419 }
420 for (const DxfLink &link : links)
421 {
422 try
423 {
424 doc->IncrementReferens(data->GeometricObject<VPointF>(link.point)->getIdTool());
425 }
426 catch (const VException &)
427 {
428 }
429 }
430 return tool;
431 }
432 return nullptr;
433}

436QVector<DxfShape> VToolImportDxf::shapesFromFile(const AamaFile &file, const QVector<int> &pieces,
437 const VContainer *data)
438{
439 QVector<DxfShape> points;
440 QVector<DxfShape> curves;
441 QStringList taken;
442
443 const auto unique = [&taken](const QString &name)
444 {
445 return VContainer::IsUnique(name) && not taken.contains(name);
446 };
447 const auto flip = [](const QPointF &p) { return QPointF(p.x(), -p.y()); };
448 Q_UNUSED(data)
449
450 for (const int index : pieces)
451 {
452 if (index < 0 || index >= file.pieces.size())
453 {
454 continue;
455 }
456 const AamaPiece &piece = file.pieces.at(index);
457
458 // a short name made of letters and digits, not used yet in the pattern
459 QString base = piece.name;
460 base.remove(QRegularExpression(QStringLiteral("[^\\p{L}\\p{Nd}]")));
461 if (base.isEmpty() || base.at(0).isDigit())
462 {
463 base.prepend(QLatin1Char('P'));
464 }
465 QString prefix = base;
466 for (int k = 2; not unique(prefix + QStringLiteral("_1")); ++k)
467 {
468 prefix = base + QString::number(k);
469 }
470
471 // corners
472 QVector<int> corners;
473 QVector<QString> cornerNames;
474 for (int i = 0; i < piece.boundary.size(); ++i)
475 {
476 if (piece.turn.at(i))
477 {
478 corners.append(i);
479 const QString name = prefix + QLatin1Char('_') + QString::number(corners.size());
480 cornerNames.append(name);
481 taken.append(name);
482 points.append({true, name, {flip(piece.boundary.at(i))}});
483 }
484 }
485 // notches
486 for (int i = 0; i < piece.notches.size(); ++i)
487 {
488 const QString name = prefix + QStringLiteral("_N") + QString::number(i + 1);
489 taken.append(name);
490 points.append({true, name, {flip(piece.notches.at(i))}});
491 }
492 // the outline, from corner to corner
493 const int n = piece.boundary.size();
494 for (int c = 0; c < corners.size(); ++c)
495 {
496 const int from = corners.at(c);
497 const int to = corners.at((c + 1) % corners.size());
498 QVector<QPointF> line;
499 int i = from;
500 do
501 {
502 line.append(flip(piece.boundary.at(i)));
503 i = (i + 1) % n;
504 } while (i != to);
505 line.append(flip(piece.boundary.at(to)));
506 if (corners.size() == 1 && line.size() < 3)
507 {
508 continue;
509 }
510 const QString name = QStringLiteral("Spl_") + cornerNames.at(c) + QLatin1Char('_')
511 + cornerNames.at((c + 1) % corners.size());
512 curves.append({false, name, line});
513 }
514 // internal lines (darts...): a point at each end and at each corner, straight or curved between them
515 for (int l = 0; l < piece.internals.size(); ++l)
516 {
517 const QString lineName = prefix + QStringLiteral("_L") + QString::number(l + 1);
518 QVector<QPointF> line;
519 for (const QPointF &p : piece.internals.at(l))
520 {
521 line.append(flip(p));
522 }
523 QVector<int> breaks {0};
524 for (int i = 1; i + 1 < line.size(); ++i)
525 {
526 const qreal a1 = QLineF(line.at(i - 1), line.at(i)).angle();
527 const qreal a2 = QLineF(line.at(i), line.at(i + 1)).angle();
528 qreal change = std::fmod(qAbs(a2 - a1), 360.0);
529 change = qMin(change, 360.0 - change);
530 if (change > 30)
531 {
532 breaks.append(i);
533 }
534 }
535 breaks.append(line.size() - 1);
536
537 QStringList names;
538 for (int b = 0; b < breaks.size(); ++b)
539 {
540 QString name;
541 if (b == 0)
542 {
543 name = lineName + QLatin1Char('a');
544 }
545 else if (b == breaks.size() - 1)
546 {
547 name = lineName + QLatin1Char('b');
548 }
549 else
550 {
551 name = lineName + QLatin1Char('_') + QString::number(b);
552 }
553 names.append(name);
554 taken.append(name);
555 points.append({true, name, {line.at(breaks.at(b))}});
556 }
557 for (int b = 0; b + 1 < breaks.size(); ++b)
558 {
559 curves.append({false, QStringLiteral("Spl_") + names.at(b) + QLatin1Char('_') + names.at(b + 1),
560 line.mid(breaks.at(b), breaks.at(b + 1) - breaks.at(b) + 1)});
561 }
562 }
563 // grain line
564 if (piece.grain.size() == 2)
565 {
566 curves.append({false, QStringLiteral("Spl_") + prefix + QStringLiteral("_Grain"),
567 {flip(piece.grain.at(0)), flip(piece.grain.at(1))}});
568 }
569 }
570 return points + curves;
571}

574QVector<DxfShape> VToolImportDxf::readShapes(const QDomElement &domElement)
575{
576 QVector<DxfShape> shapes;
577 const QDomElement tag = domElement.firstChildElement(TagShapes);
578 for (QDomElement e = tag.firstChildElement(); not e.isNull(); e = e.nextSiblingElement())
579 {
580 DxfShape shape;
581 shape.point = e.tagName() == QLatin1String("point");
582 shape.name = e.attribute(QStringLiteral("name"));
583 for (const QString &pair : e.attribute(QStringLiteral("xy")).split(QLatin1Char(' '), Qt::SkipEmptyParts))
584 {
585 const QStringList xy = pair.split(QLatin1Char(','));
586 if (xy.size() == 2)
587 {
588 shape.points.append(QPointF(xy.at(0).toDouble(), xy.at(1).toDouble()));
589 }
590 }
591 if (not shape.points.isEmpty())
592 {
593 shapes.append(shape);
594 }
595 }
596 return shapes;
597}

599//---------------------------------------------------------------------------------------------------------------------
600QVector<DxfLink> VToolImportDxf::readLinks(const QString &value)
601{
602 QVector<DxfLink> links;
603 for (const QString &part : value.split(QLatin1Char(';'), Qt::SkipEmptyParts))
604 {
605 const QStringList fields = part.split(QLatin1Char(':'));
606 if (fields.size() != 3)
607 {
608 continue;
609 }
610 const QStringList xy = fields.at(2).split(QLatin1Char(','));
611 if (xy.size() != 2)
612 {
613 continue;
614 }
615 DxfLink link;
616 link.shape = fields.at(0).toInt();
617 link.point = fields.at(1).toUInt();
618 link.reference = QPointF(xy.at(0).toDouble(), xy.at(1).toDouble());
619 links.append(link);
620 }
621 return links;
622}

812void VToolImportDxf::showContextMenu(QGraphicsSceneContextMenuEvent *event, quint32 id)
813{
814 QMenu menu;
815 QAction *tieAll = menu.addAction(tr("Tie Points to the Draft..."));
816 QAction *tie = nullptr;
817 QAction *untie = nullptr;
818 const int shape = shapeOfDestination(id);
819 const bool isPoint = shape >= 0 && shape < m_shapes.size() && m_shapes.at(shape).point;
820 if (isPoint)
821 {
822 tie = menu.addAction(tr("Tie %1 to a Point of the Draft...").arg(m_shapes.at(shape).name));
823 bool tied = false;
824 for (const DxfLink &link : qAsConst(m_links))
825 {
826 tied = tied || link.shape == shape;
827 }
828 untie = menu.addAction(tr("Untie %1").arg(m_shapes.at(shape).name));
829 untie->setEnabled(tied);
830 }
831 menu.addSeparator();
832 QAction *remove = menu.addAction(QIcon::fromTheme(QStringLiteral("edit-delete")), tr("Delete DXF"));
833 remove->setEnabled(referens() <= 1);
834
835 QAction *chosen = menu.exec(event->screenPos());
836 if (chosen == nullptr)
837 {
838 return;
839 }
840 if (chosen == remove)
841 {
842 deleteTool(); // do not catch exception here
843 return;
844 }
845 if (chosen == untie)
846 {
847 linkPoint(id, NULL_ID);
848 return;
849 }
850
851 // the points of the draft (not the ones of this DXF), by name
852 QMap<QString, quint32> candidates;
853 QVector<quint32> own;
854 for (const DestinationItem &item : qAsConst(destination))
855 {
856 own.append(item.id);
857 }
858 const QHash<quint32, QSharedPointer<VGObject>> *objects = VAbstractTool::data.DataGObjects();
859 for (auto i = objects->cbegin(); i != objects->cend(); ++i)
860 {
861 if (static_cast<GOType>(i.value()->getType()) == GOType::Point && i.value()->getMode() == Draw::Calculation
862 && not own.contains(i.key()) && not i.value()->name().isEmpty())
863 {
864 candidates.insert(i.value()->name(), i.key());
865 }
866 }
867 if (candidates.isEmpty())
868 {
869 return;
870 }
871
872 if (chosen == tie)
873 {
874 bool ok = false;
875 const QString name = QInputDialog::getItem(qApp->getMainWindow(), tr("Tie DXF Point"),
876 tr("Point of the draft that %1 follows:")
877 .arg(m_shapes.at(shape).name),
878 candidates.keys(), 0, false, &ok);
879 if (ok && candidates.contains(name))
880 {
881 linkPoint(id, candidates.value(name));
882 }
883 return;
884 }
885
886 if (chosen == tieAll)
887 {
888 // one row per DXF point: which draft point it follows
889 QDialog dialog(qApp->getMainWindow());
890 dialog.setWindowTitle(tr("Tie DXF Points"));
891 auto *layout = new QVBoxLayout(&dialog);
892 auto *text = new QLabel(tr("Choose the point of the draft that each DXF point follows. The points you leave "
893 "free move in proportion to the tied ones. The point the DXF is placed on is "
894 "already tied."), &dialog);
895 text->setWordWrap(true);
896 layout->addWidget(text);
897 auto *table = new QTableWidget(&dialog);
898 table->setColumnCount(2);
899 table->setHorizontalHeaderLabels({tr("DXF point"), tr("Follows the draft point")});
900 table->horizontalHeader()->setStretchLastSection(true);
901 table->verticalHeader()->setVisible(false);
902 QVector<int> rows;
903 for (int i = 0; i < m_shapes.size(); ++i)
904 {
905 if (m_shapes.at(i).point && i != m_dxfAnchor)
906 {
907 rows.append(i);
908 }
909 }
910 table->setRowCount(rows.size());
911 QVector<QComboBox *> boxes;
912 for (int r = 0; r < rows.size(); ++r)
913 {
914 auto *name = new QTableWidgetItem(m_shapes.at(rows.at(r)).name);
915 name->setFlags(name->flags() & ~Qt::ItemIsEditable);
916 table->setItem(r, 0, name);
917 auto *box = new QComboBox(table);
918 box->addItem(tr("(free)"), static_cast<uint>(NULL_ID));
919 for (auto c = candidates.cbegin(); c != candidates.cend(); ++c)
920 {
921 box->addItem(c.key(), static_cast<uint>(c.value()));
922 }
923 for (const DxfLink &link : qAsConst(m_links))
924 {
925 if (link.shape == rows.at(r))
926 {
927 box->setCurrentIndex(qMax(0, box->findData(static_cast<uint>(link.point))));
928 }
929 }
930 table->setCellWidget(r, 1, box);
931 boxes.append(box);
932 }
933 table->resizeColumnsToContents();
934 layout->addWidget(table);
935 auto *buttons = new QDialogButtonBox(QDialogButtonBox::Ok | QDialogButtonBox::Cancel, &dialog);
936 connect(buttons, &QDialogButtonBox::accepted, &dialog, &QDialog::accept);
937 connect(buttons, &QDialogButtonBox::rejected, &dialog, &QDialog::reject);
938 layout->addWidget(buttons);
939 dialog.resize(460, 420);
940 if (dialog.exec() != QDialog::Accepted)
941 {
942 return;
943 }
944
945 QVector<DxfLink> links;
946 for (int r = 0; r < rows.size(); ++r)
947 {
948 const quint32 point = boxes.at(r)->currentData().toUInt();
949 if (point == NULL_ID)
950 {
951 continue;
952 }
953 DxfLink link;
954 link.shape = rows.at(r);
955 link.point = point;
956 // keep the reference of an unchanged tie, take the current position for a new one
957 link.reference = pixelToUnits(
958 static_cast<QPointF>(*VAbstractTool::data.GeometricObject<VPointF>(point)));
959 for (const DxfLink &old : qAsConst(m_links))
960 {
961 if (old.shape == link.shape && old.point == link.point)
962 {
963 link.reference = old.reference;
964 }
965 }
966 links.append(link);
967 }
968 m_links = links;
969 saveChange();
970 }
