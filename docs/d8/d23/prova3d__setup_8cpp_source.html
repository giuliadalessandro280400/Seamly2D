80sim::P2 reflect(const sim::P2 &p, const sim::P2 &a, const sim::P2 &b)
81{
82 const double dx = b.x - a.x, dy = b.y - a.y;
83 const double l = std::hypot(dx, dy);
84 if (l < 1e-9) return {2 * a.x - p.x, p.y};
85 const double ux = dx / l, uy = dy / l;
86 const double vx = p.x - a.x, vy = p.y - a.y;
87 const double d = vx * ux + vy * uy;
88 return {a.x + 2 * d * ux - vx, a.y + 2 * d * uy - vy};
89}

91double distanceToLine(const sim::P2 &p, const sim::P2 &a, const sim::P2 &b)
92{
93 const double dx = b.x - a.x, dy = b.y - a.y;
94 const double l = std::hypot(dx, dy);
95 if (l < 1e-9) return std::hypot(p.x - a.x, p.y - a.y);
96 return std::fabs((p.x - a.x) * dy - (p.y - a.y) * dx) / l;
97}

139Setup Setup::load(const VAbstractPattern *doc)
140{
141 Setup s;
142 if (doc == nullptr) return s;
143 const QDomElement root = doc->documentElement().firstChildElement(tagRoot);
144 for (QDomElement e = root.firstChildElement(); !e.isNull(); e = e.nextSiblingElement())
145 {
146 if (e.tagName() == QLatin1String("piece"))
147 {
148 PieceSetup p;
149 p.id = e.attribute(QStringLiteral("piece")).toUInt(); // not "id": ids are unique in the whole file
150 p.use = e.attribute(QStringLiteral("use"), QStringLiteral("true")) != QLatin1String("false");
151 p.mode = modeFromKey(e.attribute(QStringLiteral("mode")));
152 p.center1 = e.attribute(QStringLiteral("center1")).toUInt();
153 p.center2 = e.attribute(QStringLiteral("center2")).toUInt();
154 p.position = positionFromKey(e.attribute(QStringLiteral("position")));
155 s.pieces.append(p);
156 }
157 else if (e.tagName() == QLatin1String("seam"))
158 {
159 SeamSetup m;
160 m.pieceA = e.attribute(QStringLiteral("pieceA")).toUInt();
161 m.a1 = e.attribute(QStringLiteral("a1")).toUInt();
162 m.a2 = e.attribute(QStringLiteral("a2")).toUInt();
163 m.pieceB = e.attribute(QStringLiteral("pieceB")).toUInt();
164 m.b1 = e.attribute(QStringLiteral("b1")).toUInt();
165 m.b2 = e.attribute(QStringLiteral("b2")).toUInt();
166 m.reverse = e.attribute(QStringLiteral("reverse")) == QLatin1String("true");
167 m.sides = sidesFromKey(e.attribute(QStringLiteral("sides")));
168 s.seams.append(m);
169 }
170 }
171 return s;
172}

174void Setup::save(VAbstractPattern *doc) const
175{
176 if (doc == nullptr) return;
177 QDomElement root = doc->documentElement();
178 QDomElement old = root.firstChildElement(tagRoot);
179 if (!old.isNull()) root.removeChild(old);
180 QDomElement el = doc->createElement(tagRoot);
181 for (const PieceSetup &p : pieces)
182 {
183 QDomElement e = doc->createElement(QStringLiteral("piece"));
184 e.setAttribute(QStringLiteral("piece"), p.id);
185 e.setAttribute(QStringLiteral("use"), p.use ? QStringLiteral("true") : QStringLiteral("false"));
186 e.setAttribute(QStringLiteral("mode"), modeKey(p.mode));
187 if (p.center1 != 0) e.setAttribute(QStringLiteral("center1"), p.center1);
188 if (p.center2 != 0) e.setAttribute(QStringLiteral("center2"), p.center2);
189 e.setAttribute(QStringLiteral("position"), positionKey(p.position));
190 el.appendChild(e);
191 }
192 for (const SeamSetup &m : seams)
193 {
194 QDomElement e = doc->createElement(QStringLiteral("seam"));
195 e.setAttribute(QStringLiteral("pieceA"), m.pieceA);
196 e.setAttribute(QStringLiteral("a1"), m.a1);
197 e.setAttribute(QStringLiteral("a2"), m.a2);
198 e.setAttribute(QStringLiteral("pieceB"), m.pieceB);
199 e.setAttribute(QStringLiteral("b1"), m.b1);
200 e.setAttribute(QStringLiteral("b2"), m.b2);
201 if (m.reverse) e.setAttribute(QStringLiteral("reverse"), QStringLiteral("true"));
202 e.setAttribute(QStringLiteral("sides"), sidesKey(m.sides));
203 el.appendChild(e);
204 }
205 root.appendChild(el);
206 emit doc->patternChanged(false);
207}

249bool buildInput(const Setup &setup, const VContainer *data, Input &out, QString *error)
250{
251 out = Input();
252 QMap<quint32, QVector<Copy>> copies; // piece id -> its copies in the simulation
253 QMap<quint32, QMap<quint32, int>> nodeIndex; // piece id -> node id -> index in the outline of the half
254
255 for (const PieceSetup &ps : setup.pieces)
256 {
257 if (!ps.use) continue;
258 VPiece piece;
259 try
260 {
261 piece = data->GetPiece(ps.id);
262 }
263 catch (...)
264 {
265 continue;
266 }
267
268 // the seam line, in cm
269 const QVector<QPointF> path = piece.mainPathPoints(data);
270 std::vector<sim::P2> pts;
271 std::vector<QPointF> px;
272 for (const QPointF &p : path)
273 {
274 if (!px.empty() && QLineF(px.back(), p).length() < 1e-3) continue;
275 px.push_back(p);
276 pts.push_back({FromPixel(p.x(), Unit::Cm), FromPixel(p.y(), Unit::Cm)});
277 }
278 if (px.size() > 2 && QLineF(px.front(), px.back()).length() < 1e-3)
279 {
280 px.pop_back();
281 pts.pop_back();
282 }
283 if (pts.size() < 3) continue;
284
285 QMap<quint32, int> &nodes = nodeIndex[ps.id];
286 for (const VPieceNode &node : piece.GetPath().getNodes())
287 {
288 if (node.GetTypeTool() != Tool::NodePoint || node.isExcluded()) continue;
289 const QPointF at = data->GeometricObject<VPointF>(node.GetId())->toQPointF();
290 int best = -1;
291 double bd = 1e18;
292 for (size_t k = 0; k < px.size(); ++k)
293 {
294 const double d = QLineF(px[k], at).length();
295 if (d < bd) { bd = d; best = int(k); }
296 }
297 if (best >= 0 && bd < 1.0) nodes.insert(node.GetId(), best);
298 }
299
300 // grain: the grainline of the piece, vertical when it has none
301 double grain = 90;
302 {
303 bool ok = false;
304 const double r = piece.GetGrainlineGeometry().getRotation().toDouble(&ok);
305 if (ok && piece.GetGrainlineGeometry().IsVisible()) grain = -r;
306 }
307
308 const int n = int(pts.size());
309 const bool hasCenter = nodes.contains(ps.center1) && nodes.contains(ps.center2) && ps.center1 != ps.center2;
310 sim::P2 c1{0, 0}, c2{0, 0};
311 double centroidX = 0, minY = 1e18;
312 for (const sim::P2 &p : pts) { centroidX += p.x / n; minY = std::min(minY, p.y); }
313 if (hasCenter)
314 {
315 c1 = pts[size_t(nodes.value(ps.center1))];
316 c2 = pts[size_t(nodes.value(ps.center2))];
317 }
318 else if (ps.mode != Mode::Single)
319 {
320 out.warnings << QCoreApplication::translate("Prova3dSetup", "%1: no centre line, taken as whole")
321 .arg(piece.GetName());
322 }
323 const Mode mode = hasCenter ? ps.mode : Mode::Single;
324 const double centerX = hasCenter ? (c1.x + c2.x) / 2 : centroidX;
325 const double side = centroidX >= centerX ? 1.0 : -1.0; // which way the half goes from the centre line
326 const bool isBack = ps.position == Position::Back || ps.position == Position::BackWaist;
327 const bool fromWaist = ps.position == Position::FrontWaist || ps.position == Position::BackWaist;
328
329 auto makeInput = [&](const std::vector<sim::P2> &outline, const QString &suffix)
330 {
331 sim::PieceInput in;
332 in.name = (piece.GetName() + suffix).toStdString();
333 in.outline = outline;
334 in.grainDeg = grain;
335 in.centerX = centerX;
336 in.angleDeg = isBack ? 180 : 0;
337 // the half of copy 0 always goes to the same side of the body, front and back alike
338 in.mirrored = isBack ? (side > 0) : (side < 0);
339 in.topY = minY;
340 in.height = fromWaist ? 102.0 : 143.0;
341 return in;
342 };
343
344 QVector<Copy> list;
345 if (mode == Mode::Fold)
346 {
347 const int i1 = nodes.value(ps.center1), i2 = nodes.value(ps.center2);
348 auto walkStraight = [&](int from, int to)
349 {
350 for (int k = (from + 1) % n; k != to; k = (k + 1) % n)
351 if (distanceToLine(pts[size_t(k)], c1, c2) > 0.2) return false;
352 return true;
353 };
354 // the fold is the straight stretch between the two nodes; the rest is the half to mirror
355 int e0, e1;
356 if (walkStraight(i1, i2)) { e0 = i2; e1 = i1; }
357 else { e0 = i1; e1 = i2; }
358 std::vector<int> half;
359 for (int k = e0;; k = (k + 1) % n) { half.push_back(k); if (k == e1) break; }
360 const int h = int(half.size());
361 std::vector<sim::P2> full;
362 for (int k : half) full.push_back(pts[size_t(k)]);
363 for (int j = h - 2; j >= 1; --j) full.push_back(reflect(pts[size_t(half[size_t(j)])], c1, c2));
364 Copy a, b;
365 a.map.assign(size_t(n), -1);
366 b.map.assign(size_t(n), -1);
367 for (int q = 0; q < h; ++q)
368 {
369 a.map[size_t(half[size_t(q)])] = q;
370 b.map[size_t(half[size_t(q)])] = (q == 0 || q == h - 1) ? q : h + (h - 2 - q);
371 }
372 a.index = b.index = int(out.pieces.size());
373 out.pieces.push_back(makeInput(full, QString()));
374 list << a << b;
375 }
376 else if (mode == Mode::Pair)
377 {
378 Copy a, b;
379 a.map.resize(size_t(n));
380 b.map.resize(size_t(n));
381 std::vector<sim::P2> mirrored;
382 for (int k = n - 1; k >= 0; --k) mirrored.push_back(reflect(pts[size_t(k)], c1, c2));
383 for (int k = 0; k < n; ++k) { a.map[size_t(k)] = k; b.map[size_t(k)] = n - 1 - k; }
384 a.index = int(out.pieces.size());
385 out.pieces.push_back(makeInput(pts, QStringLiteral(" dx")));
386 b.index = int(out.pieces.size());
387 out.pieces.push_back(makeInput(mirrored, QStringLiteral(" sx")));
388 list << a << b;
389 }
390 else
391 {
392 Copy a;
393 a.map.resize(size_t(n));
394 for (int k = 0; k < n; ++k) a.map[size_t(k)] = k;
395 a.index = int(out.pieces.size());
396 out.pieces.push_back(makeInput(pts, QString()));
397 list << a;
398 }
399 copies.insert(ps.id, list);
400
401 // pins on the centre line, a little inside the piece
402 if (hasCenter && !fromWaist)
403 {
404 const double dx = c2.x - c1.x, dy = c2.y - c1.y;
405 const double l = std::hypot(dx, dy);
406 double nx = -dy / l, ny = dx / l;
407 if ((centroidX - c1.x) * nx < 0) { nx = -nx; ny = -ny; }
408 const double inset = mode == Mode::Fold ? 0.0 : 1.5;
409 for (double t : {0.15, 0.85})
410 {
411 const sim::P2 p{c1.x + dx * t + nx * inset, c1.y + dy * t + ny * inset};
412 out.pins.push_back({list.first().index, p});
413 if (mode == Mode::Pair) out.pins.push_back({list.at(1).index, reflect(p, c1, c2)});
414 }
415 }
416 else if (hasCenter)
417 {
418 const double dx = c2.x - c1.x, dy = c2.y - c1.y;
419 const double l = std::hypot(dx, dy);
420 double nx = -dy / l, ny = dx / l;
421 if ((centroidX - c1.x) * nx < 0) { nx = -nx; ny = -ny; }
422 const double inset = mode == Mode::Fold ? 0.0 : 1.5;
423 const sim::P2 top = c1.y < c2.y ? c1 : c2;
424 const sim::P2 p{top.x + nx * inset, top.y + 3.0 + ny * inset};
425 out.pins.push_back({list.first().index, p});
426 if (mode == Mode::Pair) out.pins.push_back({list.at(1).index, reflect(p, c1, c2)});
427 }
428 }
429
430 if (out.pieces.empty())
431 {
432 if (error) *error = QCoreApplication::translate("Prova3dSetup", "No piece is set up for the 3D fitting");
433 return false;
434 }
435
436 // seams: once on the half, on both halves unless separated
437 for (const SeamSetup &m : setup.seams)
438 {
439 if (!copies.contains(m.pieceA) || !copies.contains(m.pieceB)) continue;
440 const QVector<Copy> &ca = copies.value(m.pieceA), &cb = copies.value(m.pieceB);
441 const QMap<quint32, int> &na = nodeIndex.value(m.pieceA), &nb = nodeIndex.value(m.pieceB);
442 if (!na.contains(m.a1) || !na.contains(m.a2) || !nb.contains(m.b1) || !nb.contains(m.b2)) continue;
443 const QString name = QStringLiteral("%1 · %2").arg(pieceName(data, m.pieceA), pieceName(data, m.pieceB));
444
445 auto add = [&](int k, const QString &suffix)
446 {
447 const int ka = (k == 1 && ca.size() > 1) ? 1 : 0;
448 const int kb = (k == 1 && cb.size() > 1) ? 1 : 0;
449 const Copy &A = ca.at(ka);
450 const Copy &B = cb.at(kb);
451 const bool swapA = ka == 1, swapB = kb == 1; // a mirrored half runs the other way round
452 const quint32 aStart = swapA ? m.a2 : m.a1, aEnd = swapA ? m.a1 : m.a2;
453 const quint32 bStart = swapA ? m.b2 : m.b1, bEnd = swapA ? m.b1 : m.b2;
454 sim::SeamInput s;
455 s.pieceA = A.index;
456 s.pieceB = B.index;
457 s.fromA = A.map[size_t(na.value(aStart))];
458 s.toA = A.map[size_t(na.value(aEnd))];
459 s.fromB = B.map[size_t(nb.value(bStart))];
460 s.toB = B.map[size_t(nb.value(bEnd))];
461 s.reverseB = m.reverse != swapA;
462 if (swapB) s.reverseB = !s.reverseB;
463 if (s.fromA < 0 || s.toA < 0 || s.fromB < 0 || s.toB < 0) return;
464 s.name = (name + suffix).toStdString();
465 s.autoDirection = true;
466 out.seams.push_back(s);
467 };
468
469 const bool bothMirrored = ca.size() > 1 && cb.size() > 1;
470 const bool anyMirrored = ca.size() > 1 || cb.size() > 1;
471 switch (m.sides)
472 {
473 case Sides::Right:
474 add(0, anyMirrored ? QStringLiteral(" dx") : QString());
475 break;
476 case Sides::Left:
477 add(1, anyMirrored ? QStringLiteral(" sx") : QString());
478 break;
479 default:
480 add(0, bothMirrored ? QStringLiteral(" dx") : QString());
481 if (bothMirrored) add(1, QStringLiteral(" sx"));
482 break;
483 }
484 }
485 return true;
486}
