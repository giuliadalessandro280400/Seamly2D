16double pchip(const std::vector<double> &xs, const std::vector<double> &ys, double x)
17{
18 const int n = int(xs.size());
19 if (x <= xs.front()) return ys.front();
20 if (x >= xs.back()) return ys.back();
21 int k = int(std::upper_bound(xs.begin(), xs.end(), x) - xs.begin()) - 1;
22 auto slope = [&](int i) { return (ys[i + 1] - ys[i]) / (xs[i + 1] - xs[i]); };
23 auto deriv = [&](int i) {
24 if (i == 0) return slope(0);
25 if (i == n - 1) return slope(n - 2);
26 const double s0 = slope(i - 1), s1 = slope(i);
27 if (s0 * s1 <= 0) return 0.0;
28 return 2.0 / (1.0 / s0 + 1.0 / s1);
29 };
30 const double h = xs[k + 1] - xs[k], t = (x - xs[k]) / h;
31 const double d0 = deriv(k) * h, d1 = deriv(k + 1) * h;
32 const double t2 = t * t, t3 = t2 * t;
33 return (2 * t3 - 3 * t2 + 1) * ys[k] + (t3 - 2 * t2 + t) * d0 + (-2 * t3 + 3 * t2) * ys[k + 1] + (t3 - t2) * d1;
34}

229// squared Euclidean distance transform, 1D (Felzenszwalb & Huttenlocher)
230void edt1d(const std::vector<double> &f, std::vector<double> &d, int n)
231{
232 std::vector<int> v(n);
233 std::vector<double> z(n + 1);
234 int k = 0;
235 v[0] = 0; z[0] = -1e20; z[1] = 1e20;
236 for (int q = 1; q < n; ++q) {
237 double s;
238 while (true) {
239 s = ((f[q] + double(q) * q) - (f[v[k]] + double(v[k]) * v[k])) / (2.0 * q - 2.0 * v[k]);
240 if (s <= z[k] && k > 0) { --k; continue; }
241 break;
242 }
243 if (s <= z[k]) { v[0] = q; z[0] = -1e20; z[1] = 1e20; k = 0; continue; }
244 ++k; v[k] = q; z[k] = s; z[k + 1] = 1e20;
245 }
246 k = 0;
247 for (int q = 0; q < n; ++q) {
248 while (z[k + 1] < q) ++k;
249 d[q] = double(q - v[k]) * (q - v[k]) + f[v[k]];
250 }
251}

253std::vector<double> edt3d(const std::vector<char> &target, int nx, int ny, int nz)
254{
255 const double INF = 1e20;
256 std::vector<double> g(size_t(nx) * ny * nz);
257 for (size_t i = 0; i < g.size(); ++i) g[i] = target[i] ? 0 : INF;
258 auto at = [&](int x, int y, int z) -> double & { return g[(size_t(z) * ny + y) * nx + x]; };
259 std::vector<double> f, d;
260 f.resize(std::max({nx, ny, nz})); d.resize(f.size());
261 for (int z = 0; z < nz; ++z) for (int y = 0; y < ny; ++y) {
262 for (int x = 0; x < nx; ++x) f[x] = at(x, y, z);
263 edt1d(f, d, nx);
264 for (int x = 0; x < nx; ++x) at(x, y, z) = d[x];
265 }
266 for (int z = 0; z < nz; ++z) for (int x = 0; x < nx; ++x) {

269 for (int y = 0; y < ny; ++y) at(x, y, z) = d[y];

272 for (int z = 0; z < nz; ++z) f[z] = at(x, y, z);

305double DressForm::sdf(const V3 &p) const
306{
307 const double fx = (p.x - m_origin.x) / m_cell, fy = (p.y - m_origin.y) / m_cell, fz = (p.z - m_origin.z) / m_cell;
308 if (fx < 0 || fy < 0 || fz < 0 || fx >= m_nx - 1 || fy >= m_ny - 1 || fz >= m_nz - 1) return 1e9;
309 const int x = int(fx), y = int(fy), z = int(fz);
310 const double tx = fx - x, ty = fy - y, tz = fz - z;
311 auto at = [&](int i, int j, int k) { return double(m_sdf[(size_t(k) * m_ny + j) * m_nx + i]); };
312 const double c00 = at(x, y, z) * (1 - tx) + at(x + 1, y, z) * tx;
313 const double c10 = at(x, y + 1, z) * (1 - tx) + at(x + 1, y + 1, z) * tx;
314 const double c01 = at(x, y, z + 1) * (1 - tx) + at(x + 1, y, z + 1) * tx;
315 const double c11 = at(x, y + 1, z + 1) * (1 - tx) + at(x + 1, y + 1, z + 1) * tx;
316 return (c00 * (1 - ty) + c10 * ty) * (1 - tz) + (c01 * (1 - ty) + c11 * ty) * tz;
317}

449 const P2 gw{std::cos(g), std::sin(g)}; // warp direction
450 const P2 gf{-gw.y, gw.x}; // weft direction
451 auto toUV = [&](const P2 &q) { return P2{q.x * gf.x + q.y * gf.y, q.x * gw.x + q.y * gw.y}; };
452 auto toPattern = [&](const P2 &uv) { return P2{uv.x * gf.x + uv.y * gw.x, uv.x * gf.y + uv.y * gw.y}; };
453
454 std::vector<P2> uvOutline;
455 for (const P2 &q : p.outline) uvOutline.push_back(toUV(q));
456
457 const int first = int(m_x.size());
458 // emits the triangles of one region of a cell (snapping to grid crossings and outline corners)
459 auto emit = [&](const std::vector<P2> &region) {
460 if (region.size() < 3 || std::fabs(area(region)) < 1e-6) return;
461 std::vector<int> ids;
462 for (P2 q : region) {
463 bool snapped = false;
464 for (const P2 &v : uvOutline) {
465 if (std::hypot(q.x - v.x, q.y - v.y) < 0.25 * h) { q = v; snapped = true; break; }
466 }
467 if (!snapped) {
468 const P2 gc{std::round(q.x / h) * h, std::round(q.y / h) * h};
469 if (std::hypot(q.x - gc.x, q.y - gc.y) < 0.25 * h && pointInPolygon(uvOutline, gc)) q = gc;
470 }
471 const int id = nodeFor(pieceIndex, q, toPattern(q));
472 if (std::find(ids.begin(), ids.end(), id) == ids.end()) ids.push_back(id);
473 }
474 if (ids.size() < 3) return;
475 std::vector<P2> poly;
476 for (int id : ids) poly.push_back(m_nodeUV[id]);
477 for (const auto &t : earClip(poly)) {
478 std::array<int, 3> tr{ids[t[0]], ids[t[1]], ids[t[2]]};
479 const P2 &pa = m_nodeUV[tr[0]], &pb = m_nodeUV[tr[1]], &pc = m_nodeUV[tr[2]];
480 const double cr = (pb.x - pa.x) * (pc.y - pa.y) - (pb.y - pa.y) * (pc.x - pa.x);
481 if (std::fabs(cr) < 1e-4 * h * h) continue;
482 if (cr < 0) std::swap(tr[1], tr[2]);
483 m_tris.push_back(tr);
484 m_triPiece.push_back(pieceIndex);
485 }
486 };
487 auto segCross = [](const P2 &p1, const P2 &p2, const P2 &q1, const P2 &q2) {
488 auto orient = [](const P2 &a, const P2 &b, const P2 &c) { return (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x); };
489 const double d1 = orient(q1, q2, p1), d2 = orient(q1, q2, p2), d3 = orient(p1, p2, q1), d4 = orient(p1, p2, q2);
490 return ((d1 > 0) != (d2 > 0)) && ((d3 > 0) != (d4 > 0));
491 };
492 const auto ears = earClip(uvOutline);
493 double bx0 = 1e18, by0 = 1e18, bx1 = -1e18, by1 = -1e18;
494 for (const P2 &q : uvOutline) { bx0 = std::min(bx0, q.x); bx1 = std::max(bx1, q.x); by0 = std::min(by0, q.y); by1 = std::max(by1, q.y); }
495 const int nv0 = int(uvOutline.size());
496 for (int i = int(std::floor(bx0 / h)); i * h < bx1; ++i)
497 for (int j = int(std::floor(by0 / h)); j * h < by1; ++j) {
498 const double x0 = i * h, y0 = j * h, x1 = x0 + h, y1 = y0 + h;
499 const P2 sq[4] = {{x0, y0}, {x1, y0}, {x1, y1}, {x0, y1}};
500 int crossings = 0, inside = 0;
501 for (int e = 0; e < nv0; ++e) {
502 const P2 &a = uvOutline[e], &b = uvOutline[(e + 1) % nv0];
503 if (std::max(a.x, b.x) < x0 || std::min(a.x, b.x) > x1 || std::max(a.y, b.y) < y0 || std::min(a.y, b.y) > y1) continue;
504 for (int k = 0; k < 4; ++k) if (segCross(a, b, sq[k], sq[(k + 1) % 4])) ++crossings;
505 if (a.x > x0 && a.x < x1 && a.y > y0 && a.y < y1) ++inside;
506 }
507 if (crossings == 0 && inside == 0) {
508 if (pointInPolygon(uvOutline, P2{x0 + h / 2, y0 + h / 2})) emit({sq[0], sq[1], sq[2], sq[3]}); // a whole cell
509 continue;
510 }
511 if (crossings <= 2) {
512 emit(clipConvex(uvOutline, x0, y0, x1, y1)); // one piece of outline through the cell
513 continue;
514 }
515 // tricky cells (dart tips, narrow parts): clip the triangles of the outline, only here
516 for (const auto &t : ears) {
517 const std::vector<P2> tri{uvOutline[t[0]], uvOutline[t[1]], uvOutline[t[2]]};
518 emit(clipConvex(tri, x0, y0, x1, y1));
519 }
520 }
521 const int last = int(m_x.size());
522
523 // initial placement: wrapped on a cylinder
524 for (int k = first; k < last; ++k) {
525 const P2 &q = m_nodePattern[k];
526 const double th = p.angleDeg * kPi / 180 + (p.mirrored ? -1 : 1) * (q.x - p.centerX) / R;
527 m_x[k] = V3(R * std::sin(th), p.height - (q.y - p.topY), R * std::cos(th));
528 }
529
530 // nodes that ended up in no triangle (merged corners) are left out
531 std::vector<char> used(last - first, 0);
532 for (size_t t = 0; t < m_tris.size(); ++t)
533 if (m_triPiece[t] == pieceIndex)
534 for (int k : m_tris[t]) used[k - first] = 1;
535 for (int k = first; k < last; ++k) if (!used[k - first]) m_orphan.push_back(k);
536
537 // boundary: nodes lying on the outline, in order
538 Boundary &bd = m_boundary[pieceIndex];
539 double acc = 0;
540 std::vector<std::pair<double, int>> onEdge;
541 const int nv = int(uvOutline.size());
542 for (int e = 0; e < nv; ++e) {
543 const P2 a = uvOutline[e], b = uvOutline[(e + 1) % nv];
544 const double len = std::hypot(b.x - a.x, b.y - a.y);
545 bd.vertexParam.push_back(acc);
546 for (int k = first; k < last; ++k) {
547 if (!used[k - first]) continue;
548 const P2 &q = m_nodeUV[k];
549 const double t = ((q.x - a.x) * (b.x - a.x) + (q.y - a.y) * (b.y - a.y)) / (len * len);
550 if (t < -1e-9 || t > 1 + 1e-9) continue;
551 const double dx = a.x + t * (b.x - a.x) - q.x, dy = a.y + t * (b.y - a.y) - q.y;
552 if (std::hypot(dx, dy) > 1e-5) continue;
553 if (t > 1 - 1e-9) continue; // belongs to the next edge's start
554 onEdge.push_back({acc + t * len, k});
555 }
556 acc += len;
557 }
558 bd.length = acc;
559 std::sort(onEdge.begin(), onEdge.end());
560 for (const auto &pe : onEdge) {
561 if (!bd.nodes.empty() && bd.nodes.back() == pe.second) continue;
562 bd.param.push_back(pe.first);
563 bd.nodes.push_back(pe.second);
564 }
565
566 // the border: a ribbon along the outline that closes the net (chords over 1, 2 and 3 border nodes)
567 {
568 const int m = int(bd.nodes.size());
569 for (int i = 0; i < m; ++i)
570 for (int s = 1; s <= 3; ++s) {
571 const int a = bd.nodes[i], b = bd.nodes[(i + s) % m];
572 if (a == b) continue;
573 Dist d{};
574 d.a = a; d.b = b; d.kind = Edge;
575 const double du = m_nodeUV[b].x - m_nodeUV[a].x, dv = m_nodeUV[b].y - m_nodeUV[a].y;
576 d.rest = std::hypot(du, dv);
577 if (d.rest < 1e-4) continue;
578 d.dir = std::fabs(std::sin(2 * std::atan2(dv, du))); // 0 on grain, 1 on the bias
579 m_border.push_back(d);
580 }
581 }
582
583 // edges
584 std::map<std::pair<int, int>, bool> seen;
585 for (size_t t = 0; t < m_tris.size(); ++t) {
586 if (m_triPiece[t] != pieceIndex) continue;
587 for (int e = 0; e < 3; ++e) {
588 int a = m_tris[t][e], b = m_tris[t][(e + 1) % 3];
589 if (a > b) std::swap(a, b);
590 if (seen.count({a, b})) continue;
591 seen[{a, b}] = true;
592 const P2 &ua = m_nodeUV[a], &ub = m_nodeUV[b];
593 const double du = ub.x - ua.x, dv = ub.y - ua.y;
594 Dist d{};
595 d.a = a; d.b = b;
596 d.rest = std::hypot(du, dv);
597 if (d.rest < 1e-6) continue;
598 if (std::fabs(du) < 1e-7) { d.kind = Thread; d.dir = 1; } // warp
599 else if (std::fabs(dv) < 1e-7) { d.kind = Thread; d.dir = 0; } // weft
600 else { d.kind = Shear; d.dir = std::fabs(std::sin(2 * std::atan2(dv, du))); }
601 m_dist.push_back(d);
602 }
603 }
604
605 // bending along the threads: consecutive nodes on the same grid line
606 std::map<long long, std::vector<std::pair<double, int>>> warpLines, weftLines;
607 for (int k = first; k < last; ++k) {
608 const P2 &q = m_nodeUV[k];
609 const double fu = q.x / h, fv = q.y / h;
610 if (std::fabs(fu - std::round(fu)) < 1e-6) warpLines[std::llround(fu)].push_back({q.y, k});
611 if (std::fabs(fv - std::round(fv)) < 1e-6) weftLines[std::llround(fv)].push_back({q.x, k});
612 }
613 auto addBends = [&](std::map<long long, std::vector<std::pair<double, int>>> &lines, bool warp) {
614 for (auto &kv : lines) {
615 auto &v = kv.second;
616 std::sort(v.begin(), v.end());
617 for (size_t i = 1; i + 1 < v.size(); ++i) {
618 const int a = v[i - 1].second, b = v[i].second, c = v[i + 1].second;
619 auto connected = [&](int x, int y) { return seen.count({std::min(x, y), std::max(x, y)}) > 0; };
620 if (!connected(a, b) || !connected(b, c)) continue;
621 Bend bd2{};
622 bd2.a = a; bd2.b = c; bd2.warp = warp;
623 bd2.rest = std::fabs(v[i + 1].first - v[i - 1].first);
624 m_bend.push_back(bd2);
625 }
626 }
627 };
628 addBends(warpLines, true);
629 addBends(weftLines, false);
630
631 // bending across every inner edge (isometric bending, Bergou et al.): zero when the cloth is flat
632 {
633 std::map<std::pair<int, int>, std::vector<int>> edgeTris;
634 for (size_t t = 0; t < m_tris.size(); ++t) {
635 if (m_triPiece[t] != pieceIndex) continue;
636 for (int e = 0; e < 3; ++e) {
637 int a = m_tris[t][e], b = m_tris[t][(e + 1) % 3];
638 edgeTris[{std::min(a, b), std::max(a, b)}].push_back(int(t));
639 }
640 }
641 auto cot = [](const P2 &a, const P2 &b) {
642 const double cr = std::fabs(a.x * b.y - a.y * b.x);
643 const double c = (a.x * b.x + a.y * b.y) / std::max(cr, 1e-9);
644 return std::max(-8.0, std::min(8.0, c));
645 };
646 auto triArea = [&](int t) {
647 const P2 &a = m_nodeUV[m_tris[t][0]], &b = m_nodeUV[m_tris[t][1]], &c = m_nodeUV[m_tris[t][2]];
648 return std::fabs((b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x)) / 2;
649 };
650 for (const auto &kv : edgeTris) {
651 if (kv.second.size() != 2) continue;
652 const int x0 = kv.first.first, x1 = kv.first.second;
653 auto opposite = [&](int t) { for (int k : m_tris[t]) if (k != x0 && k != x1) return k; return -1; };
654 const int x2 = opposite(kv.second[0]), x3 = opposite(kv.second[1]);
655 if (x2 < 0 || x3 < 0) continue;
656 const double A = triArea(kv.second[0]) + triArea(kv.second[1]);
657 if (A < 1e-3 * h * h) continue;
658 auto d = [&](int a, int b) { return P2{m_nodeUV[b].x - m_nodeUV[a].x, m_nodeUV[b].y - m_nodeUV[a].y}; };
659 const P2 e0 = d(x0, x1), e1 = d(x0, x2), e2 = d(x0, x3), e3 = d(x1, x2), e4 = d(x1, x3);
660 const P2 me0{-e0.x, -e0.y};
661 const double c01 = cot(e0, e1), c02 = cot(e0, e2), c03 = cot(me0, e3), c04 = cot(me0, e4);
662 Hinge hg{};
663 hg.i[0] = x0; hg.i[1] = x1; hg.i[2] = x2; hg.i[3] = x3;
664 hg.K[0] = c03 + c04; hg.K[1] = c01 + c02; hg.K[2] = -c01 - c03; hg.K[3] = -c02 - c04;
665 hg.scale = 3.0 / A;
666 hg.kk = 0;
667 for (double k : hg.K) hg.kk += k * k;
668 m_hinge.push_back(hg);
669 }
670 }
671
672 // long-range tethers: grid crossings far apart can't get further than the cloth allows
673 std::map<std::pair<long long, long long>, int> grid;
674 for (int k = first; k < last; ++k) {
675 const P2 &q = m_nodeUV[k];
676 const double fu = q.x / h, fv = q.y / h;
677 if (std::fabs(fu - std::round(fu)) < 1e-6 && std::fabs(fv - std::round(fv)) < 1e-6) grid[{std::llround(fu), std::llround(fv)}] = k;
678 }
679 auto insideSegment = [&](const P2 &a, const P2 &b) {
680 const double len = std::hypot(b.x - a.x, b.y - a.y);
681 const int n = std::max(2, int(len / (0.5 * h)));
682 for (int i = 1; i < n; ++i) {
683 const P2 q{a.x + (b.x - a.x) * i / n, a.y + (b.y - a.y) * i / n};
684 if (!pointInPolygon(uvOutline, q)) return false;
685 }
686 return true;
687 };
688 const int dirs[4][2] = {{1, 0}, {0, 1}, {1, 1}, {1, -1}};
689 for (const auto &kv : grid) {
690 for (int s : {2, 4, 8, 16, 32}) {
691 for (const auto &d : dirs) {
692 auto it = grid.find({kv.first.first + d[0] * s, kv.first.second + d[1] * s});
693 if (it == grid.end()) continue;
694 const int a = kv.second, b = it->second;
695 if (!insideSegment(m_nodeUV[a], m_nodeUV[b])) continue;
696 Dist t{};
697 t.a = a; t.b = b; t.kind = Edge;
698 t.rest = std::hypot(m_nodeUV[b].x - m_nodeUV[a].x, m_nodeUV[b].y - m_nodeUV[a].y);
699 t.dir = (d[0] != 0 && d[1] != 0) ? 1.0 : (d[1] != 0 ? -1.0 : -2.0); // 1 diagonal, -1 warp, -2 weft
700 m_tether.push_back(t);
701 }
702 }
703 }
704}

706void Garment::setMaterial(const Material &m)
707{
708 m_mat = m;
709 const double sinLock = std::sin(m.lockAngleDeg * kPi / 180);
710 for (Dist &d : m_dist) {
711 if (m.paper) {
712 d.lo = d.hi = d.rest;
713 d.compliance = 0;
714 continue;
715 }
716 if (d.kind == Thread) {
717 const double eps = d.dir > 0.5 ? m.stretchWarp : m.stretchWeft;
718 d.lo = d.rest * 0.97;
719 d.hi = d.rest * (1 + eps);
720 d.compliance = 0;
721 } else {
722 const double eps = std::max(m.stretchWarp, m.stretchWeft);
723 d.lo = d.rest * std::sqrt(std::max(0.0, 1 - d.dir * sinLock));
724 d.hi = d.rest * std::sqrt(1 + d.dir * sinLock) * (1 + eps);
725 d.compliance = m.shearCompliance;
726 }
727 }
728 for (Bend &b : m_bend) b.compliance = m.bendCompliance;
729 for (Dist &d : m_border) {
730 if (m.paper || m.stayStitch) { d.lo = d.hi = d.rest; d.compliance = 0; continue; }
731 // fabric: the cut edge follows the cloth (a bias edge gives), only a light pull to keep it smooth
732 d.lo = 0;
733 d.hi = 1e9;
734 d.compliance = m.shearCompliance * 10;
735 }
736 for (Dist &t : m_tether) {
737 double k = 1;
738 if (!m.paper) {
739 if (t.dir > 0) k = std::sqrt(1 + sinLock) * (1 + std::max(m.stretchWarp, m.stretchWeft));
740 else k = 1 + (t.dir > -1.5 ? m.stretchWarp : m.stretchWeft);
741 }
742 t.hi = t.rest * k;
743 }
744 for (size_t k = 0; k < m_x.size(); ++k) {
745 const double kg = m_mass[k] * 1e-4 * m.density; // cm^2 -> m^2
746 m_w.resize(m_x.size());
747 m_w[k] = kg > 0 ? 1.0 / kg : 0;
748 (void)0;
749 }
750}

766
767Garment::Stats Garment::settle(double gravity, int maxSteps, double tol, double seamForceN)
768{
769 const auto t0 = std::chrono::steady_clock::now();
770 const double dt = std::getenv("SIMDT") ? std::atof(std::getenv("SIMDT")) : 1.0 / 120;
771 const double dt2 = dt * dt;
772 const double maxLambda = seamForceN * 100 * dt2; // N -> kg cm / s^2, times dt^2
773 (void)maxLambda;
774 const double sewSpeed = std::getenv("SEWSPD") ? std::atof(std::getenv("SEWSPD")) : 0.3; // cm per step a seam may close
775 const int iterations = std::getenv("SIMIT") ? std::atoi(std::getenv("SIMIT")) : 6;
776 const double momentum = std::getenv("SIMMOM") ? std::atof(std::getenv("SIMMOM")) : 0.5;
777 Stats st;
778 std::vector<double> w = m_w;
779 for (const Pin &p : m_pins) w[p.node] = 0;
780 const V3 g(0, -gravity, 0);
781 std::vector<double> seamTarget(m_seam.size());
782 std::vector<char> touching(m_x.size());
783 std::vector<double> pushed(m_x.size());
784 int quiet = 0;
785
786 auto seamPoints = [&](const SeamPt &s, V3 &pa, V3 &pb) {
787 pa = m_x[s.a0] * (1 - s.wa) + m_x[s.a1] * s.wa;
788 pb = m_x[s.b0] * (1 - s.wb) + m_x[s.b1] * s.wb;
789 };
790
791 for (int step = 0; step < maxSteps; ++step) {
792 for (size_t k = 0; k < m_x.size(); ++k) {
793 m_prev[k] = m_x[k];
794 if (w[k] == 0) continue;
795 m_v[k] += g * dt;
796 m_x[k] += m_v[k] * dt;
797 }
798 for (const Pin &p : m_pins) m_x[p.node] = p.to;
799 for (size_t i = 0; i < m_seam.size(); ++i) {
800 V3 pa, pb;
801 seamPoints(m_seam[i], pa, pb);
802 const double gap = (pb - pa).len();
803 const bool dart = m_seamInput[m_seam[i].seam].pieceA == m_seamInput[m_seam[i].seam].pieceB;
804 // sewn only when the edges really touch (zero ease); darts are closed first and always hold
805 if (gap < 0.05 && (m_latch || dart)) m_seam[i].sewn = true;
806 if (!m_seam[i].sewn && m_latch && gap < 3.0) {
807 // a single point left out between sewn ones (or at the end of a sewn stretch) is the mesh, not the fit
808 const bool hasPrev = i > 0 && m_seam[i - 1].seam == m_seam[i].seam;
809 const bool hasNext = i + 1 < m_seam.size() && m_seam[i + 1].seam == m_seam[i].seam;
810 const bool prevOk = !hasPrev || m_seam[i - 1].sewn;
811 const bool nextOk = !hasNext || m_seam[i + 1].sewn;
812 if (prevOk && nextOk && (hasPrev || hasNext)) m_seam[i].sewn = true;
813 }
814 seamTarget[i] = m_seam[i].sewn ? 0.0 : std::max(0.0, gap - sewSpeed);
815 m_seam[i].lambda = 0;
816 }
817 std::fill(touching.begin(), touching.end(), 0);
818 std::fill(pushed.begin(), pushed.end(), 0.0);
819
820 for (int it = 0; it < iterations; ++it) {
821 // seams: pulled closed, but never with more than the maximum force
822 for (size_t i = 0; i < m_seam.size(); ++i) {
823 SeamPt &s = m_seam[i];
824 if (s.sewn) continue;
825 V3 pa, pb;
826 seamPoints(s, pa, pb);
827 V3 n = pb - pa;
828 const double len = n.len();
829 const double C = len - seamTarget[i];
830 if (C <= 0 || len < 1e-9) continue;
831 n = n * (1.0 / len);
832 const double wa0 = 1 - s.wa, wa1 = s.wa, wb0 = 1 - s.wb, wb1 = s.wb;
833 const double ws = wa0 * wa0 * w[s.a0] + wa1 * wa1 * w[s.a1] + wb0 * wb0 * w[s.b0] + wb1 * wb1 * w[s.b1];
834 if (ws == 0) continue;
835 double dl = -C / ws;
836 if (!s.sewn && s.lambda + dl < -maxLambda) dl = -maxLambda - s.lambda;
837 if (dl >= 0) continue;
838 s.lambda += dl;
839 m_x[s.a0] -= n * (dl * wa0 * w[s.a0]);
840 m_x[s.a1] -= n * (dl * wa1 * w[s.a1]);
841 m_x[s.b0] += n * (dl * wb0 * w[s.b0]);
842 m_x[s.b1] += n * (dl * wb1 * w[s.b1]);
843 }
844 // threads and shear band
845 for (Dist &d : m_dist) {
846 const double wa = w[d.a], wb = w[d.b], ws = wa + wb;
847 if (ws == 0) continue;
848 V3 n = m_x[d.b] - m_x[d.a];
849 const double len = n.len();
850 if (len < 1e-12) continue;
851 n = n * (1.0 / len);
852 double C, alpha;
853 if (len < d.lo) { C = len - d.lo; alpha = 0; }
854 else if (len > d.hi) { C = len - d.hi; alpha = 0; }
855 else if (d.compliance > 0) { C = len - d.rest; alpha = d.compliance / dt2; }
856 else continue;

937 for (size_t k = 0; k < m_x.size(); ++k) {

971 }
