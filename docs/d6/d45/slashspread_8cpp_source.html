50qreal normalized(qreal angle) // -180 .. 180
51{
52 angle = std::fmod(angle, 360.0);
53 if (angle > 180)
54 {
55 angle -= 360;
56 }
57 if (angle <= -180)
58 {
59 angle += 360;
60 }
61 return angle;
62}

64QVector<QPointF> polylineAt(const QVector<QPointF> &line, const QVector<qreal> &fractions)
65{
66 QVector<qreal> lengths {0};
67 for (int i = 1; i < line.size(); ++i)
68 {
69 lengths.append(lengths.last() + QLineF(line.at(i - 1), line.at(i)).length());
70 }
71 const qreal total = lengths.last();
72 QVector<QPointF> points;
73 for (const qreal f : fractions)
74 {
75 const qreal s = f * total;
76 int i = 1;
77 while (i < lengths.size() - 1 && lengths.at(i) < s)
78 {
79 ++i;
80 }
81 const qreal segment = lengths.at(i) - lengths.at(i - 1);
82 const qreal k = segment > 0 ? (s - lengths.at(i - 1)) / segment : 0;
83 points.append(line.at(i - 1) + (line.at(i) - line.at(i - 1)) * k);
84 }
85 return points;
86}

90SlashSpread::SlashSpread(const QPointF &hinge, const QVector<QPointF> &edge, int slashes, qreal spread, bool percent,
91 Fixed fixed)
92 : m_hinge(hinge)
93{
94 if (edge.size() < 2 || slashes < 1)
95 {
96 return;
97 }
98
99 qreal edgeLength = 0;
100 for (int i = 1; i < edge.size(); ++i)
101 {
102 edgeLength += QLineF(edge.at(i - 1), edge.at(i)).length();
103 }
104 m_total = percent ? edgeLength * spread / 100.0 : spread;
105
106 // Slashes evenly spaced along the edge
107 QVector<qreal> fractions;
108 for (int i = 1; i <= slashes; ++i)
109 {
110 fractions.append(static_cast<qreal>(i) / (slashes + 1));
111 }
112 m_slashEnds = polylineAt(edge, fractions);
113
114 m_startAngle = QLineF(hinge, edge.first()).angle();
115 const qreal endPosition = normalized(QLineF(hinge, edge.last()).angle() - m_startAngle);
116 if (qFuzzyIsNull(endPosition))
117 {
118 return; // the edge is in line with the hinge: nothing to open
119 }
120 m_sweep = endPosition > 0 ? 1 : -1;
121
122 for (const QPointF &end : m_slashEnds)
123 {
124 m_slashPositions.append(fanPosition(end));
125 }
126
127 // Each slash opens so that the edge gets longer by the same amount there
128 const qreal gap = m_total / slashes;
129 QVector<qreal> cumulative {0};
130 for (const QPointF &end : m_slashEnds)
131 {
132 const qreal r = QLineF(hinge, end).length();
133 qreal angle = 0;
134 if (r > 0)
135 {
136 angle = qRadiansToDegrees(2 * qAsin(qBound(-1.0, gap / (2 * r), 1.0)));
137 }
138 cumulative.append(cumulative.last() + m_sweep * angle);
139 }
140
141 qreal fixedRotation = 0;
142 switch (fixed)
143 {
144 case Fixed::End:
145 fixedRotation = cumulative.last();
146 break;
147 case Fixed::Both:
148 {
149 const int low = slashes / 2;
150 const int high = (slashes + 1) / 2;
151 fixedRotation = (cumulative.at(low) + cumulative.at(high)) / 2;
152 break;
153 }
154 case Fixed::Start:
155 default:
156 break;
157 }
158 for (const qreal r : cumulative)
159 {
160 m_rotations.append(r - fixedRotation);
161 }
162 m_valid = true;
163}

172int SlashSpread::section(const QPointF &p) const
173{
174 if (QLineF(m_hinge, p).length() < 1e-6)
175 {
176 return -1; // the hinge itself does not move
177 }
178 const qreal position = fanPosition(p);
179 int section = 0;
180 for (const qreal slash : m_slashPositions)
181 {
182 if (position > slash + 1e-9)
183 {
184 ++section;
185 }
186 }
187 return section;
188}

191qreal SlashSpread::rotation(int section) const
192{
193 if (not m_valid || section < 0 || section >= m_rotations.size())
194 {
195 return 0;
196 }
197 return m_rotations.at(section);
198}

201QPointF SlashSpread::map(const QPointF &p) const
202{
203 const qreal angle = rotation(section(p));
204 if (qFuzzyIsNull(angle))
205 {
206 return p;
207 }
208 QLineF line(m_hinge, p);
209 line.setAngle(line.angle() + angle);
210 return line.p2();
211}

214VAbstractCurve *SlashSpread::opened(const QSharedPointer<VAbstractCurve> &curve, const QString &suffix) const
215{
216 const QVector<QPointF> points = curve->getPoints();
217 int first = -2;
218 bool oneSection = true;
219 for (const QPointF &p : points)
220 {
221 const int s = section(p);
222 if (s == -1)
223 {
224 continue; // on the hinge
225 }
226 if (first == -2)
227 {
228 first = s;
229 }
230 else if (s != first)
231 {
232 oneSection = false;
233 break;
234 }
235 }
236
237 if (oneSection)
238 {
239 const qreal angle = rotation(qMax(first, 0));
240 switch (static_cast<GOType>(curve->getType()))
241 {
242 case GOType::Arc:
243 return (new VArc(qSharedPointerDynamicCast<VArc>(curve)->Rotate(m_hinge, angle, suffix)));
244 case GOType::EllipticalArc:
245 return (new VEllipticalArc(
246 qSharedPointerDynamicCast<VEllipticalArc>(curve)->Rotate(m_hinge, angle, suffix)));
247 case GOType::Spline:
248 return (new VSpline(qSharedPointerDynamicCast<VSpline>(curve)->Rotate(m_hinge, angle, suffix)));
249 case GOType::SplinePath:
250 return (new VSplinePath(qSharedPointerDynamicCast<VSplinePath>(curve)->Rotate(m_hinge, angle, suffix)));
251 case GOType::CubicBezier:
252 return (new VCubicBezier(
253 qSharedPointerDynamicCast<VCubicBezier>(curve)->Rotate(m_hinge, angle, suffix)));
254 case GOType::CubicBezierPath:
255 return (new VCubicBezierPath(
256 qSharedPointerDynamicCast<VCubicBezierPath>(curve)->Rotate(m_hinge, angle, suffix)));
257 default:
258 break;
259 }
260 }
261
262 // Cut by a slash: the opened pieces joined again by one smooth curve
263 QVector<QPointF> moved;
264 moved.reserve(points.size());
265 for (const QPointF &p : points)
266 {
267 moved.append(map(p));
268 }
269 const QVector<qreal> splits = PieceCurveEdit::computeSplits(moved, ToPixel(0.3, Unit::Mm));
270 VSplinePath fitted = PieceCurveEdit::fitCurve(moved, splits);
271 const QVector<VSplinePoint> splinePoints = fitted.GetSplinePath();
272 auto *result = new VSplinePath(splinePoints);
273 result->setName(curve->name() + suffix);
274 result->setLineColor(curve->getLineColor());
275 result->SetPenStyle(curve->GetPenStyle());
276 result->setLineWeight(curve->getLineWeight());
277 return result;
278}

281QVector<QPointF> SlashSpread::edgeOf(const VContainer *data, quint32 startId, quint32 endId,
282 const QVector<quint32> &objects)
283{
284 const QPointF start = static_cast<QPointF>(*data->GeometricObject<VPointF>(startId));
285 const QPointF end = static_cast<QPointF>(*data->GeometricObject<VPointF>(endId));
286 for (const quint32 id : objects)
287 {
288 const QSharedPointer<VGObject> obj = data->GetGObject(id);
289 if (static_cast<GOType>(obj->getType()) == GOType::Point)
290 {
291 continue;
292 }
293 const QSharedPointer<VAbstractCurve> curve = data->GeometricObject<VAbstractCurve>(id);
294 QVector<QPointF> points = curve->getPoints();
295 if (points.size() < 2)
296 {
297 continue;
298 }
299 const qreal tolerance = 0.5; // pixels
300 if (QLineF(points.first(), start).length() < tolerance && QLineF(points.last(), end).length() < tolerance)
301 {
302 return points;
303 }
304 if (QLineF(points.first(), end).length() < tolerance && QLineF(points.last(), start).length() < tolerance)
305 {
306 std::reverse(points.begin(), points.end());
307 return points;
308 }
309 }
310 return {start, end};
311}
