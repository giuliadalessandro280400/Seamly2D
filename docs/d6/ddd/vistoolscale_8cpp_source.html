63void VisToolScale::RefreshGeometry()
64{
65 if (objects.isEmpty())
66 {
67 return;
68 }
69
70 QPointF origin;
71
72 if (object1Id != NULL_ID)
73 {
74 origin = static_cast<QPointF>(*Visualization::data->GeometricObject<VPointF>(object1Id));
75 DrawPoint(m_point, origin, supportColor2);
76
77 Visualization::toolTip = tr("Scale X = %1%3, Y = %2%3")
78 .arg(m_xScale).arg(m_yScale).arg(VToolScale::scalePostfix());
79 }
80
81 int iPoint = -1;
82 int iCurve = -1;
83 for (int i = 0; i < objects.size(); ++i)
84 {
85 const quint32 id = objects.at(i);
86 const QSharedPointer<VGObject> obj = Visualization::data->GetGObject(id);
87
88 // This check helps to find missed objects in the switch
89 Q_STATIC_ASSERT_X(static_cast<int>(GOType::Unknown) == 7, "Not all objects was handled.");
90
91 switch(static_cast<GOType>(obj->getType()))
92 {
93 case GOType::Point:
94 {
95 const QSharedPointer<VPointF> p = Visualization::data->GeometricObject<VPointF>(id);
96
97 ++iPoint;
98 VScaledEllipse *point = GetPoint(static_cast<quint32>(iPoint), supportColor2);
99 DrawPoint(point, static_cast<QPointF>(*p), supportColor2);
100
101 ++iPoint;
102 point = GetPoint(static_cast<quint32>(iPoint), supportColor);
103
104 if (object1Id != NULL_ID)
105 {
106 DrawPoint(point, VPointF::ScalePF(origin, static_cast<QPointF>(*p), m_xScale, m_yScale),
107 supportColor);
108 }
109 break;
110 }
111 case GOType::Arc:
112 iCurve = addCurve<VArc>(origin, id, iCurve);
113 break;
114 case GOType::EllipticalArc:
115 iCurve = addCurve<VEllipticalArc>(origin, id, iCurve);
116 break;
117 case GOType::Spline:
118 iCurve = addCurve<VSpline>(origin, id, iCurve);
119 break;
120 case GOType::SplinePath:
121 iCurve = addCurve<VSplinePath>(origin, id, iCurve);
122 break;
123 case GOType::CubicBezier:
124 iCurve = addCurve<VCubicBezier>(origin, id, iCurve);
125 break;
126 case GOType::CubicBezierPath:
127 iCurve = addCurve<VCubicBezierPath>(origin, id, iCurve);
128 break;
129 case GOType::Unknown:
130 case GOType::Curve:
131 case GOType::Path:
132 case GOType::AllCurves:
133 default:
134 break;
135 }
136 }
137}

159template <class Item>
160int VisToolScale::addCurve(const QPointF &origin, quint32 id, int i)
161{
162 const QSharedPointer<Item> curve = Visualization::data->template GeometricObject<Item>(id);
163
164 ++i;
165 VCurvePathItem *path = GetCurve(static_cast<quint32>(i), supportColor2);
166 DrawPath(path, curve->GetPath(), curve->DirectionArrows(), supportColor2, Qt::SolidLine,
167 lineWeight, Qt::RoundCap);
168
169 ++i;
170 path = GetCurve(static_cast<quint32>(i), supportColor);
171 if (object1Id != NULL_ID)
172 {
173 const Item scaled = curve->Scale(origin, m_xScale, m_yScale);
174 DrawPath(path, scaled.GetPath(), scaled.DirectionArrows(), supportColor, Qt::SolidLine,
175 lineWeight, Qt::RoundCap);
176 }
177
178 return i;
179}
