45/**
46 * @brief A DXF point tied to a point of the draft. The position of the draft point when the link was made is kept:
47 * when the draft changes (another size), the DXF follows how the tied points moved, and every other point of the DXF
48 * moves in proportion (moving least squares: near points follow the near links).
49 */
50struct DxfLink
51{
52 int shape {-1}; // index of the DXF point among the shapes
53 quint32 point {NULL_ID}; // point of the draft
54 QPointF reference; // where the draft point was when tied (pattern units)
55};
56
57/**
58 * @brief One object of an imported DXF: a point, or a curve through points. Coordinates in millimetres, y down,
59 * as in the file (only flipped).
60 */
61struct DxfShape
62{
63 bool point {true};
64 QString name;
65 QVector<QPointF> points;
66};

68/**
69 * @brief The VToolImportDxf class is a DXF pattern laid on the draft. The geometry of the file is kept in the pattern,

71 * optionally, turning it so that one of its lines follows a line of the draft); the angle can be a formula. The
