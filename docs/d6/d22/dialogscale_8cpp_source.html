251//---------------------------------------------------------------------------------------------------------------------
252void DialogScale::ChosenObject(quint32 id, const SceneObject &type)
253{
254 if (not m_stage1 && not prepare)// After first choose we ignore all objects
255 {
256 if (type == SceneObject::Point)
257 {
258 VisToolScale *operation = qobject_cast<VisToolScale *>(vis);
259 SCASSERT(operation != nullptr)
260
261 auto object = std::find_if(m_objects.begin(), m_objects.end(),
262 [id](const SourceItem &item) { return item.id == id; });
263
264 if (object != m_objects.end())
265 {
266 if (m_objects.size() > 1)
267 {
268 // The origin point does not move when scaled, so silently remove it from the list.
269 m_objects.erase(object);
270 operation->setObjects(sourceToObjects(m_objects));
271 }
272 else
273 {
274 emit ToolTip(tr("Select an origin point that is not part of the list of objects"));
275 return;
276 }
277 }
278
279 if (SetObject(id, ui->originPoint_ComboBox, ""))
280 {
281 VAbstractMainWindow *window = qobject_cast<VAbstractMainWindow *>(qApp->getMainWindow());
282 SCASSERT(window != nullptr)
283 connect(operation, &Visualization::ToolTip, window, &VAbstractMainWindow::setStatusMessage);
284
285 operation->setOriginPointId(id);
286 operation->RefreshGeometry();
287
288 prepare = true;
289 }
290 }
291 }

294//---------------------------------------------------------------------------------------------------------------------
295void DialogScale::SelectedObject(bool selected, quint32 id, quint32 tool)
296{
297 Q_UNUSED(tool)
298 if (m_stage1)
299 {
300 auto object = std::find_if(m_objects.begin(), m_objects.end(),
301 [id](const SourceItem &item) { return item.id == id; });
302
303 if (selected)
304 {
305 if (object == m_objects.cend())
306 {
307 SourceItem item;
308 item.id = id;
309 m_objects.append(item);
310 }
311 }
312 else
313 {
314 if (object != m_objects.end())
315 {
316 m_objects.erase(object);
317 }
318 }
319 }

385//---------------------------------------------------------------------------------------------------------------------
386void DialogScale::suffixChanged()
387{
388 QLineEdit* edit = qobject_cast<QLineEdit*>(sender());
389 if (edit)
390 {
391 const QString suffix = edit->text();
392 if (suffix.isEmpty())
393 {
394 flagName = false;
395 ChangeColor(ui->suffix_Label, Qt::red);
396 CheckState();
397 return;
398 }
399 else
400 {
401 if (m_suffix != suffix)
402 {
403 QRegularExpression rx(NameRegExp());
404 const QStringList uniqueNames = VContainer::AllUniqueNames();
405 for (int i=0; i < uniqueNames.size(); ++i)
406 {
407 const QString name = uniqueNames.at(i) + suffix;
408 if (not rx.match(name).hasMatch() || not data->IsUnique(name))
409 {
410 flagName = false;
411 ChangeColor(ui->suffix_Label, Qt::red);
412 CheckState();
413 return;
414 }
415 }
416 }
417 }
418
419 flagName = true;
420 ChangeColor(ui->suffix_Label, okColor);
421 }
422 CheckState();
423}
