39QDomElement findNode(VAbstractPattern *doc, quint32 pieceId, quint32 nodeId)
40{
41 const QDomElement piece = doc->elementById(pieceId, VAbstractPattern::TagPiece);
42 if (not piece.isElement())
43 {
44 return QDomElement();
45 }
46 QDomElement node = piece.firstChildElement(VAbstractPattern::TagNodes).firstChildElement(VAbstractPattern::TagNode);
47 while (not node.isNull())
48 {
49 if (node.attribute(AttrIdObject).toUInt() == nodeId)
50 {
51 return node;
52 }
53 node = node.nextSiblingElement(VAbstractPattern::TagNode);
54 }
55 return QDomElement();
56}

72QPointF MovePieceNode::readOffset(VAbstractPattern *doc, quint32 pieceId, quint32 nodeId)
73{
74 const QDomElement node = findNode(doc, pieceId, nodeId);
75 if (node.isNull())
76 {
77 return QPointF();
78 }
79 return QPointF(node.attribute(VAbstractPattern::AttrNodeOffsetX, "0").toDouble(),
80 node.attribute(VAbstractPattern::AttrNodeOffsetY, "0").toDouble());
81}

83//---------------------------------------------------------------------------------------------------------------------
84bool MovePieceNode::writeOffset(VAbstractPattern *doc, quint32 pieceId, quint32 nodeId, const QPointF &offset)
85{
86 QDomElement node = findNode(doc, pieceId, nodeId);
87 if (node.isNull())
88 {
89 qCDebug(vUndo, "Can't find node %u of piece %u.", nodeId, pieceId);
90 return false;
91 }
92
93 // Values below 0.01 mm are noise
94 if (qAbs(offset.x()) < 1e-4 && qAbs(offset.y()) < 1e-4)
95 {
96 node.removeAttribute(VAbstractPattern::AttrNodeOffsetX);
97 node.removeAttribute(VAbstractPattern::AttrNodeOffsetY);
98 }
99 else
100 {
101 doc->SetAttribute(node, VAbstractPattern::AttrNodeOffsetX, offset.x());
102 doc->SetAttribute(node, VAbstractPattern::AttrNodeOffsetY, offset.y());
103 }
104 return true;
105}

126bool MovePieceNode::mergeWith(const QUndoCommand *command)
127{
128 const MovePieceNode *other = static_cast<const MovePieceNode *>(command);
129 SCASSERT(other != nullptr)
130 if (other->m_pieceId != m_pieceId || other->nodeId != nodeId)
131 {
132 return false;
133 }
134 m_newOffset = other->m_newOffset;
135 return true;
