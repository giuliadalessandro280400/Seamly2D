44QDomElement findNode(VAbstractPattern *doc, quint32 pieceId, quint32 nodeId)
45{
46 const QDomElement piece = doc->elementById(pieceId, VAbstractPattern::TagPiece);
47 if (not piece.isElement())
48 {
49 return QDomElement();
50 }
51 QDomElement node = piece.firstChildElement(VAbstractPattern::TagNodes).firstChildElement(VAbstractPattern::TagNode);
52 while (not node.isNull())
53 {
54 if (node.attribute(AttrIdObject).toUInt() == nodeId)
55 {
56 return node;
57 }
58 node = node.nextSiblingElement(VAbstractPattern::TagNode);
59 }
60 return QDomElement();
61}
62} // namespace

77QPointF MovePieceNode::readOffset(VAbstractPattern *doc, quint32 pieceId, quint32 nodeId)
78{
79 const QDomElement node = findNode(doc, pieceId, nodeId);
80 if (node.isNull())
81 {
82 return QPointF();
83 }
84 return QPointF(node.attribute(VAbstractPattern::AttrNodeOffsetX, "0").toDouble(),
85 node.attribute(VAbstractPattern::AttrNodeOffsetY, "0").toDouble());

88//---------------------------------------------------------------------------------------------------------------------
89bool MovePieceNode::writeOffset(VAbstractPattern *doc, quint32 pieceId, quint32 nodeId, const QPointF &offset)
90{
91 QDomElement node = findNode(doc, pieceId, nodeId);
92 if (node.isNull())
93 {
94 qCDebug(vUndo, "Can't find node %u of piece %u.", nodeId, pieceId);
95 return false;
96 }
97
98 // Values below 0.01 mm are noise
99 if (qAbs(offset.x()) < 1e-4 && qAbs(offset.y()) < 1e-4)
100 {
101 node.removeAttribute(VAbstractPattern::AttrNodeOffsetX);
102 node.removeAttribute(VAbstractPattern::AttrNodeOffsetY);
103 }
104 else
105 {
106 doc->SetAttribute(node, VAbstractPattern::AttrNodeOffsetX, offset.x());
107 doc->SetAttribute(node, VAbstractPattern::AttrNodeOffsetY, offset.y());
108 }
109 return true;
110}

150MovePieceNodes::MovePieceNodes(VAbstractPattern *doc, quint32 pieceId, const QMap<quint32, QPointF> &newOffsets,
151 int dragKey, QUndoCommand *parent)
152 : VUndoCommand(QDomElement(), doc, parent)

155 , m_oldOffsets()
156 , m_newOffsets(newOffsets)

189bool MovePieceNodes::mergeWith(const QUndoCommand *command)
190{
191 const MovePieceNodes *other = static_cast<const MovePieceNodes *>(command);
192 SCASSERT(other != nullptr)
193 if (other->m_pieceId != m_pieceId || other->m_dragKey != m_dragKey)
194 {
195 return false;
196 }
197 for (auto i = other->m_newOffsets.cbegin(); i != other->m_newOffsets.cend(); ++i)
198 {
199 if (not m_oldOffsets.contains(i.key()))
200 {
201 m_oldOffsets.insert(i.key(), other->m_oldOffsets.value(i.key()));
202 }
203 m_newOffsets.insert(i.key(), i.value());
204 }
205 return true;
206}
