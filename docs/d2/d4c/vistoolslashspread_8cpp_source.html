67void VisToolSlashSpread::RefreshGeometry()
68{
69 if (objects.isEmpty())
70 {
71 return;
72 }
73
74 for (VScaledLine *line : m_slashLines)
75 {
76 line->setVisible(false);
77 }
78
79 if (m_hinge != NULL_ID)
80 {
81 DrawPoint(m_hingePoint, static_cast<QPointF>(*Visualization::data->GeometricObject<VPointF>(m_hinge)),
82 supportColor2);
83 }
84
85 const bool ready = m_hinge != NULL_ID && m_edgeStart != NULL_ID && m_edgeEnd != NULL_ID
86 && m_edgeStart != m_edgeEnd;
87 SlashSpread geometry(QPointF(), {}, 0, 0, false, m_fixed);
88 if (ready)
89 {
90 geometry = VToolSlashSpread::makeSpread(Visualization::data, m_hinge, m_edgeStart, m_edgeEnd, m_slashes,
91 m_spread, m_percent, m_fixed, objects);
92 const QVector<QPointF> ends = geometry.slashEnds();
93 while (m_slashLines.size() < ends.size())
94 {
95 m_slashLines.append(InitItem<VScaledLine>(supportColor2, this));
96 }
97 for (int i = 0; i < ends.size(); ++i)
98 {
99 DrawLine(m_slashLines.at(i), QLineF(geometry.hinge(), ends.at(i)), supportColor2, lineWeight,
100 Qt::DashLine);
101 }
102 Visualization::toolTip = tr("Slash and spread: %1 slashes, the edge gets %2 longer")
103 .arg(m_slashes).arg(qApp->fromPixel(geometry.totalSpread()), 0, 'f', 2);
104 }
105
106 int iPoint = -1;
107 int iCurve = -1;
108 for (const quint32 id : objects)
109 {
110 const QSharedPointer<VGObject> obj = Visualization::data->GetGObject(id);
111 if (static_cast<GOType>(obj->getType()) == GOType::Point)
112 {
113 const QSharedPointer<VPointF> point = Visualization::data->GeometricObject<VPointF>(id);
114 ++iPoint;
115 DrawPoint(GetPoint(static_cast<quint32>(iPoint), supportColor2), static_cast<QPointF>(*point),
116 supportColor2);
117 ++iPoint;
118 VScaledEllipse *item = GetPoint(static_cast<quint32>(iPoint), supportColor);
119 if (ready && geometry.isValid())
120 {
121 DrawPoint(item, geometry.map(static_cast<QPointF>(*point)), supportColor);
122 }
123 continue;
124 }
125
126 const QSharedPointer<VAbstractCurve> curve = Visualization::data->GeometricObject<VAbstractCurve>(id);
127 ++iCurve;
128 DrawPath(GetCurve(static_cast<quint32>(iCurve), supportColor2), curve->GetPath(), curve->DirectionArrows(),
129 supportColor2, Qt::SolidLine, lineWeight, Qt::RoundCap);
130 ++iCurve;
131 VCurvePathItem *path = GetCurve(static_cast<quint32>(iCurve), supportColor);
132 if (ready && geometry.isValid())
133 {
134 QScopedPointer<VAbstractCurve> opened(geometry.opened(curve, QString()));
135 DrawPath(path, opened->GetPath(), opened->DirectionArrows(), supportColor, Qt::SolidLine, lineWeight,
136 Qt::RoundCap);
137 }
138 }
139}
