59// closed polyline helpers
60QVector<qreal> cumulativeLengths(const QVector<QPointF> &points)
61{
62 QVector<qreal> lengths;
63 if (points.isEmpty())
64 {
65 return lengths;
66 }
67 lengths.append(0);
68 for (int i = 0; i < points.size(); ++i)
69 {
70 const QPointF &a = points.at(i);
71 const QPointF &b = points.at((i + 1) % points.size());
72 lengths.append(lengths.last() + QLineF(a, b).length());
73 }
74 return lengths; // size = points + 1, last = perimeter
75}
76
77qreal wrap(qreal t, qreal perimeter)
78{
79 if (perimeter <= 0)
80 {
81 return 0;
82 }
83 t = std::fmod(t, perimeter);
84 if (t < 0)
85 {
86 t += perimeter;
87 }
88 return t;
89}
90
91QPointF pointAt(const QVector<QPointF> &points, const QVector<qreal> &lengths, qreal t)
92{
93 if (points.isEmpty())
94 {
95 return QPointF();
96 }
97 const qreal perimeter = lengths.last();
98 t = wrap(t, perimeter);
99 for (int i = 0; i < points.size(); ++i)
100 {
101 if (t <= lengths.at(i + 1))
102 {
103 const qreal segment = lengths.at(i + 1) - lengths.at(i);
104 const qreal k = segment > 0 ? (t - lengths.at(i)) / segment : 0;
105 const QPointF &a = points.at(i);
106 const QPointF &b = points.at((i + 1) % points.size());
107 return a + (b - a) * k;
108 }
109 }
110 return points.first();
111}
112
113qreal project(const QVector<QPointF> &points, const QVector<qreal> &lengths, const QPointF &p)
114{
115 qreal best = 0;
116 qreal bestDistance = (std::numeric_limits<qreal>::max)();
117 for (int i = 0; i < points.size(); ++i)
118 {
119 const QPointF &a = points.at(i);
120 const QPointF &b = points.at((i + 1) % points.size());
121 const QPointF ab = b - a;
122 const qreal len2 = QPointF::dotProduct(ab, ab);
123 qreal k = len2 > 0 ? QPointF::dotProduct(p - a, ab) / len2 : 0;
124 k = qBound(0.0, k, 1.0);
125 const QPointF q = a + ab * k;
126 const qreal d = QLineF(p, q).length();
127 if (d < bestDistance)
128 {
129 bestDistance = d;
130 best = lengths.at(i) + k * (lengths.at(i + 1) - lengths.at(i));
131 }
132 }
133 return best;
134}
135
136qreal signedArea(const QVector<QPointF> &points)
137{
138 qreal area = 0;
139 for (int i = 0; i < points.size(); ++i)
140 {
141 const QPointF &a = points.at(i);
142 const QPointF &b = points.at((i + 1) % points.size());
143 area += a.x() * b.y() - b.x() * a.y();
144 }
145 return area / 2;
146}
147
148QPointF mirrored(const QPointF &p)
149{
150 return QPointF(p.x(), -p.y());
151}
152
153QString mm(qreal pixels)
154{
155 return QString::number(FromPixel(pixels, Unit::Mm), 'f', 1);
156}

239//---------------------------------------------------------------------------------------------------------------------
240bool SeamCheck::eventFilter(QObject *watched, QEvent *event)
241{
242 if (event->type() == QEvent::MouseButtonPress)
243 {
244 // the scene only knows the position of the last move: take the one of the click
245 if (VMainGraphicsView *view = qApp->getSceneView())
246 {
247 auto *mouseEvent = static_cast<QMouseEvent *>(event);
248 if (watched == view->viewport())
249 {
250 m_pressPos = view->mapToScene(mouseEvent->position().toPoint());
251 m_hasPressPos = true;
252 }
253 }
254 }
255 if (s_active == this && event->type() == QEvent::ShortcutOverride)
256 {
257 // keep our keys away from the shortcuts of the main window
258 const int key = static_cast<QKeyEvent *>(event)->key();
259 if (key == Qt::Key_Space || key == Qt::Key_F || key == Qt::Key_Return || key == Qt::Key_Enter
260 || key == Qt::Key_Escape)
261 {
262 event->accept();
263 return true;
264 }
265 }
266 if (s_active == this && event->type() == QEvent::KeyPress)
267 {
268 auto *keyEvent = static_cast<QKeyEvent *>(event);
269 switch (keyEvent->key())
270 {
271 case Qt::Key_Space:
272 switchSide();
273 return true;
274 case Qt::Key_F:
275 toggleMirror();
276 return true;
277 case Qt::Key_Return:
278 case Qt::Key_Enter:
279 toggleWalk();
280 return true;
281 case Qt::Key_Escape:
282 separate();
283 return true;
284 default:
285 break;
286 }
287 }
288 return QObject::eventFilter(watched, event);
289}

345PatternPieceTool *SeamCheck::pieceAt(const QPointF &scenePos, PatternPieceTool *exclude) const
346{
347 const QList<QGraphicsItem *> items = m_scene->items(scenePos);
348 for (QGraphicsItem *item : items)
349 {
350 for (QGraphicsItem *current = item; current != nullptr; current = current->parentItem())
351 {
352 if (current->type() == PatternPieceTool::Type)
353 {
354 auto *piece = static_cast<PatternPieceTool *>(current);
355 if (piece != exclude)
356 {
357 return piece;
358 }

360 }
361 }
362 }
363 return nullptr;

366//---------------------------------------------------------------------------------------------------------------------
367int SeamCheck::markAt(PatternPieceTool *piece, const QPointF &scenePos) const
368{
369 qreal scale = 1;
370 if (VMainGraphicsView *view = qApp->getSceneView())
371 {
372 scale = qMax(view->transform().m11(), 0.0001);
373 }
374 const qreal tolerance = 15.0 / scale;
375
376 const QVector<PatternPieceTool::ContourMark> marks = piece->editorMarks();
377 int best = -1;
378 qreal bestDistance = tolerance;
379 for (int i = 0; i < marks.size(); ++i)
380 {
381 const qreal d = QLineF(piece->mapToScene(marks.at(i).position), scenePos).length();
382 if (d <= bestDistance)
383 {
384 bestDistance = d;
385 best = i;
386 }
387 }
388 return best;
389}

392/**
393 * @brief updateContours reads both contours again (they change when points are corrected).
394 *
395 * The base contour is in scene coordinates, the checked one in its own coordinates (mirrored when flipped), so that
396 * placing it does not depend on where it is shown now.
397 */
398void SeamCheck::updateContours()
399{
400 if (m_base.isNull() || m_checked.isNull())
401 {
402 return;
403 }
404
405 m_baseContour.clear();
406 for (const QPointF &p : m_base->editorContour())
407 {
408 m_baseContour.append(m_base->mapToScene(p));
409 }
410 m_baseLengths = cumulativeLengths(m_baseContour);
411
412 m_checkedContour = m_checked->editorContour();
413 if (m_mirror)
414 {
415 for (QPointF &p : m_checkedContour)
416 {
417 p = mirrored(p);
418 }
419 }
420 m_checkedLengths = cumulativeLengths(m_checkedContour);
421
422 m_baseMarks.clear();
423 for (const PatternPieceTool::ContourMark &mark : m_base->editorMarks())
424 {
425 WalkMark walkMark;
426 walkMark.name = mark.name;
427 walkMark.notch = mark.notch;
428 walkMark.position = project(m_baseContour, m_baseLengths, m_base->mapToScene(mark.position));
429 m_baseMarks.append(walkMark);
430 }
431
432 m_checkedMarks.clear();
433 for (const PatternPieceTool::ContourMark &mark : m_checked->editorMarks())
434 {
435 WalkMark walkMark;
436 walkMark.name = mark.name;
437 walkMark.notch = mark.notch;
438 walkMark.position = project(m_checkedContour, m_checkedLengths,
439 m_mirror ? mirrored(mark.position) : mark.position);
440 m_checkedMarks.append(walkMark);
441 }
442
443 if (m_baseMark >= 0 && m_baseMark < m_baseMarks.size())
444 {
445 m_baseStart = m_baseMarks.at(m_baseMark).position;
446 }
447 if (m_checkedMark >= 0 && m_checkedMark < m_checkedMarks.size())
448 {
449 m_checkedStart = m_checkedMarks.at(m_checkedMark).position;
450 }
451}

454/**
455 * @brief checkedDirection the checked piece walks its contour so that both pieces lie on the same side of the seam,
456 * one on top of the other, whichever way the contours were drawn.
457 */
458int SeamCheck::checkedDirection() const
459{
460 const qreal baseArea = signedArea(m_baseContour);
461 const qreal checkedArea = signedArea(m_checkedContour);
462 const int same = ((baseArea >= 0) == (checkedArea >= 0)) ? 1 : -1;
463 return m_direction * same;
464}

466//---------------------------------------------------------------------------------------------------------------------
467QTransform SeamCheck::placement(qreal walked) const
468{
469 const qreal step = ToPixel(5, Unit::Mm);
470 const int checkedDir = checkedDirection();
471
472 const qreal tb = m_baseStart + m_direction * walked;
473 const qreal to = m_checkedStart + checkedDir * walked;
474
475 const QPointF b = pointAt(m_baseContour, m_baseLengths, tb);
476 const QPointF bNext = pointAt(m_baseContour, m_baseLengths, tb + m_direction * step);
477 const QPointF o = pointAt(m_checkedContour, m_checkedLengths, to);
478 const QPointF oNext = pointAt(m_checkedContour, m_checkedLengths, to + checkedDir * step);
479
480 const qreal angle = qAtan2(bNext.y() - b.y(), bNext.x() - b.x()) - qAtan2(oNext.y() - o.y(), oNext.x() - o.x());
481
482 QTransform t;
483 t.translate(b.x(), b.y());
484 t.rotateRadians(angle);
485 t.translate(-o.x(), -o.y());
486 if (m_mirror)
487 {
488 t.scale(1, -1);
489 }
490 return t;
491}

504//---------------------------------------------------------------------------------------------------------------------
505/**
506 * @brief updateReadout shows the walked length and, for every point met on the way, how far it is on both pieces.
507 */
508void SeamCheck::updateReadout()
509{
510 if (m_readout == nullptr || m_baseLengths.isEmpty() || m_checkedLengths.isEmpty())
511 {
512 return;
513 }
514
515 const qreal basePerimeter = m_baseLengths.last();
516 const qreal checkedPerimeter = m_checkedLengths.last();
517 const int checkedDir = checkedDirection();
518
519 struct Met { qreal distance; QString name; bool notch; };
520 auto collect = [](const QVector<WalkMark> &marks, qreal start, int direction, qreal perimeter)
521 {
522 QVector<Met> met;
523 for (const WalkMark &mark : marks)
524 {
525 qreal d = wrap((mark.position - start) * direction, perimeter);
526 if (perimeter - d < 0.01)
527 {
528 d = 0; // the start point itself
529 }
530 met.append({d, mark.name, mark.notch});
531 }
532 std::sort(met.begin(), met.end(), [](const Met &a, const Met &b) { return a.distance < b.distance; });
533 return met;
534 };
535
536 const QVector<Met> base = collect(m_baseMarks, m_baseStart, m_direction, basePerimeter);
537 const QVector<Met> checked = collect(m_checkedMarks, m_checkedStart, checkedDir, checkedPerimeter);
538
539 auto label = [](const Met &m)
540 {
541 return (m.notch ? (QString(QChar(0x25C6)) + QLatin1Char(' ')) : QString()) + m.name.toHtmlEscaped();
542 };
543
544 QString html = QStringLiteral("<p><b>%1</b> %2 mm</p>").arg(tr("Walked:"), mm(m_walked));
545 html += QStringLiteral("<table cellspacing='0' cellpadding='3'><tr><th align='left'>%1</th><th>mm</th>"
546 "<th align='left'>%2</th><th>mm</th><th>%3</th></tr>")
547 .arg(tr("Base"), tr("Checked"), tr("Difference"));
548
549 QString lastDifference;
550 const int rows = qMin(qMax(base.size(), checked.size()), 12);
551 for (int i = 0; i < rows; ++i)
552 {
553 const bool hasBase = i < base.size();
554 const bool hasChecked = i < checked.size();
555 // the mouse walks the base piece: a row is reached when its base point is
556 const bool passed = hasBase && base.at(i).distance <= m_walked + ToPixel(0.5, Unit::Mm);
557
558 QString diff;
559 if (hasBase && hasChecked)
560 {
561 const qreal d = checked.at(i).distance - base.at(i).distance;
562 diff = (d > 0 ? QStringLiteral("+") : QString()) + mm(d);
563 if (passed)
564 {
565 lastDifference = diff;
566 }
567 }
568
569 const QString style = passed ? QStringLiteral(" style='font-weight:bold'") : QString();
570 html += QStringLiteral("<tr%1><td>%2</td><td align='right'>%3</td><td>%4</td><td align='right'>%5</td>"
571 "<td align='right'>%6</td></tr>")
572 .arg(style,
573 hasBase ? label(base.at(i)) : QString(),
574 hasBase ? mm(base.at(i).distance) : QString(),
575 hasChecked ? label(checked.at(i)) : QString(),
576 hasChecked ? mm(checked.at(i).distance) : QString(),
577 diff);
578 }
579 html += QStringLiteral("</table>");
580 if (not lastDifference.isEmpty())
581 {
582 html += QStringLiteral("<p><b>%1</b> %2 mm</p>").arg(tr("Difference at the last point reached:"),
583 lastDifference);
584 }
585 m_readout->setText(html);
586 m_lastDifference = lastDifference;
587}

589//---------------------------------------------------------------------------------------------------------------------
590void SeamCheck::mouseMoved(const QPointF &scenePos)
591{
592 if (m_stage != Stage::Walking || m_baseLengths.isEmpty())
593 {
594 return;
595 }
596 const qreal t = project(m_baseContour, m_baseLengths, scenePos);
597 const qreal perimeter = m_baseLengths.last();
598 qreal walked = wrap((t - m_baseStart) * m_direction, perimeter);
599 if (m_walked < perimeter / 4 && walked > perimeter * 3 / 4)
600 {
601 walked = 0; // the mouse went back behind the start point: stay there instead of jumping around the piece
602 }
603 m_walked = walked;
604 place();
605}

607//---------------------------------------------------------------------------------------------------------------------
608void SeamCheck::leftPressed()
609{
610 if (s_active != this || m_checked.isNull())
611 {
612 return;
613 }
614 const QPointF pos = m_hasPressPos ? m_pressPos : m_scene->getScenePos();
615 m_hasPressPos = false;
616
617 switch (m_stage)
618 {
619 case Stage::PickBase:
620 {
621 PatternPieceTool *base = pieceAt(pos, m_checked);
622 if (base == nullptr)
623 {
624 return;
625 }
626 m_base = base;
627 connect(base, &QObject::destroyed, this, &SeamCheck::separate);
628 m_stage = Stage::PickCheckedPoint;
629 setStatus(tr("Click the start point on the piece to check (colored)."));
630 break;
631 }
632 case Stage::PickCheckedPoint:
633 {
634 const int mark = markAt(m_checked, pos);
635 if (mark < 0)
636 {
637 return;
638 }
639 m_checkedMark = mark;
640 m_stage = Stage::PickBasePoint;
641 setStatus(tr("Click the point of the base piece where it starts."));
642 break;
643 }
644 case Stage::PickBasePoint:
645 {
646 if (m_base.isNull())
647 {
648 return;
649 }
650 const int mark = markAt(m_base, pos);
651 if (mark < 0)
652 {
653 return;
654 }
655 m_baseMark = mark;
656 updateContours();
657 m_walked = 0;
658 startWalking();
659 mouseMoved(pos);
660 place();
661 break;
662 }
663 case Stage::Walking:
664 toggleWalk(); // a click stops it, so the points can be corrected
665 break;
666 case Stage::Stopped:
667 default:
668 break; // clicks belong to the piece editor now
669 }
670}
