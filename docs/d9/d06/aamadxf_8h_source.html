36/**
37 * A piece of an AAMA DXF file (ASTM D6673). Coordinates are in millimetres, y up as in the file.
38 * layer 1 boundary (closed polyline) and texts layer 2 turn points (corners)
39 * layer 3 curve points layer 4 notches
40 * layer 7 grain line layer 8 internal lines
41 */
42struct AamaPiece
43{
44 QString name;
45 QVector<QPointF> boundary; // vertices of the outline
46 QVector<bool> turn; // true where the vertex is a corner
47 QVector<QPointF> notches;
48 QVector<QVector<QPointF>> internals;
49 QVector<QPointF> grain; // empty or two points
50 QStringList texts;
51};
52
53struct AamaFile
54{
55 QVector<AamaPiece> pieces;
56 QString units; // what the file says about its units ("mm", "cm", "inch", or empty)
57 QString error; // empty if the file could be read
58};
