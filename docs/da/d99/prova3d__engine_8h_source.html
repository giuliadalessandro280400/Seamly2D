21struct V3 {
22 double x{0}, y{0}, z{0};
23 V3() = default;
24 V3(double a, double b, double c) : x(a), y(b), z(c) {}
25 V3 operator+(const V3 &o) const { return {x + o.x, y + o.y, z + o.z}; }
26 V3 operator-(const V3 &o) const { return {x - o.x, y - o.y, z - o.z}; }
27 V3 operator*(double s) const { return {x * s, y * s, z * s}; }
28 V3 &operator+=(const V3 &o) { x += o.x; y += o.y; z += o.z; return *this; }
29 V3 &operator-=(const V3 &o) { x -= o.x; y -= o.y; z -= o.z; return *this; }
30 double dot(const V3 &o) const { return x * o.x + y * o.y + z * o.z; }
31 double len() const { return std::sqrt(dot(*this)); }
32};
33inline V3 cross(const V3 &a, const V3 &b) { return {a.y * b.z - a.z * b.y, a.z * b.x - a.x * b.z, a.x * b.y - a.y * b.x}; }

77struct Material {
78 bool paper{true};
79 double stretchWarp{0}; // allowed thread elongation (0.03 = 3%)
80 double stretchWeft{0};
81 double lockAngleDeg{35}; // max shear angle before the threads jam (fabric only)
82 double shearCompliance{2e-3};// softness of the shear before locking (fabric)
83 double bendCompliance{1e-1}; // bending softness (bigger = softer)
84 double density{0.08}; // kg/m^2 (paper 80 g/m^2)
85 double thickness{0.15}; // cm, collision gap
86 bool stayStitch{false}; // fabric only: border taped, it doesn't stretch
87 double friction{std::getenv("FRIC") ? std::atof(std::getenv("FRIC")) : 0.6};
88
89 static Material paperDefault() { Material m; m.paper = true; m.bendCompliance = std::getenv("PBEND") ? std::atof(std::getenv("PBEND")) : 3.0; m.density = 0.08; return m; }
90 static Material fabricDefault() { Material m; m.paper = false; m.bendCompliance = std::getenv("FBEND") ? std::atof(std::getenv("FBEND")) : 5000.0; m.density = 0.15; return m; }
