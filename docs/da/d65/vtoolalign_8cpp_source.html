70AlignTransform VToolAlign::computeTransform(const QPointF &p, const QPointF &a, const QPointF &t, const QPointF &b,
71 qreal extraAngle)
72{
73 AlignTransform transform;
74 const QLineF move(p, t);
75 transform.length = move.length();
76 transform.angle = move.angle();
77 transform.origin = t;
78
79 const QLineF sourceLine(p, a);
80 const QLineF targetLine(t, b);
81 qreal rotation = extraAngle;
82 if (not qFuzzyIsNull(sourceLine.length()) && not qFuzzyIsNull(targetLine.length()))
83 {
84 rotation += targetLine.angle() - sourceLine.angle();
85 }
86 transform.rotation = rotation;
87 return transform;
88}

141//---------------------------------------------------------------------------------------------------------------------
142VToolAlign *VToolAlign::Create(const quint32 _id, quint32 sourcePoint, quint32 sourceDirection,
143 quint32 targetPoint, quint32 targetDirection, QString &angle,
144 const QString &suffix, const QVector<SourceItem> &source,
145 const QVector<DestinationItem> &destination, VMainGraphicsScene *scene,
146 VAbstractPattern *doc, VContainer *data, const Document &parse,
147 const Source &typeCreation)
148{
149 const qreal extraAngle = CheckFormula(_id, angle, data);
150
151 const QSharedPointer<VPointF> p = data->GeometricObject<VPointF>(sourcePoint);
152 const QSharedPointer<VPointF> a = data->GeometricObject<VPointF>(sourceDirection);
153 const QSharedPointer<VPointF> t = data->GeometricObject<VPointF>(targetPoint);
154 const QSharedPointer<VPointF> b = data->GeometricObject<VPointF>(targetDirection);
155
156 const AlignTransform transform = computeTransform(static_cast<QPointF>(*p), static_cast<QPointF>(*a),
157 static_cast<QPointF>(*t), static_cast<QPointF>(*b),
158 extraAngle);
159
160 QVector<DestinationItem> dest = destination;
161
162 quint32 id = _id;
163 const bool fromGui = typeCreation == Source::FromGui;
164 if (fromGui)
165 {
166 dest.clear();// Try to avoid mistake, value must be empty
167 id = VContainer::getNextId();//Just reserve id for tool
168 }
169
170 for (int i = 0; i < source.size(); ++i)
171 {
172 const quint32 objectId = source.at(i).id;
173 const QSharedPointer<VGObject> obj = data->GetGObject(objectId);
174
175 // This check helps to find missed objects in the switch
176 Q_STATIC_ASSERT_X(static_cast<int>(GOType::Unknown) == 7, "Not all objects were handled.");
177
178QT_WARNING_PUSH
179QT_WARNING_DISABLE_GCC("-Wswitch-default")
180 switch(static_cast<GOType>(obj->getType()))
181 {
182 case GOType::Point:
183 if (fromGui)
184 {
185 dest.append(createPoint(id, objectId, transform, suffix, data));
186 }
187 else
188 {
189 updatePoint(id, objectId, transform, suffix, data, dest.at(i));
190 }
191 break;
192 case GOType::Arc:
193 fromGui ? dest.append(createItem<VArc>(id, objectId, transform, suffix, data))
194 : updateItem<VArc>(id, objectId, transform, suffix, data, dest.at(i).id);
195 break;
196 case GOType::EllipticalArc:
197 fromGui ? dest.append(createItem<VEllipticalArc>(id, objectId, transform, suffix, data))
198 : updateItem<VEllipticalArc>(id, objectId, transform, suffix, data, dest.at(i).id);
199 break;
200 case GOType::Spline:
201 fromGui ? dest.append(createItem<VSpline>(id, objectId, transform, suffix, data))
202 : updateItem<VSpline>(id, objectId, transform, suffix, data, dest.at(i).id);
203 break;
204 case GOType::SplinePath:
205 fromGui ? dest.append(createItem<VSplinePath>(id, objectId, transform, suffix, data))
206 : updateItem<VSplinePath>(id, objectId, transform, suffix, data, dest.at(i).id);
207 break;
208 case GOType::CubicBezier:
209 fromGui ? dest.append(createItem<VCubicBezier>(id, objectId, transform, suffix, data))
210 : updateItem<VCubicBezier>(id, objectId, transform, suffix, data, dest.at(i).id);
211 break;
212 case GOType::CubicBezierPath:
213 fromGui ? dest.append(createItem<VCubicBezierPath>(id, objectId, transform, suffix, data))
214 : updateItem<VCubicBezierPath>(id, objectId, transform, suffix, data, dest.at(i).id);
215 break;
216 case GOType::Unknown:
217 case GOType::Curve:
218 case GOType::Path:
219 case GOType::AllCurves:
220 default:
221 break;
222 }
223QT_WARNING_POP
224 }
225
226 if (not fromGui && parse != Document::FullParse)
227 {
228 doc->UpdateToolData(id, data);
229 }
230
231 if (parse == Document::FullParse)
232 {
233 VDrawTool::AddRecord(id, Tool::Align, doc);
234 VToolAlign *tool = new VToolAlign(doc, data, id, sourcePoint, sourceDirection, targetPoint, targetDirection,
235 angle, suffix, source, dest, typeCreation);
236 scene->addItem(tool);
237 initOperationToolConnections(scene, tool);
238 VAbstractPattern::AddTool(id, tool);
239 doc->IncrementReferens(p->getIdTool());
240 doc->IncrementReferens(a->getIdTool());
241 doc->IncrementReferens(t->getIdTool());
242 doc->IncrementReferens(b->getIdTool());
243 for (int i = 0; i < source.size(); ++i)
244 {
245 doc->IncrementReferens(data->GetGObject(source.at(i).id)->getIdTool());
246 }
247 return tool;
248 }
249 return nullptr;
250}

396DestinationItem VToolAlign::createPoint(quint32 idTool, quint32 idItem, const AlignTransform &t,
397 const QString &suffix, VContainer *data)
398{
399 const QSharedPointer<VPointF> point = data->GeometricObject<VPointF>(idItem);
400 VPointF moved = point->Move(t.length, t.angle, suffix).Rotate(t.origin, t.rotation);
401 moved.setIdObject(idTool);
402
403 DestinationItem item;
404 item.mx = moved.mx();
405 item.my = moved.my();
406 item.showPointName = moved.isShowPointName();
407 item.id = data->AddGObject(new VPointF(moved));
408 return item;
409}

412template <class Item>
413void VToolAlign::registerItem(VContainer *data, quint32 id)
414{
415 if constexpr (std::is_same_v<Item, VArc> || std::is_same_v<Item, VEllipticalArc>)
416 {
417 data->AddArc(data->GeometricObject<Item>(id), id);

419 else if constexpr (std::is_same_v<Item, VSpline> || std::is_same_v<Item, VCubicBezier>)
420 {
421 data->AddSpline(data->GeometricObject<Item>(id), id);
422 }
423 else
424 {
425 data->AddCurveWithSegments(data->GeometricObject<Item>(id), id);
426 }
