26double wrapAngle(double a)
27{
28 while (a > pi) a -= 2 * pi;
29 while (a < -pi) a += 2 * pi;

33QString number(double v)
34{
35 const double r = std::round(v * 2) / 2; // half centimetres
36 QString s = QLocale().toString(std::fabs(r), 'f', std::fabs(r - std::round(r)) > 0.01 ? 1 : 0);
37 return (r > 0 ? QStringLiteral("+") : QStringLiteral("−")) + s;

40// the meridian nearest to theta
41const Tape &nearestMeridian(const QVector<Tape> &list, double theta)
42{
43 int best = 0;
44 double bd = 1e9;
45 for (int i = 0; i < list.size(); ++i)
46 {
47 const double d = std::fabs(wrapAngle(theta - list.at(i).value));
48 if (d < bd) { bd = d; best = i; }
49 }
50 return list.at(best);
51}

65// the angle reached going `arc` cm around the section at height y from angle t0
66double angleAtArc(const sim::DressForm &form, double y, double t0, double arc)
67{
68 if (std::fabs(arc) < 1e-6) return t0;
69 double lo = t0, hi = t0 + (arc > 0 ? pi / 2 : -pi / 2);
70 for (int i = 0; i < 40; ++i)
71 {
72 const double mid = (lo + hi) / 2;
73 const double a = form.arcLength(y, t0, mid);
74 if ((arc > 0 && a < arc) || (arc < 0 && a > arc)) lo = mid;
75 else hi = mid;
76 }
77 return (lo + hi) / 2;
78}

103QString gridLabel(const sim::DressForm &form, double theta, double y)
104{
105 for (const auto &l : landmarks())
106 {
107 const sim::V3 a = form.surfacePoint(l.second.first, l.second.second);
108 const sim::V3 b = form.surfacePoint(theta, y);
109 if ((a - b).len() < 0.35) return l.first;
110 }
111
112 const QVector<Tape> ls = levels();
113 const Tape level = nearestLevel(ls, y);
114 const double dy = y - level.value;
115 QString vertical = level.name;
116 if (std::fabs(dy) >= 0.25) vertical += QStringLiteral(" ") + number(dy);
117
118 const QVector<Tape> ms = meridians();
119 const Tape m = nearestMeridian(ms, theta);
120 double arc = form.arcLength(y, m.value, m.value + wrapAngle(theta - m.value));
121 // + towards the back, - towards the front; on the centre lines + is the right side
122 const bool centre = std::fabs(std::sin(m.value)) < 1e-6;
123 if (!centre && m.value < 0) arc = -arc;
124 QString around = m.name;
125 if (std::fabs(arc) >= 0.25) around += QStringLiteral(" ") + number(arc);
126
127 return vertical + QStringLiteral(" · ") + around + QStringLiteral(" cm");
128}

130void snapToGrid(const sim::DressForm &form, double &theta, double &y)
131{
132 for (const auto &l : landmarks())
133 {
134 const sim::V3 a = form.surfacePoint(l.second.first, l.second.second);
135 const sim::V3 b = form.surfacePoint(theta, y);
136 if ((a - b).len() < 0.9) { theta = l.second.first; y = l.second.second; return; }
137 }
138 const QVector<Tape> ls = levels();
139 const Tape level = nearestLevel(ls, y);
140 const double ys = level.value + std::round((y - level.value) / gridStep) * gridStep;
141 if (std::fabs(y - ys) < 0.7) y = ys;
142
143 const QVector<Tape> ms = meridians();
144 const Tape m = nearestMeridian(ms, theta);
145 const double arc = form.arcLength(y, m.value, m.value + wrapAngle(theta - m.value));
146 const double snapped = std::round(arc / gridStep) * gridStep;
147 if (std::fabs(arc - snapped) < 0.7) theta = wrapAngle(angleAtArc(form, y, m.value, snapped));
148}

150QVector<GridLine> gridLines(const sim::DressForm &form)
151{
152 QVector<GridLine> out;
153 auto ring = [&](double y, bool tape)
154 {
155 GridLine g;
156 g.tape = tape;
157 for (int i = 0; i <= 144; ++i) g.points.push_back(form.surfacePoint(-pi + 2 * pi * i / 144, y));
158 out << g;
159 };
160 auto upright = [&](double theta, bool tape, double top)
161 {
162 GridLine g;
163 g.tape = tape;
164 for (double y = gridBottom; y <= top + 1e-6; y += 0.5) g.points.push_back(form.surfacePoint(theta, y));
165 out << g;
166 };
167
168 const QVector<Tape> ls = levels();
169 for (const Tape &t : ls) ring(t.value, true);
170 for (double y = 101.0 - 12 * gridStep; y <= 140.0; y += gridStep)
171 {
172 if (y < gridBottom) continue;
173 bool onTape = false;
174 for (const Tape &t : ls) if (std::fabs(t.value - y) < 0.3) onTape = true;
175 if (!onTape) ring(y, false);
176 }
177
178 const QVector<Tape> ms = meridians();
179 for (const Tape &t : ms)
180 upright(t.value, true, std::fabs(std::fabs(t.value) - pi / 2) < 1e-6 ? gridTop : gridTop);
181 // the fine uprights: every 2 cm around the bust, between each tape and the next one
182 QVector<double> angles;
183 for (const Tape &t : ms) angles << wrapAngle(t.value);
184 std::sort(angles.begin(), angles.end());
185 for (int i = 0; i < angles.size(); ++i)
186 {
187 const double a0 = angles.at(i);
188 double a1 = i + 1 < angles.size() ? angles.at(i + 1) : angles.first() + 2 * pi;
189 const double span = form.arcLength(121.0, a0, a1);
190 for (double s = gridStep; s < span - gridStep * 0.5; s += gridStep)
191 upright(wrapAngle(angleAtArc(form, 121.0, a0, s)), false, 140.0);
192 }
193 return out;
194}
