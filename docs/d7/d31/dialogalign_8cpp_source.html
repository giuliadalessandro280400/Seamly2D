223//---------------------------------------------------------------------------------------------------------------------
224void DialogAlign::ChosenObject(quint32 id, const SceneObject &type)
225{
226 if (m_stage1 || prepare || type != SceneObject::Point)
227 {
228 return;
229 }
230
231 VisToolAlign *operation = qobject_cast<VisToolAlign *>(vis);
232 SCASSERT(operation != nullptr)
233
234 switch (m_pick)
235 {
236 case 0:
237 if (SetObject(id, ui->sourcePoint_ComboBox, tr("Select the second point of the group line (A)")))
238 {
239 operation->setSourcePointId(id);
240 ++m_pick;
241 }
242 break;
243 case 1:
244 if (id == getSourcePointId())
245 {
246 emit ToolTip(tr("Select a point different from P"));
247 return;
248 }
249 if (SetObject(id, ui->sourceDirection_ComboBox, tr("Select the target point (T)")))
250 {
251 operation->setSourceDirectionId(id);
252 ++m_pick;
253 }
254 break;
255 case 2:
256 if (SetObject(id, ui->targetPoint_ComboBox, tr("Select the second point of the target line (B)")))
257 {
258 operation->setTargetPointId(id);
259 ++m_pick;
260 }
261 break;
262 case 3:
263 if (id == getTargetPointId())
264 {
265 emit ToolTip(tr("Select a point different from T"));
266 return;
267 }
268 if (SetObject(id, ui->targetDirection_ComboBox, QString()))
269 {
270 operation->setTargetDirectionId(id);
271 ++m_pick;
272 prepare = true;
273 }
274 break;
275 default:
276 break;
277 }
278 operation->RefreshGeometry();
279}

281//---------------------------------------------------------------------------------------------------------------------
282void DialogAlign::SelectedObject(bool selected, quint32 id, quint32 tool)
283{
284 Q_UNUSED(tool)
285 if (m_stage1)
286 {
287 auto object = std::find_if(m_objects.begin(), m_objects.end(),
288 [id](const SourceItem &item) { return item.id == id; });
289
290 if (selected)
291 {
292 if (object == m_objects.cend())
293 {
294 SourceItem item;
295 item.id = id;
296 m_objects.append(item);
297 }
298 }
299 else if (object != m_objects.end())
300 {
301 m_objects.erase(object);
302 }
303 }
304}

328//---------------------------------------------------------------------------------------------------------------------
329void DialogAlign::suffixChanged()
330{
331 QLineEdit* edit = qobject_cast<QLineEdit*>(sender());
332 if (edit)
333 {
334 const QString suffix = edit->text();
335 if (suffix.isEmpty())
336 {
337 flagName = false;
338 ChangeColor(ui->suffix_Label, Qt::red);
339 CheckState();
340 return;
341 }
342
343 if (m_suffix != suffix)
344 {
345 QRegularExpression rx(NameRegExp());
346 const QStringList uniqueNames = VContainer::AllUniqueNames();
347 for (int i=0; i < uniqueNames.size(); ++i)
348 {
349 const QString name = uniqueNames.at(i) + suffix;
350 if (not rx.match(name).hasMatch() || not data->IsUnique(name))
351 {
352 flagName = false;
353 ChangeColor(ui->suffix_Label, Qt::red);
354 CheckState();
355 return;
356 }
357 }
358 }
359
360 flagName = true;
361 ChangeColor(ui->suffix_Label, okColor);
362 }
363 CheckState();
364}

420void DialogAlign::updateVisualization()
421{
422 VisToolAlign *operation = qobject_cast<VisToolAlign *>(vis);
423 // Preview while creating (after the points were chosen) or while editing an existing tool.
424 if (operation == nullptr || (m_stage1 && toolId == NULL_ID) || (m_pick < 4 && toolId == NULL_ID))
425 {
426 return;
427 }
428
429 operation->setObjects(sourceToObjects(m_objects));
430 operation->setSourcePointId(getSourcePointId());
431 operation->setSourceDirectionId(getSourceDirectionId());
432 operation->setTargetPointId(getTargetPointId());
433 operation->setTargetDirectionId(getTargetDirectionId());
434 if (m_angleFlag)
435 {
436 operation->setAngle(ui->plainTextEditFormula->toPlainText());
437 }
438 operation->RefreshGeometry();
439}
