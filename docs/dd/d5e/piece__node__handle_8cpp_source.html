40PieceNodeHandle::PieceNodeHandle(quint32 nodeId, QGraphicsItem *parent)
41 : QObject()
42 , QGraphicsRectItem(-handleSize/2, -handleSize/2, handleSize, handleSize, parent)
43 , m_nodeId(nodeId)
44 , m_dragging(false)
45 , m_modified(false)
46 , m_grabOffset()
47{
48 setFlag(QGraphicsItem::ItemIgnoresTransformations, true); // same size at every zoom
49 setAcceptHoverEvents(true);
50 setAcceptedMouseButtons(Qt::LeftButton);
51 setCursor(Qt::SizeAllCursor);
52 setZValue(1000); // above the point marks of the piece (z 100)
53 setToolTip(tr("Drag to move this point of the piece (the draft does not change). "
54 "Double click to put it back."));
55 updateLook(false);
56}

65//---------------------------------------------------------------------------------------------------------------------
66void PieceNodeHandle::updateLook(bool hover)
67{
68 QPen pen(m_modified ? QColor(200, 40, 40) : QColor(30, 90, 200));
69 pen.setCosmetic(true);
70 pen.setWidthF(1.5);
71 setPen(pen);
72 setBrush(hover || m_dragging ? pen.color() : QColor(Qt::white));
73}

75//---------------------------------------------------------------------------------------------------------------------
76void PieceNodeHandle::mousePressEvent(QGraphicsSceneMouseEvent *event)
77{
78 if (event->button() != Qt::LeftButton)
79 {
80 event->ignore();
81 return;
82 }
83 m_dragging = true;
84 m_grabOffset = pos() - parentItem()->mapFromScene(event->scenePos());
85 updateLook(true);
86 event->accept();
87}
88
89//---------------------------------------------------------------------------------------------------------------------
90void PieceNodeHandle::mouseMoveEvent(QGraphicsSceneMouseEvent *event)
91{
92 if (not m_dragging)
93 {
94 return;
95 }
96 const QPointF position = parentItem()->mapFromScene(event->scenePos()) + m_grabOffset;
97 setPos(position);
98 emit moved(m_nodeId, position);
99 event->accept();
100}
101
102//---------------------------------------------------------------------------------------------------------------------
103void PieceNodeHandle::mouseReleaseEvent(QGraphicsSceneMouseEvent *event)
104{
105 m_dragging = false;
106 updateLook(isUnderMouse());
107 event->accept();
108}

111void PieceNodeHandle::mouseDoubleClickEvent(QGraphicsSceneMouseEvent *event)
