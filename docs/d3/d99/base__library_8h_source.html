58/**
59 * Appends the draft blocks of the base at @p path to @p doc:
60 * - every id is moved above the highest id of the pattern;
61 * - a block with an existing name gets a number;
62 * - a point whose name is taken gets a number, and the formulas follow;
63 * - the base's variables the pattern lacks are copied;
64 * - the blocks are moved so the first base point lands on @p start (pattern units).
65 * @p knownNames are the measurement / variable names the pattern already has; any other name used by a
66 * formula is reported in InsertResult::missing.
67 * @p isTaken tells whether a point name is already used in the pattern.
68 */
69InsertResult insertBase(VAbstractPattern *doc, const QString &path, const QPointF &start,
70 const QSet<QString> &knownNames,
71 const std::function<bool(const QString &)> &isTaken);
