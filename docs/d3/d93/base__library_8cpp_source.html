139// Renames the point names inside the built-in variables of @p formula.
140QString renameInFormula(const QString &formula, const QHash<QString, QString> &renames)
141{
142 if (renames.isEmpty())
143 {
144 return formula;
145 }
146 QString result;
147 int last = 0;
148 QRegularExpressionMatchIterator it = identifierRe().globalMatch(formula);
149 while (it.hasNext())
150 {
151 const QRegularExpressionMatch m = it.next();
152 result += formula.mid(last, m.capturedStart() - last);
153 QString token = m.captured();
154 if (isPointVariable(token))
155 {
156 QStringList parts = token.split(QLatin1Char('_'));
157 for (int i = 1; i < parts.size(); ++i)
158 {
159 if (renames.contains(parts.at(i)))
160 {
161 parts[i] = renames.value(parts.at(i));
162 }
163 }
164 token = parts.join(QLatin1Char('_'));
165 }
166 result += token;
167 last = m.capturedEnd();
168 }
169 result += formula.mid(last);
170 return result;
171}
172
173// Names used by @p formula that are neither built-in variables nor functions.
174QStringList namesInFormula(const QString &formula)
175{
176 QStringList names;
177 QRegularExpressionMatchIterator it = identifierRe().globalMatch(formula);
178 while (it.hasNext())
179 {
180 const QRegularExpressionMatch m = it.next();
181 const QString token = m.captured();
182 // a function call: name followed by '('
183 int next = m.capturedEnd();
184 while (next < formula.size() && formula.at(next).isSpace())
185 {
186 ++next;
187 }
188 if (next < formula.size() && formula.at(next) == QLatin1Char('('))
189 {
190 continue;
191 }
192 if (isPointVariable(token) || token == currentLength || token == currentSeamAllowance)
193 {
194 continue;
195 }
196 if (token == QLatin1String("pi") || token == QLatin1String("e") || token == QLatin1String("_pi")
197 || token == QLatin1String("_e"))
198 {
199 continue;
200 }
201 names.append(token);
202 }
203 return names;
204}

244void collect(const QString &root, bool builtIn, QVector<BaseLibrary::Entry> &out)
245{
246 QDirIterator it(root, QStringList{QStringLiteral("*.sm2d"), QStringLiteral("*.val")}, QDir::Files,
247 QDirIterator::Subdirectories);
248 QVector<BaseLibrary::Entry> found;
249 while (it.hasNext())
250 {
251 const QString path = it.next();
252 BaseLibrary::Entry entry;
253 entry.path = path;
254 entry.builtIn = builtIn;
255 entry.title = titleFromFile(path);
256 entry.folder = QDir(root).relativeFilePath(QFileInfo(path).path());
257 if (entry.folder == QLatin1String("."))
258 {
259 entry.folder.clear();
260 }
261 entry.description = descriptionOf(path);
262 found.append(entry);
263 }
264 std::sort(found.begin(), found.end(), [](const BaseLibrary::Entry &a, const BaseLibrary::Entry &b)
265 {
266 if (a.folder != b.folder)
267 {
268 return a.folder.localeAwareCompare(b.folder) < 0;
269 }
270 return a.title.localeAwareCompare(b.title) < 0;
271 });
272 out += found;
273}

328bool BaseLibrary::readBase(const QString &path, QDomDocument &out, QString *error)
329{
330 try
331 {
332 // The converter needs a real file: resources are copied out first.
333 QString source = path;
334 QString temporary;
335 if (path.startsWith(QLatin1Char(':')))
336 {
337 temporary = QDir::temp().filePath(QStringLiteral("seamly2d_base_%1.sm2d")
338 .arg(QCoreApplication::applicationPid()));
339 QFile::remove(temporary);
340 if (!QFile::copy(path, temporary))
341 {
342 if (error)
343 {
344 *error = QObject::tr("Can't read the base %1").arg(path);
345 }
346 return false;
347 }
348 QFile::setPermissions(temporary, QFile::ReadOwner | QFile::WriteOwner);
349 source = temporary;
350 }
351
352 VPatternConverter converter(source);
353 const QString converted = converter.Convert();
354 QFile file(converted);
355 const bool opened = file.open(QIODevice::ReadOnly);
356 const bool parsed = opened && out.setContent(&file);
357 file.close();
358 if (!temporary.isEmpty())
359 {
360 QFile::remove(temporary);
361 }
362 if (!parsed)
363 {
364 if (error)
365 {
366 *error = QObject::tr("Can't read the base %1").arg(path);
367 }
368 return false;
369 }
370 return true;
371 }
372 catch (VException &e)
373 {
374 if (error)
375 {
376 *error = e.ErrorMessage();
377 }
378 return false;
379 }
380}

383BaseLibrary::InsertResult BaseLibrary::insertBase(VAbstractPattern *doc, const QString &path, const QPointF &start,
384 const QSet<QString> &knownNames,
385 const std::function<bool(const QString &)> &isTaken)
386{
387 InsertResult result;
388
389 QDomDocument base;
390 if (!readBase(path, base, &result.error))
391 {
392 return result;
393 }
394
395 const QDomElement baseRoot = base.documentElement();
396 result.baseUnit = baseRoot.firstChildElement(QStringLiteral("unit")).text();
397
398 QVector<QDomElement> blocks;
399 for (QDomElement b = baseRoot.firstChildElement(QStringLiteral("draftBlock")); !b.isNull();
400 b = b.nextSiblingElement(QStringLiteral("draftBlock")))
401 {
402 blocks.append(b);
403 }
404 if (blocks.isEmpty())
405 {
406 result.error = QObject::tr("The base %1 has no draft blocks").arg(path);
407 return result;
408 }
409
410 // 1. New ids above the pattern's highest one.
411 const quint32 offset = maxId(doc->documentElement());
412 QSet<quint32> baseIds;
413 for (const QDomElement &block : blocks)
414 {
415 forEachElement(block, [&baseIds](const QDomElement &e)
416 {
417 bool ok = false;
418 const quint32 id = e.attribute(QStringLiteral("id")).toUInt(&ok);
419 if (ok && id > 0)
420 {
421 baseIds.insert(id);
422 }
423 });
424 }
425 auto remap = [&baseIds, offset](const QString &value, bool *changed) -> QString
426 {
427 bool ok = false;
428 const quint32 id = value.toUInt(&ok);
429 if (ok && baseIds.contains(id))
430 {
431 *changed = true;
432 return QString::number(id + offset);
433 }
434 *changed = false;
435 return value;
436 };
437
438 // 2. Point names that the pattern already uses.
439 QSet<QString> baseNames;
440 for (const QDomElement &block : blocks)
441 {
442 forEachElement(block.firstChildElement(QStringLiteral("calculation")), [&baseNames](const QDomElement &e)
443 {
444 for (const QString &attr : nameAttributes())
445 {
446 const QString name = e.attribute(attr);
447 if (!name.isEmpty())
448 {
449 baseNames.insert(name);
450 }
451 }
452 });
453 }
454 QHash<QString, QString> renames;
455 QStringList conflicts;
456 for (const QString &name : baseNames)
457 {
458 if (isTaken(name))
459 {
460 conflicts.append(name);
461 }
462 }
463 if (!conflicts.isEmpty())
464 {
465 // One number for the whole base, so that A, B, F0 become A1, B1, F01.
466 std::sort(conflicts.begin(), conflicts.end());
467 for (int n = 1; n < 10000; ++n)
468 {
469 bool free = true;
470 for (const QString &name : conflicts)
471 {
472 const QString candidate = name + QString::number(n);
473 if (isTaken(candidate) || baseNames.contains(candidate))
474 {
475 free = false;
476 break;
477 }
478 }
479 if (free)
480 {
481 for (const QString &name : conflicts)
482 {
483 renames.insert(name, name + QString::number(n));
484 result.renamedPoints.append(name + QStringLiteral(" \u2192 ") + name + QString::number(n));
485 }
486 break;
487 }
488 }
489 }
490
491 // 3. Where the blocks go: the first base point lands on start.
492 QPointF shift;
493 {
494 bool found = false;
495 for (const QDomElement &block : blocks)
496 {
497 forEachElement(block.firstChildElement(QStringLiteral("calculation")),
498 [&found, &shift, &start](const QDomElement &e)
499 {
500 if (!found && e.tagName() == QLatin1String("point")
501 && e.attribute(QStringLiteral("type")) == QLatin1String("single"))
502 {
503 shift = start - QPointF(e.attribute(QStringLiteral("x")).toDouble(),
504 e.attribute(QStringLiteral("y")).toDouble());
505 found = true;
506 }
507 });
508 }
509 }
510
511 // 4. Rewrite every element of the blocks.
512 QSet<QString> formulaNames;
513 for (QDomElement &block : blocks)
514 {
515 forEachElement(block, [&](QDomElement e)
516 {
517 const QDomNamedNodeMap attrs = e.attributes();
518 QVector<QPair<QString, QString>> changes;
519 for (int i = 0; i < attrs.count(); ++i)
520 {
521 const QDomAttr attr = attrs.item(i).toAttr();
522 const QString name = attr.name();
523 const QString value = attr.value();
524
525 if (idAttributes().contains(name))
526 {
527 bool changed = false;
528 const QString v = remap(value, &changed);
529 if (changed)
530 {
531 changes.append(qMakePair(name, v));
532 continue;
533 }
534 }
535
536 if (nameAttributes().contains(name) && renames.contains(value))
537 {
538 changes.append(qMakePair(name, renames.value(value)));
539 continue;
540 }
541
542 if (name == QLatin1String("links"))
543 {
544 // shape:point:x,y;... (DXF import ties)
545 QStringList links = value.split(QLatin1Char(';'), Qt::SkipEmptyParts);
546 for (QString &link : links)
547 {
548 QStringList parts = link.split(QLatin1Char(':'));
549 if (parts.size() >= 2)
550 {
551 bool changed = false;
552 parts[1] = remap(parts.at(1), &changed);
553 link = parts.join(QLatin1Char(':'));
554 }
555 }
556 changes.append(qMakePair(name, links.join(QLatin1Char(';'))));
557 continue;
558 }
559
560 if (!notFormulas().contains(name) && !idAttributes().contains(name) && looksLikeFormula(value))
561 {
562 const QString renamed = renameInFormula(value, renames);
563 if (renamed != value)
564 {
565 changes.append(qMakePair(name, renamed));
566 }
567 for (const QString &n : namesInFormula(renamed))
568 {
569 formulaNames.insert(n);
570 }
571 }
572 }
573
574 if (e.tagName() == QLatin1String("point") && e.attribute(QStringLiteral("type")) == QLatin1String("single"))
575 {
576 const double x = e.attribute(QStringLiteral("x")).toDouble() + shift.x();
577 const double y = e.attribute(QStringLiteral("y")).toDouble() + shift.y();
578 changes.append(qMakePair(QStringLiteral("x"), QString::number(x, 'f', 4)));
579 changes.append(qMakePair(QStringLiteral("y"), QString::number(y, 'f', 4)));
580 }
581
582 for (const auto &c : changes)
583 {
584 e.setAttribute(c.first, c.second);
585 }
586
587 if (idTextElements().contains(e.tagName()) && e.firstChildElement().isNull())
588 {
589 bool changed = false;
590 const QString v = remap(e.text().trimmed(), &changed);
591 if (changed)
592 {
593 while (e.hasChildNodes())
594 {
595 e.removeChild(e.firstChild());
596 }
597 e.appendChild(e.ownerDocument().createTextNode(v));
598 }
599 }
600 });
601 }
602
603 // 5. Variables of the base the pattern doesn't have yet.
604 QDomElement docRoot = doc->documentElement();
605 QSet<QString> baseVariables;
606 {
607 const QDomElement baseVars = baseRoot.firstChildElement(QStringLiteral("variables"));
608 QDomElement docVars = docRoot.firstChildElement(QStringLiteral("variables"));
609 QSet<QString> existing;
610 for (QDomElement v = docVars.firstChildElement(); !v.isNull(); v = v.nextSiblingElement())
611 {
612 existing.insert(v.attribute(QStringLiteral("name")));
613 }
614 for (QDomElement v = baseVars.firstChildElement(); !v.isNull(); v = v.nextSiblingElement())
615 {
616 const QString name = v.attribute(QStringLiteral("name"));
617 baseVariables.insert(name);
618 for (const QString &n : namesInFormula(v.attribute(QStringLiteral("formula"))))
619 {
620 formulaNames.insert(n);
621 }
622 if (existing.contains(name) || docVars.isNull())
623 {
624 continue;
625 }
626 docVars.appendChild(doc->importNode(v, true));
627 }
628 }
629
630 // 6. Names the formulas need that nobody provides.
631 for (const QString &n : formulaNames)
632 {
633 if (!knownNames.contains(n) && !baseVariables.contains(n) && !result.missing.contains(n))
634 {
635 result.missing.append(n);
636 }
637 }
638 std::sort(result.missing.begin(), result.missing.end());
639
640 // 7. Block names and the insertion itself.
641 const QStringList existingBlocks = blockNames(*doc);
642 QDomElement after;
643 for (QDomElement b = docRoot.firstChildElement(QStringLiteral("draftBlock")); !b.isNull();
644 b = b.nextSiblingElement(QStringLiteral("draftBlock")))
645 {
646 after = b;
647 }
648 QStringList usedBlocks = existingBlocks;
649 for (const QDomElement &block : blocks)
650 {
651 QDomElement copy = doc->importNode(block, true).toElement();
652 QString name = copy.attribute(QStringLiteral("name"));
653 if (usedBlocks.contains(name))
654 {
655 int n = 2;
656 QString candidate;
657 do
658 {
659 candidate = QStringLiteral("%1 %2").arg(name).arg(n++);
660 } while (usedBlocks.contains(candidate));
661 name = candidate;
662 copy.setAttribute(QStringLiteral("name"), name);
663 }
664 usedBlocks.append(name);
665 result.blocks.append(name);
666
667 if (after.isNull())
668 {
669 docRoot.appendChild(copy);
670 }
671 else
672 {
673 docRoot.insertAfter(copy, after);
674 }
675 after = copy;
676 }
677
678 result.ok = true;
679 return result;
680}

683QString BaseLibrary::saveAsBase(const VAbstractPattern *doc, const QString &title, QString *error)
684{
685 QString fileName = title.trimmed();
686 fileName.replace(QRegularExpression(QStringLiteral("[\\\\/:*?\"<>|]")), QStringLiteral("_"));
687 if (fileName.isEmpty())
688 {
689 fileName = QObject::tr("Base");
690 }
691 const QString path = userFolder() + QLatin1Char('/') + fileName + QStringLiteral(".sm2d");
692
693 QDomDocument copy = doc->cloneNode(true).toDocument();
694 QDomElement root = copy.documentElement();
695 // A base doesn't carry a measurement file: it uses the one of the pattern it goes into.
696 const QDomElement measurements = root.firstChildElement(QStringLiteral("measurements"));
697 if (!measurements.isNull())
698 {
699 QDomElement empty = copy.createElement(QStringLiteral("measurements"));
700 root.replaceChild(empty, measurements);
701 }
702
703 QFile file(path);
704 if (!file.open(QIODevice::WriteOnly | QIODevice::Truncate))
705 {
706 if (error)
707 {
708 *error = file.errorString();
709 }
710 return QString();
711 }
712 QTextStream out(&file);
713 out << copy.toString(4);
714 return path;
