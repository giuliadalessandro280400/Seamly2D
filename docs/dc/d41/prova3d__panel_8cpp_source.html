145void Prova3dView::paintEvent(QPaintEvent *)
146{
147 QPainter p(this);
148 p.setRenderHint(QPainter::Antialiasing, true);
149 p.fillRect(rect(), paperLight);
150
151 // a thin frame, like a photo pasted on the page
152 p.setPen(QPen(rule, 1));
153 p.drawRect(rect().adjusted(0, 0, -1, -1));
154
155 const double cy = std::cos(m_yaw), sy = std::sin(m_yaw), cp = std::cos(m_pitch), sp = std::sin(m_pitch);
156 const double scale = std::min(width() / 70.0, height() / 92.0) * m_zoom;
157 const QPointF centre(width() / 2.0, height() / 2.0);
158 const double midY = 116.0;
159 auto view = [&](const sim::V3 &v) {
160 const double x = v.x * cy + v.z * sy;
161 const double z0 = -v.x * sy + v.z * cy;
162 const double y0 = v.y - midY;
163 const double y = y0 * cp - z0 * sp;
164 const double z = y0 * sp + z0 * cp;
165 return sim::V3(x, y, z);
166 };
167 auto screen = [&](const sim::V3 &v) { return QPointF(centre.x() + v.x * scale, centre.y() - v.y * scale); };
168
169 const sim::V3 light = [] { sim::V3 l(-0.35, 0.55, 1.0); return l * (1.0 / l.len()); }();
170
171 struct Item { double depth; int kind; int index; }; // kind 0 form, 1 garment, 2 border
172 std::vector<Item> items;
173 std::vector<sim::V3> fv(m_formVerts.size());
174 for (size_t i = 0; i < m_formVerts.size(); ++i) fv[i] = view(m_formVerts[i]);
175 for (size_t t = 0; t < m_formTris.size(); ++t)
176 {
177 const auto &tr = m_formTris[t];
178 const sim::V3 n = sim::cross(fv[tr[1]] - fv[tr[0]], fv[tr[2]] - fv[tr[0]]);
179 if (n.z <= 0) continue; // the form is closed: its back faces are hidden
180 items.push_back({(fv[tr[0]].z + fv[tr[1]].z + fv[tr[2]].z) / 3 - 0.2, 0, int(t)});
181 }
182
183 std::vector<sim::V3> gv;
184 std::vector<std::array<int, 3>> gt;
185 std::vector<int> gp;
186 std::vector<std::pair<int, int>> border;
187 std::vector<sim::V3> gn;
188 if (m_garment)
189 {
190 const std::vector<sim::V3> x = m_garment->shownPositions();
191 gt = m_garment->meshTriangles();
192 gp = m_garment->meshTrianglePiece();
193 gv.resize(x.size());
194 for (size_t i = 0; i < x.size(); ++i) gv[i] = view(x[i]);
195 std::map<std::pair<int, int>, int> edgeCount;
196 for (const auto &tr : gt)
197 for (int e = 0; e < 3; ++e)
198 {
199 const int a = tr[e], b = tr[(e + 1) % 3];
200 ++edgeCount[{std::min(a, b), std::max(a, b)}];
201 }
202 for (const auto &kv : edgeCount)
203 if (kv.second == 1) border.push_back(kv.first);
204 // smooth shading: normals averaged on the vertices
205 gn.assign(gv.size(), sim::V3());
206 for (const auto &tr : gt)
207 {
208 sim::V3 n = sim::cross(gv[tr[1]] - gv[tr[0]], gv[tr[2]] - gv[tr[0]]);
209 const double l = n.len();
210 if (l < 1e-12) continue;
211 n = n * (1.0 / l);
212 // the cloth has two sides: keep the normals of a vertex on one side
213 for (int k : tr) gn[k] += (gn[k].dot(n) < 0 ? n * -1.0 : n);
214 }
215 for (sim::V3 &n : gn) { const double l = n.len(); if (l > 1e-12) n = n * (1.0 / l); }
216 for (size_t t = 0; t < gt.size(); ++t)
217 {
218 const auto &tr = gt[t];
219 items.push_back({(gv[tr[0]].z + gv[tr[1]].z + gv[tr[2]].z) / 3 + 0.25, 1, int(t)});
220 }
221 for (size_t e = 0; e < border.size(); ++e)
222 items.push_back({(gv[border[e].first].z + gv[border[e].second].z) / 2 + 0.35, 2, int(e)});
223 }
224 std::sort(items.begin(), items.end(), [](const Item &a, const Item &b) { return a.depth < b.depth; });
225
226 const QColor pieceColors[2] = {QColor(0xFD, 0xFC, 0xF8), QColor(0xF0, 0xEC, 0xE2)};
227 QPen inkPen(ink, 1.1);
228 inkPen.setCapStyle(Qt::RoundCap);
229 for (const Item &it : items)
230 {
231 if (it.kind == 2)
232 {
233 p.setPen(inkPen);
234 p.drawLine(screen(gv[border[it.index].first]), screen(gv[border[it.index].second]));
235 continue;
236 }
237 const bool isForm = it.kind == 0;
238 const auto &tr = isForm ? m_formTris[it.index] : gt[it.index];
239 const std::vector<sim::V3> &vv = isForm ? fv : gv;
240 sim::V3 n = sim::cross(vv[tr[1]] - vv[tr[0]], vv[tr[2]] - vv[tr[0]]);
241 const double nl = n.len();
242 if (nl < 1e-12) continue;
243 n = n * (1.0 / nl);
244 if (!isForm)
245 {
246 sim::V3 s = gn[tr[0]] + gn[tr[1]] + gn[tr[2]];
247 if (s.len() > 1e-9) n = s * (1.0 / s.len());
248 }
249 const double shade = 0.62 + 0.38 * std::fabs(n.dot(light));
250 QColor base = isForm ? formColor : pieceColors[std::min(1, std::max(0, gp[it.index]))];
251 QColor c = QColor::fromRgbF(base.redF() * shade, base.greenF() * shade, base.blueF() * shade);
252 p.setPen(QPen(c, 0.6));
253 p.setBrush(c);
254 const QPointF poly[3] = {screen(vv[tr[0]]), screen(vv[tr[1]]), screen(vv[tr[2]])};
255 p.drawPolygon(poly, 3);
256 }
257
258 // the seams that don't close: a red pencil line across the gap and how much is missing
259 if (m_garment)
260 {
261 QFont hand;
262 hand.setFamilies(handFamilies());
263 hand.setPointSizeF(11);
264 p.setFont(hand);
265 QPen red(redPencil, 1.2, Qt::DashLine);
266 QList<QRectF> taken;
267 const QFontMetricsF fm(hand);
268 for (const sim::Garment::OpenSeam &os : m_garment->openSeams())
269 {
270 if (os.pairs.empty() || os.maxGap < 0.3) continue;
271 p.setPen(red);
272 QPointF labelAt;
273 double best = -1;
274 for (const auto &pr : os.pairs)
275 {
276 const QPointF a = screen(view(pr.first)), b = screen(view(pr.second));
277 p.drawLine(a, b);
278 const double g = (pr.second - pr.first).len();
279 if (g > best) { best = g; labelAt = (a + b) / 2; }
280 }
281 p.setPen(redPencil);
282 const QString text = QStringLiteral("%1 cm").arg(QLocale().toString(os.maxGap, 'f', 1));
283 // to the outside of the figure, and never on top of another note
284 const bool right = labelAt.x() >= centre.x();
285 QRectF box = fm.boundingRect(text);
286 box.moveTopLeft(labelAt + QPointF(right ? 8 : -8 - box.width(), -box.height() / 2));
287 for (int guard = 0; guard < 20; ++guard)
288 {
289 bool clash = false;
290 for (const QRectF &r : taken) if (r.intersects(box)) { clash = true; break; }
291 if (!clash) break;
292 box.translate(0, box.height() * 0.9);
293 }
294 taken.append(box);
295 p.drawText(box, Qt::AlignLeft | Qt::AlignVCenter, text);
296 }
297 }

306void Prova3dView::mouseMoveEvent(QMouseEvent *event)
307{
308 if (!(event->buttons() & Qt::LeftButton)) return;
309 const QPoint d = event->pos() - m_lastPos;
310 m_lastPos = event->pos();
311 m_yaw += d.x() * 0.012;
312 m_pitch = std::max(-0.9, std::min(0.9, m_pitch + d.y() * 0.008));
313 update();
314}

490void Prova3dPanel::runPhase(int phase)
491{
492 if (m_worker != nullptr) return;
493 setBusy(true);
494 m_notes->clear();
495 m_status->setText(phase == 1 ? tr("Putting it on and fitting in paper…") : tr("Pins out, the cloth drops…"));
496
497 const double width = m_width->value() / 100.0;
498 const sim::Material cloth = fabric();
499 const bool needForm = !m_form;
500 QPointer<Prova3dView> view = m_view;
501
502 m_worker = QThread::create([this, phase, width, cloth, needForm, view]()
503 {
504 QElapsedTimer timer;
505 timer.start();
506 if (needForm)
507 {
508 m_form = std::make_unique<sim::DressForm>(sim::DressForm::standard42());
509 const sim::DressForm *form = m_form.get();
510 QMetaObject::invokeMethod(this, [view, form]() { if (view) view->setForm(form); }, Qt::QueuedConnection);
511 }
512 auto repaint = [view]() { QMetaObject::invokeMethod(view, "update", Qt::QueuedConnection); };
513
514 const double seamForce = 5e-4;
515 QString summary;
516 bool pinsKept = false;
517 if (phase == 1)
518 {
519 const sim::Demo demo = sim::demoBodice(width);
520 auto garment = std::make_shared<sim::Garment>(demo.pieces, demo.seams, *m_form, 1.5, 23.0);
521 garment->setProgress(repaint);
522 std::vector<QString> names;
523 for (const sim::SeamInput &s : demo.seams) names.push_back(QString::fromStdString(s.name));
524 m_seamNames = names;
525 m_garment = garment;
526 QMetaObject::invokeMethod(this, [this, view, garment, names]()
527 {
528 if (view) view->setGarment(garment, names);
529 }, Qt::QueuedConnection);
530
531 // putting it on: soft and slippery
532 sim::Material dressing = sim::Material::fabricDefault();
533 dressing.friction = 0;
534 dressing.lockAngleDeg = 45;
535 garment->setMaterial(dressing);
536 garment->setLatchSeams(false);
537 garment->settle(0, 4000, 0.002, seamForce * 10);
538
539 // paper: pinned at centre front and back, assembled, then its weight
540 garment->setMaterial(sim::Material::paperDefault());
541 garment->setLatchSeams(true);
542 std::vector<sim::PinInput> pins;
543 for (const auto &pp : demo.pins) pins.push_back({pp.first, pp.second, garment->nodeAt(pp.first, pp.second)});
544 garment->setPins(pins);
545 garment->settle(0, 4000, 0.002, seamForce);
546 garment->settle(981, 4000, 0.002, seamForce);
547 summary = tr("Paper fitting: %1 s").arg(QLocale().toString(timer.elapsed() / 1000.0, 'f', 1));
548 }
549 else if (m_garment)
550 {
551 // a seam still open can't hold the garment: the pins stay, as they would on the form
552 bool open = false;
553 for (const sim::Garment::OpenSeam &os : m_garment->openSeams())
554 {
555 const sim::SeamInput &si = m_garment->seams()[size_t(os.seam)];
556 if (si.pieceA != si.pieceB && os.sewnFraction < 0.5) open = true;
557 }
558 if (!open) m_garment->releasePins();
559 pinsKept = open;
560 m_garment->setMaterial(cloth);
561 m_garment->settle(981, 6000, 0.002, seamForce);
562 summary = tr("Drop: %1 s").arg(QLocale().toString(timer.elapsed() / 1000.0, 'f', 1));
563 if (pinsKept) summary += QStringLiteral(" \u2014 ") + tr("some seams are open, the pins stay in");
564 }
565 emit phaseFinished(summary);
