104bool MoveSpline::mergeWith(const QUndoCommand *command)
105{
106 const MoveSpline *moveCommand = static_cast<const MoveSpline *>(command);
107 SCASSERT(moveCommand != nullptr)
108 const quint32 id = moveCommand->getSplineId();
109
110 if (id != nodeId)

126void MoveSpline::Do(const VSpline &spl)
127{
128 QDomElement domElement = doc->elementById(nodeId, VAbstractPattern::TagSpline);
129 if (domElement.isElement())
130 {
131 doc->SetAttribute(domElement, AttrPoint1, spl.GetP1().id());
132 doc->SetAttribute(domElement, AttrPoint4, spl.GetP4().id());
133 doc->SetAttribute(domElement, AttrAngle1, spl.GetStartAngleFormula());
134 doc->SetAttribute(domElement, AttrAngle2, spl.GetEndAngleFormula());
135 doc->SetAttribute(domElement, AttrLength1, spl.GetC1LengthFormula());
136 doc->SetAttribute(domElement, AttrLength2, spl.GetC2LengthFormula());
137
138 // Proportional handles: store the dragged lengths relative to the reference chord.
139 const qreal refChord = domElement.attribute(ProportionalHandles::attrRefChord(), QStringLiteral("0")).toDouble();
140 if (refChord > 0)
141 {
142 const qreal currentChord = ProportionalHandles::chord(static_cast<QPointF>(spl.GetP1()),
143 static_cast<QPointF>(spl.GetP4()));
144 if (ProportionalHandles::isNumber(spl.GetC1LengthFormula()))
145 {
146 doc->SetAttribute(domElement, AttrLength1, ProportionalHandles::toStored(
147 qApp->fromPixel(spl.GetC1Length()), currentChord, refChord));
148 }
149 if (ProportionalHandles::isNumber(spl.GetC2LengthFormula()))
150 {
151 doc->SetAttribute(domElement, AttrLength2, ProportionalHandles::toStored(
152 qApp->fromPixel(spl.GetC2Length()), currentChord, refChord));
153 }
154 }
155
156 emit NeedLiteParsing(Document::LiteParse);
157 }
158 else
159 {
160 qCDebug(vUndo, "Can't find spline with id = %u.", nodeId);
161 return;
162 }
163}
