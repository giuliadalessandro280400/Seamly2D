41/**
42 * Proportional handles
43 *
44 * Normally the length of a curve handle is a fixed value. When the pattern is graded the end points move but the
45 * handles keep their length, so a curve drawn nicely in one size looks too round or too square in the others.
46 *
47 * With proportional handles a curve remembers its chord (the distance between the two end points of a segment) in
48 * the size where it was edited. The stored handle lengths belong to that reference chord; in every other size they
49 * are multiplied by currentChord / referenceChord. The handle angles do not change.
50 *
51 * Simple curves store the reference in the attribute "refChord", curve paths store one value per segment in
52 * "refChords" (separated by ';'). Values are in pattern units.
53 */

68/** @brief chord returns the distance between two points in pattern units. */
69inline qreal chord(const QPointF &p1, const QPointF &p2)
70{
71 return qApp->fromPixel(QLineF(p1, p2).length());
72}
73
74/** @brief factor returns the multiplier for the handles of a segment, 1 when the feature is off. */
75inline qreal factor(qreal currentChord, qreal referenceChord)
76{
77 if (referenceChord <= 0 || qFuzzyIsNull(referenceChord) || currentChord <= 0)
78 {
79 return 1.0;
80 }
81 return currentChord / referenceChord;
82}

84/** @brief isNumber true if a formula is a single number that can be safely rewritten. */
85inline bool isNumber(const QString &formula)
86{
87 return qmu::QmuTokenParser::IsSingle(formula);
88}

90/** @brief toStored converts an actual handle length (pattern units) to the value stored for the reference chord. */
91inline QString toStored(qreal actualLength, qreal currentChord, qreal referenceChord)
92{
93 return QString::number(actualLength / factor(currentChord, referenceChord));
94}

96inline QVector<qreal> parseList(const QString &text)
97{
98 QVector<qreal> values;
99 const QStringList parts = text.split(QLatin1Char(';'), Qt::SkipEmptyParts);
100 for (const QString &part : parts)
101 {
102 bool ok = false;
103 const qreal value = part.toDouble(&ok);
104 values.append(ok ? value : 0);
105 }
106 return values;
107}
108
109inline QString formatList(const QVector<qreal> &values)
110{
111 QStringList parts;
112 for (const qreal value : values)
113 {
114 parts.append(QString::number(value));
115 }
116 return parts.join(QLatin1Char(';'));
117}
