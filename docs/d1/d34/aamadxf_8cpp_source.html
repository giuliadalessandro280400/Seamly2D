67QVector<Pair> readPairs(const QString &path, QString &error)
68{
69 QVector<Pair> pairs;
70 QFile file(path);
71 if (not file.open(QIODevice::ReadOnly))
72 {
73 error = QObject::tr("Can't open the file.");
74 return pairs;
75 }
76 const QByteArray data = file.readAll();
77 QString content = QString::fromUtf8(data);
78 if (content.contains(QChar::ReplacementCharacter))
79 {
80 content = QString::fromLatin1(data); // older files use the local code page
81 }
82 const QStringList lines = content.split(QLatin1Char('\n'));
83 for (int i = 0; i + 1 < lines.size(); i += 2)
84 {
85 bool ok = false;
86 const int code = lines.at(i).trimmed().toInt(&ok);
87 if (not ok)
88 {
89 if (lines.at(i).trimmed().isEmpty())
90 {
91 --i; // tolerate empty lines
92 continue;
93 }
94 error = QObject::tr("This is not a text DXF file (binary DXF is not supported).");
95 pairs.clear();
96 return pairs;
97 }
98 QString value = lines.at(i + 1);
99 if (value.endsWith(QLatin1Char('\r')))
100 {
101 value.chop(1);
102 }
103 pairs.append({code, value.trimmed()});
104 }
105 return pairs;
106}

109// Reads the entity that starts at pairs[i] (a code 0); returns the index of the next code 0.
110int readEntity(const QVector<Pair> &pairs, int i, Entity &entity)
111{
112 entity.type = pairs.at(i).value;
113 ++i;
114 double x = 0, y = 0;
115 bool haveX = false;
116 double x2 = 0, y2 = 0;
117 bool haveSecond = false;
118 while (i < pairs.size() && pairs.at(i).code != 0)
119 {
120 const Pair &p = pairs.at(i);
121 switch (p.code)
122 {
123 case 8: entity.layer = p.value; break;
124 case 1: entity.text = p.value; break;
125 case 2: entity.blockName = p.value; break;
126 case 10:
127 if (entity.type == QLatin1String("LWPOLYLINE") && haveX)
128 {
129 entity.points.append(QPointF(x, y));
130 }
131 x = p.value.toDouble();
132 haveX = true;
133 break;
134 case 20: y = p.value.toDouble(); break;
135 case 11: x2 = p.value.toDouble(); haveSecond = true; break;
136 case 21: y2 = p.value.toDouble(); break;
137 case 41: entity.scaleX = p.value.toDouble(); break;
138 case 42: entity.scaleY = p.value.toDouble(); break;
139 case 50: entity.rotation = p.value.toDouble(); break;
140 case 70: entity.closed = (p.value.toInt() & 1) != 0; break;
141 default: break;
142 }
143 ++i;
144 }
145 if (haveX)
146 {
147 entity.points.append(QPointF(x, y));
148 }
149 if (haveSecond)
150 {
151 entity.points.append(QPointF(x2, y2));
152 }
153 return i;
154}

156//---------------------------------------------------------------------------------------------------------------------
157// Reads entities from pairs[i] until one of the end markers; POLYLINE + VERTEX + SEQEND become one entity.
158int readEntities(const QVector<Pair> &pairs, int i, const QStringList &endMarkers, QVector<Entity> &entities)
159{
160 while (i < pairs.size())
161 {
162 if (pairs.at(i).code != 0)
163 {
164 ++i;
165 continue;
166 }
167 if (endMarkers.contains(pairs.at(i).value))
168 {
169 return i;
170 }
171 Entity entity;
172 i = readEntity(pairs, i, entity);
173 if (entity.type == QLatin1String("POLYLINE"))
174 {
175 entity.points.clear(); // the position of a POLYLINE is not a vertex
176 while (i < pairs.size() && pairs.at(i).code == 0 && pairs.at(i).value == QLatin1String("VERTEX"))
177 {
178 Entity vertex;
179 i = readEntity(pairs, i, vertex);
180 if (not vertex.points.isEmpty())
181 {
182 entity.points.append(vertex.points.first());
183 }
184 }
185 if (i < pairs.size() && pairs.at(i).value == QLatin1String("SEQEND"))
186 {
187 Entity end;
188 i = readEntity(pairs, i, end);
189 }
190 }
191 entities.append(entity);
192 }
193 return i;
194}

197QPointF transformed(const QPointF &p, const Entity &insert)
198{
199 const double a = qDegreesToRadians(insert.rotation);
200 const double x = p.x() * insert.scaleX;
201 const double y = p.y() * insert.scaleY;
202 const QPointF base = insert.points.isEmpty() ? QPointF() : insert.points.first();
203 return base + QPointF(x * std::cos(a) - y * std::sin(a), x * std::sin(a) + y * std::cos(a));
204}

207bool near(const QPointF &a, const QPointF &b, double tolerance)
208{
209 return QLineF(a, b).length() <= tolerance;
210}

213AamaPiece makePiece(const Block &block, const Entity &insert, double unit)
214{
215 AamaPiece piece;
216 piece.name = block.name;
217
218 const auto map = [&](const QPointF &p) { return transformed(p, insert) * unit; };
219
220 // The outline: the polyline of layer 1 with most vertices
221 const Entity *outline = nullptr;
222 QVector<QPointF> turnPoints;
223 QVector<QPointF> curvePoints;
224 for (const Entity &e : block.entities)
225 {
226 const bool polyline = e.type == QLatin1String("POLYLINE") || e.type == QLatin1String("LWPOLYLINE");
227 if (polyline && e.layer == QLatin1String("1") && (outline == nullptr
228 || e.points.size() > outline->points.size()))
229 {
230 outline = &e;
231 }
232 else if (e.type == QLatin1String("POINT") && not e.points.isEmpty())
233 {
234 if (e.layer == QLatin1String("2"))
235 {
236 turnPoints.append(map(e.points.first()));
237 }
238 else if (e.layer == QLatin1String("3"))
239 {
240 curvePoints.append(map(e.points.first()));
241 }
242 else if (e.layer == QLatin1String("4"))
243 {
244 piece.notches.append(map(e.points.first()));
245 }
246 }
247 else if (e.type == QLatin1String("LINE") && e.points.size() >= 2)
248 {
249 if (e.layer == QLatin1String("4"))
250 {
251 piece.notches.append(map(e.points.first()));
252 }
253 else if (e.layer == QLatin1String("7"))
254 {
255 piece.grain = {map(e.points.at(0)), map(e.points.at(1))};
256 }
257 else if (e.layer == QLatin1String("8") || e.layer == QLatin1String("11"))
258 {
259 piece.internals.append({map(e.points.at(0)), map(e.points.at(1))});
260 }
261 }
262 else if (polyline && (e.layer == QLatin1String("8") || e.layer == QLatin1String("11")) && e.points.size() >= 2)
263 {
264 QVector<QPointF> line;
265 for (const QPointF &p : e.points)
266 {
267 line.append(map(p));
268 }
269 piece.internals.append(line);
270 }
271 else if ((e.type == QLatin1String("TEXT") || e.type == QLatin1String("MTEXT")) && not e.text.isEmpty())
272 {
273 piece.texts.append(e.text);
274 if (e.text.startsWith(QLatin1String("Piece Name:"), Qt::CaseInsensitive))
275 {
276 piece.name = e.text.section(QLatin1Char(':'), 1).trimmed();
277 }
278 }
279 }
280
281 if (outline == nullptr || outline->points.size() < 3)
282 {
283 return AamaPiece();
284 }
285
286 for (const QPointF &p : outline->points)
287 {
288 piece.boundary.append(map(p));
289 }
290 // a closed outline may repeat its first vertex at the end
291 if (piece.boundary.size() > 3 && near(piece.boundary.first(), piece.boundary.last(), 0.01))
292 {
293 piece.boundary.removeLast();
294 }
295
296 const double tolerance = 0.05; // mm
297 const bool marked = not turnPoints.isEmpty() || not curvePoints.isEmpty();
298 const int n = piece.boundary.size();
299 for (int i = 0; i < n; ++i)
300 {
301 const QPointF &p = piece.boundary.at(i);
302 bool turn = false;
303 if (marked)
304 {
305 for (const QPointF &t : turnPoints)
306 {
307 turn = turn || near(p, t, tolerance);
308 }
309 if (not turn)
310 {
311 bool curve = false;
312 for (const QPointF &c : curvePoints)
313 {
314 curve = curve || near(p, c, tolerance);
315 }
316 turn = not curve; // a vertex that is not marked as a curve point is a corner
317 }
318 }
319 else
320 {
321 // No marks: a vertex is a corner where the outline turns sharply
322 const QPointF &prev = piece.boundary.at((i + n - 1) % n);
323 const QPointF &next = piece.boundary.at((i + 1) % n);
324 const double a1 = QLineF(prev, p).angle();
325 const double a2 = QLineF(p, next).angle();
326 double change = std::fmod(std::fabs(a2 - a1), 360.0);
327 change = qMin(change, 360.0 - change);
328 turn = change > 30;
329 }
330 piece.turn.append(turn);
331 }
332 if (not piece.turn.contains(true))
333 {
334 piece.turn[0] = true;
335 }
336 return piece;
337}

341AamaFile readAama(const QString &path, double unitMm)
342{
343 AamaFile file;
344 const QVector<Pair> pairs = readPairs(path, file.error);
345 if (not file.error.isEmpty())
346 {
347 return file;
348 }
349
350 QHash<QString, Block> blocks;
351 QVector<Entity> inserts;
352 QVector<Entity> loose; // entities drawn directly, not in a block
353 int insUnits = -1;
354 QString unitsText;
355
356 int i = 0;
357 while (i < pairs.size())
358 {
359 if (pairs.at(i).code == 0 && pairs.at(i).value == QLatin1String("SECTION") && i + 1 < pairs.size())
360 {
361 const QString section = pairs.at(i + 1).value;
362 i += 2;
363 if (section == QLatin1String("HEADER"))
364 {
365 while (i < pairs.size() && not (pairs.at(i).code == 0 && pairs.at(i).value == QLatin1String("ENDSEC")))
366 {
367 if (pairs.at(i).code == 9 && pairs.at(i).value == QLatin1String("$INSUNITS") && i + 1 < pairs.size())
368 {
369 insUnits = pairs.at(i + 1).value.toInt();
370 }
371 ++i;
372 }
373 }
374 else if (section == QLatin1String("BLOCKS"))
375 {
376 while (i < pairs.size() && not (pairs.at(i).code == 0 && pairs.at(i).value == QLatin1String("ENDSEC")))
377 {
378 if (pairs.at(i).code == 0 && pairs.at(i).value == QLatin1String("BLOCK"))
379 {
380 Entity header;
381 i = readEntity(pairs, i, header);
382 Block block;
383 block.name = header.blockName;
384 i = readEntities(pairs, i, {QStringLiteral("ENDBLK"), QStringLiteral("ENDSEC")}, block.entities);
385 if (i < pairs.size() && pairs.at(i).value == QLatin1String("ENDBLK"))
386 {
387 Entity end;
388 i = readEntity(pairs, i, end);
389 }
390 blocks.insert(block.name, block);
391 }
392 else
393 {
394 ++i;
395 }
396 }
397 }
398 else if (section == QLatin1String("ENTITIES"))
399 {
400 QVector<Entity> entities;
401 i = readEntities(pairs, i, {QStringLiteral("ENDSEC")}, entities);
402 for (const Entity &e : entities)
403 {
404 if (e.type == QLatin1String("INSERT"))
405 {
406 inserts.append(e);
407 }
408 else
409 {
410 if (e.type == QLatin1String("TEXT") && e.text.contains(QLatin1String("UNITS"), Qt::CaseInsensitive))
411 {
412 unitsText = e.text;
413 }
414 loose.append(e);
415 }
416 }
417 }
418 }
419 else
420 {
421 ++i;
422 }
423 }
424
425 // Units
426 double unit = 1; // mm
427 if (insUnits == 1 || unitsText.contains(QLatin1String("ENGLISH"), Qt::CaseInsensitive))
428 {
429 unit = 25.4;
430 file.units = QStringLiteral("inch");
431 }
432 else if (insUnits == 5)
433 {
434 unit = 10;
435 file.units = QStringLiteral("cm");
436 }
437 else if (insUnits == 6)
438 {
439 unit = 1000;
440 file.units = QStringLiteral("m");
441 }
442 else if (insUnits == 4 || unitsText.contains(QLatin1String("METRIC"), Qt::CaseInsensitive))
443 {
444 file.units = QStringLiteral("mm");
445 }
446 if (unitMm > 0)
447 {
448 unit = unitMm;
449 }
450
451 for (const Entity &insert : inserts)
452 {
453 if (not blocks.contains(insert.blockName))
454 {
455 continue;
456 }
457 AamaPiece piece = makePiece(blocks.value(insert.blockName), insert, unit);
458 if (not piece.boundary.isEmpty())
459 {
460 file.pieces.append(piece);
461 }
462 }
463
464 // A file with the outlines drawn directly (no blocks): one piece
465 if (file.pieces.isEmpty() && not loose.isEmpty())
466 {
467 Block block;
468 block.name = QStringLiteral("PIECE");
469 block.entities = loose;
470 Entity identity;
471 identity.points = {QPointF()};
472 AamaPiece piece = makePiece(block, identity, unit);
473 if (not piece.boundary.isEmpty())
474 {
475 file.pieces.append(piece);
476 }
477 }
478
479 if (file.pieces.isEmpty())
480 {
481 file.error = QObject::tr("No pattern piece found: an AAMA file has the outline of each piece on layer 1.");
482 }
483 return file;
484}
