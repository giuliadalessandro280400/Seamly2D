37/**
38 * @brief The MovePieceNode class changes the manual offset of one point of a pattern piece.
39 * The offset is stored on the <node> of the piece, in pattern units. Consecutive moves of the same point merge into
40 * one undo step.
41 */

55 /** @brief writeOffset stores an offset on a piece node; returns false if the node was not found. */

57 static QPointF readOffset(VAbstractPattern *doc, quint32 pieceId, quint32 nodeId);

59private:

63 QPointF m_newOffset;
