44// a guess of the centre line: the two nodes furthest apart on the outermost straight vertical edge
45void guessCenter(const VContainer *data, quint32 pieceId, quint32 &c1, quint32 &c2)
46{
47 const QVector<prova3d::NodeInfo> nodes = prova3d::pieceNodes(data, pieceId);
48 if (nodes.size() < 2) return;
49 QVector<QPointF> pos;
50 for (const prova3d::NodeInfo &n : nodes) pos << data->GeometricObject<VPointF>(n.id)->toQPointF();
51 int minX = 0;
52 for (int i = 1; i < pos.size(); ++i) if (pos[i].x() < pos[minX].x()) minX = i;
53 int other = -1;
54 double span = 0;
55 const double tol = ToPixel(0.5, Unit::Cm);
56 for (int i = 0; i < pos.size(); ++i)
57 {
58 if (i == minX || std::fabs(pos[i].x() - pos[minX].x()) > tol) continue;
59 const double d = std::fabs(pos[i].y() - pos[minX].y());
60 if (d > span) { span = d; other = i; }
61 }
62 if (other < 0) return;
63 c1 = nodes[pos[minX].y() < pos[other].y() ? minX : other].id;
64 c2 = nodes[pos[minX].y() < pos[other].y() ? other : minX].id;
65}

214 m_pieces->setCellWidget(row, PMode, mode);

221 m_pieces->setCellWidget(row, PC2, c2);

227 pos->addItem(prova3d::positionName(q), int(q));
228 if (q == p.position) pos->setCurrentIndex(pos->count() - 1);

230 m_pieces->setCellWidget(row, PPos, pos);

234void Prova3dSetupDialog::addSeamRow(const prova3d::SeamSetup &s)
235{
236 const int row = m_seams->rowCount();
237 m_seams->insertRow(row);
238 auto *pa = pieceCombo(s.pieceA);
239 auto *pb = pieceCombo(s.pieceB);
240 auto *a1 = new QComboBox, *a2 = new QComboBox, *b1 = new QComboBox, *b2 = new QComboBox;
241 fillNodes(a1, pa->currentData().toUInt(), s.a1);
242 fillNodes(a2, pa->currentData().toUInt(), s.a2);
243 fillNodes(b1, pb->currentData().toUInt(), s.b1);
244 fillNodes(b2, pb->currentData().toUInt(), s.b2);
245 connect(pa, QOverload<int>::of(&QComboBox::currentIndexChanged), this, [this, pa, a1, a2]()
246 {
247 fillNodes(a1, pa->currentData().toUInt(), 0);
248 fillNodes(a2, pa->currentData().toUInt(), 0);
249 });
250 connect(pb, QOverload<int>::of(&QComboBox::currentIndexChanged), this, [this, pb, b1, b2]()
251 {
252 fillNodes(b1, pb->currentData().toUInt(), 0);

254 });
255 m_seams->setCellWidget(row, SA, pa);
256 m_seams->setCellWidget(row, SA1, a1);
257 m_seams->setCellWidget(row, SA2, a2);

294prova3d::Setup Prova3dSetupDialog::setup() const
295{
296 prova3d::Setup s;
297 for (int row = 0; row < m_pieces->rowCount(); ++row)
298 {
299 prova3d::PieceSetup p;
300 p.id = m_pieces->item(row, PName)->data(Qt::UserRole).toUInt();
301 if (QWidget *holder = m_pieces->cellWidget(row, PUse))
302 if (auto *use = holder->findChild<QCheckBox *>()) p.use = use->isChecked();
303 p.mode = prova3d::Mode(cell<QComboBox>(m_pieces, row, PMode)->currentData().toInt());
304 p.center1 = cell<QComboBox>(m_pieces, row, PC1)->currentData().toUInt();
305 p.center2 = cell<QComboBox>(m_pieces, row, PC2)->currentData().toUInt();
306 p.position = prova3d::Position(cell<QComboBox>(m_pieces, row, PPos)->currentData().toInt());
307 s.pieces << p;
308 }
309 for (int row = 0; row < m_seams->rowCount(); ++row)
310 {
311 prova3d::SeamSetup m;
312 m.pieceA = cell<QComboBox>(m_seams, row, SA)->currentData().toUInt();
313 m.a1 = cell<QComboBox>(m_seams, row, SA1)->currentData().toUInt();
314 m.a2 = cell<QComboBox>(m_seams, row, SA2)->currentData().toUInt();
315 m.pieceB = cell<QComboBox>(m_seams, row, SB)->currentData().toUInt();
316 m.b1 = cell<QComboBox>(m_seams, row, SB1)->currentData().toUInt();
317 m.b2 = cell<QComboBox>(m_seams, row, SB2)->currentData().toUInt();
318 if (QWidget *holder = m_seams->cellWidget(row, SRev))
319 if (auto *rev = holder->findChild<QCheckBox *>()) m.reverse = rev->isChecked();
320 m.sides = prova3d::Sides(cell<QComboBox>(m_seams, row, SSides)->currentData().toInt());
321 if (m.a1 == 0 || m.a2 == 0 || m.b1 == 0 || m.b2 == 0) continue;
322 s.seams << m;
323 }
324 return s;
