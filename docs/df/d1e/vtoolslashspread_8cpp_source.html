85SlashSpread VToolSlashSpread::makeSpread(const VContainer *data, quint32 hinge, quint32 edgeStart, quint32 edgeEnd,
86 int slashes, qreal spread, bool percent, SlashSpread::Fixed fixed,
87 const QVector<quint32> &objects)
88{
89 const QPointF h = static_cast<QPointF>(*data->GeometricObject<VPointF>(hinge));
90 const QVector<QPointF> edge = SlashSpread::edgeOf(data, edgeStart, edgeEnd, objects);
91 return SlashSpread(h, edge, slashes, percent ? spread : qApp->toPixel(spread), percent, fixed);
92}

131//---------------------------------------------------------------------------------------------------------------------
132VToolSlashSpread *VToolSlashSpread::Create(const quint32 _id, quint32 hinge, quint32 edgeStart, quint32 edgeEnd,
133 int slashes, QString &spread, bool percent, SlashSpread::Fixed fixed,
134 const QString &suffix, const QVector<SourceItem> &source,
135 const QVector<DestinationItem> &destination, VMainGraphicsScene *scene,
136 VAbstractPattern *doc, VContainer *data, const Document &parse,
137 const Source &typeCreation)
138{
139 const qreal spreadValue = CheckFormula(_id, spread, data);
140
141 QVector<quint32> objects;
142 for (const SourceItem &item : source)
143 {
144 objects.append(item.id);
145 }
146 const SlashSpread geometry = makeSpread(data, hinge, edgeStart, edgeEnd, slashes, spreadValue, percent, fixed,
147 objects);
148
149 QVector<DestinationItem> dest = destination;
150
151 quint32 id = _id;
152 const bool fromGui = typeCreation == Source::FromGui;
153 if (fromGui)
154 {
155 dest.clear();// Try to avoid mistake, value must be empty
156 id = VContainer::getNextId();//Just reserve id for tool
157 }
158
159 for (int i = 0; i < source.size(); ++i)
160 {
161 const quint32 objectId = source.at(i).id;
162 const QSharedPointer<VGObject> obj = data->GetGObject(objectId);
163
164 if (static_cast<GOType>(obj->getType()) == GOType::Point)
165 {
166 VPointF moved = openedPoint(geometry, objectId, suffix, data);
167 moved.setIdObject(id);
168 if (fromGui)
169 {
170 DestinationItem item;
171 item.mx = moved.mx();
172 item.my = moved.my();
173 item.showPointName = moved.isShowPointName();
174 item.id = data->AddGObject(new VPointF(moved));
175 dest.append(item);
176 }
177 else
178 {
179 const DestinationItem &item = dest.at(i);
180 moved.setMx(item.mx);
181 moved.setMy(item.my);
182 moved.setShowPointName(item.showPointName);
183 data->UpdateGObject(item.id, new VPointF(moved));
184 }
185 continue;
186 }
187
188 const QSharedPointer<VAbstractCurve> curve = data->GeometricObject<VAbstractCurve>(objectId);
189 VAbstractCurve *opened = geometry.opened(curve, suffix); // the container takes it
190 opened->setIdObject(id);
191 if (fromGui)
192 {
193 DestinationItem item;
194 item.mx = INT_MAX;
195 item.my = INT_MAX;
196 item.id = data->AddGObject(opened);
197 registerCurve(data, data->GeometricObject<VAbstractCurve>(item.id), item.id);
198 dest.append(item);
199 }
200 else
201 {
202 const quint32 destId = dest.at(i).id;
203 data->UpdateGObject(destId, opened);
204 registerCurve(data, data->GeometricObject<VAbstractCurve>(destId), destId);
205 }
206 }
207
208 if (not fromGui && parse != Document::FullParse)
209 {
210 doc->UpdateToolData(id, data);
211 }
212
213 if (parse == Document::FullParse)
214 {
215 VDrawTool::AddRecord(id, Tool::SlashSpread, doc);
216 VToolSlashSpread *tool = new VToolSlashSpread(doc, data, id, hinge, edgeStart, edgeEnd, slashes, spread,
217 percent, fixed, suffix, source, dest, typeCreation);
218 scene->addItem(tool);
219 initOperationToolConnections(scene, tool);
220 VAbstractPattern::AddTool(id, tool);
221 doc->IncrementReferens(data->GeometricObject<VPointF>(hinge)->getIdTool());
222 doc->IncrementReferens(data->GeometricObject<VPointF>(edgeStart)->getIdTool());
223 doc->IncrementReferens(data->GeometricObject<VPointF>(edgeEnd)->getIdTool());
224 for (int i = 0; i < source.size(); ++i)
225 {
226 doc->IncrementReferens(data->GetGObject(source.at(i).id)->getIdTool());
227 }
228 return tool;
229 }
230 return nullptr;
231}

233//---------------------------------------------------------------------------------------------------------------------
234VPointF VToolSlashSpread::openedPoint(const SlashSpread &spread, quint32 idItem, const QString &suffix,
235 VContainer *data)
236{
237 const QSharedPointer<VPointF> point = data->GeometricObject<VPointF>(idItem);
238 const qreal angle = spread.isValid() ? spread.rotation(spread.section(static_cast<QPointF>(*point))) : 0;
239 return point->Rotate(spread.hinge(), angle, suffix);
240}

242//---------------------------------------------------------------------------------------------------------------------
243void VToolSlashSpread::registerCurve(VContainer *data, const QSharedPointer<VAbstractCurve> &curve, quint32 id)
244{
245 switch (static_cast<GOType>(curve->getType()))
246 {
247 case GOType::Arc:
248 case GOType::EllipticalArc:
249 data->AddArc(curve, id);
250 break;
251 case GOType::Spline:
252 case GOType::CubicBezier:
253 data->AddSpline(qSharedPointerDynamicCast<VAbstractBezier>(curve), id);
254 break;
255 case GOType::SplinePath:
256 case GOType::CubicBezierPath:
257 data->AddCurveWithSegments(qSharedPointerDynamicCast<VAbstractCubicBezierPath>(curve), id);
258 break;
259 default:
260 break;
261 }
262}
