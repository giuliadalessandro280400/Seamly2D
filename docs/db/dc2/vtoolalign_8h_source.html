47/**
48 * @brief The AlignTransform struct describes the rigid movement of the Align tool: a move of length/angle followed
49 * by a rotation around the target point.
50 */
51struct AlignTransform
52{
53 qreal length {0}; // in pixels
54 qreal angle {0}; // direction of the move, degrees
55 qreal rotation {0}; // rotation after the move, degrees
56 QPointF origin; // rotation origin (the target point)
57};
